\chapter{First-Order Logic}
\section{Syntax}
We now present the mathematical definition for a \keyword{first-order language} as well as its relationship
to \aref{dfn:general-mathematical-structures}{mathematical structures}. Conceptually, a first-order language
is simply a collection of symbols  together with a finite set of rules  that determine which
strings of symbols from said collection are valid expressions of that language.

\begin{dfn}
    \phantomsection \label{dfn:first-order-language}
    A \keyword{first-order language} \( \mathcal{L} \) is associated with the following symbols:
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item Logical Symbols:
            \begin{enumerate}[topsep=2pt,label=\arabic*.]
                \item Parentheses: \( () \, , \,[ \, ] \)
                \item Sentential Connectives\footnote{All the other familiar connectives can be defined in terms of \( \lnot \), \( \land \), and \( \lor \).}:
                    \( \lnot \), \( \land \), \( \lor \)
                \item Quantifiers: \( \forall \), \( \exists \)
                \item Equality(optional): \( = \)
                \item Variables: \( v_{0}, v_{1}, \dots \)
            \end{enumerate}
        \item Non-Logical Symbols:
            \begin{enumerate}[topsep=2pt,label=\arabic*.]
                \item Predicates: A tuple \( P^{\mathcal{L}} \) of predicate symbols with an arity function
                    \( \function{\Pi^{\mathcal{L}}}{\dom P^{\mathcal{L}}}{\naturalset - \mleft\{ 0 \mright\}} \) associated with it
                \item Functions: A tuple \( F^{\mathcal{L}} \) of function symbols with an arity function
                    \( \function{\Phi^{\mathcal{L}}}{\dom F^{\mathcal{L}}}{\naturalset} \) associated with it
                    \phantomsection \label{text:first-order-language-canonical-well-ordering-for-variable-symbols}
            \end{enumerate}
    \end{enumerate}
\end{dfn}

\begin{rem}
    The traditional ``constant'' symbols are treated as \( 0 \)-ary function symbols in this approach.
    That is why \( \ran \Phi^{\mathcal{L}} \) can be the entirety of \( \naturalset \) (which includes 0) whereas \( 0 \) must
    be excluded from the range of \( \Pi^{\mathcal{L}} \).
\end{rem}

\begin{dfn}
    \phantomsection \label{dfn:first-order-language-L-symbol}
    A symbol associated with \( \mathcal{L} \) is called an \( \mathcal{L} \)-symbol.
    An \( \mathcal{L} \)-symbol that is also a variable symbol is called an \( \mathcal{L} \)-variable.
\end{dfn}

\noindent
The symbols defined in \cref{dfn:first-order-language} form the vocabulary of \( \mathcal{L} \), so to speak;
each symbol in \( \mathcal{L} \) is analogous to a word\footnote{\,Technically, the parentheses are more like punctuations.} in a natural language.
Just as the words of a natural language can be stringed together to form expressions,
the symbols of \( \mathcal{L} \) can also be stringed together to form \emph{expressions} in the language of \( \mathcal{L} \), or
\( \mathcal{L} \)-expressions for short.

\begin{dfn}
    \phantomsection \label{dfn:first-order-language-expression}
    An \( \mathcal{L} \)-expression is any finite sequence of \( \mathcal{L} \)-symbols.
\end{dfn}

\begin{notation}
    \phantomsection \label{notation:first-order-language-the-set-of-all-symbols-constituting-an-expression}
    If \( \varepsilon \) is an \( \mathcal{L} \)-expression, then \( \folsymbols{\varepsilon} \) denotes the set of all
    the \( \mathcal{L} \)-symbols consituting \( \varepsilon \).
\end{notation}

\begin{dfn}
    \phantomsection \label{dfn:first-order-language-the-set-of-all-variable-symbols-in-a-first-order-language}
    If \( \mathcal{L} \) is a first-order language, then \( \folvars{\mathcal{L}} \) denotes the (countable and non-empty) set of all \( \mathcal{L} \)-variables
    associated with \( \mathcal{L} \).
\end{dfn}

\begin{notation}
    \phantomsection \label{notation:first-order-language-the-set-of-all-variable-symbols-in-an-expression}
    If \( \varepsilon \) is an \( \mathcal{L} \)-expression, then \( \folvars{\varepsilon} \) denotes the set of all \( \mathcal{L} \)-variables
    appearing in \( \varepsilon \).
\end{notation}

\begin{cor}
    It follows then that for any \( \mathcal{L} \)-expression \( \varepsilon \), \( \folvars{\varepsilon} \subseteq \folsymbols{\varepsilon} \)
    and \( \folvars{\varepsilon} \subseteq \folvars{\mathcal{L}} \).
\end{cor}

\noindent
Of course, just having an infinite supply of expressions is not enough to fully define a language. We also need a set of rules to tell us
which expressions are valid and which are not.
For instance, the words `cats', `or', and `to' are all valid words of the English language; however, no native English speaker would deem the sequence
of words `to or cats' a valid English expression for the simple reason that `to or cats' does not conform to the grammar of the English language.
In other words, the grammar \emph{is} part of the language and should therefore be included in the language's specification.
\begin{equation*}
    \text{language} = \text{vocabulary} + \text{grammar}
\end{equation*}
\noindent
Fortunately, the syntax (grammar) of first-order languages is relatively straightforward.
In first-order languages, the most basic unit of grammar is formulas---analogous to
clauses in natural languages. But just as clauses are composed of existing words (verbs, nouns, etc.) in a natural language,
formulas are also composed of smaller units called \emph{terms}.
As such, any comprehensive definition of the grammar of a first-order language must begin with a precise definition of what a `term' means in that language.

\begin{dfn}[Terms]
    \phantomsection \label{dfn:first-order-language-construction-of-terms}
    An \aref{dfn:first-order-language-expression}{\( \mathcal{L} \)-expression} is an \( \mathcal{L} \)-term if and only if
    it is constructible in a finite number of steps via the application of the following two rules:
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item If \( t \) is an \( \mathcal{L} \)-expression such that \( \folsymbols{t} = \mleft\{ x \mright\} \) where \( x \in \folvars{\mathcal{L}} \),
             then \( t \) is an \( \mathcal{L} \)-term
        \item If \( \tau \) is a tuple of \( \mathcal{L} \)-terms such that \( \dom \tau = \Phi^{\mathcal{L}}(i) \) for some \( i \in \dom F^{\mathcal{L}} \),
             then \( F^{\mathcal{L}}_{i}(\tau) \) is also an \( \mathcal{L} \)-term
    \end{enumerate}
\end{dfn}

\begin{dfn}[Complexity]
    \phantomsection \label{dfn:first-order-language-complexity-of-terms}
    If \( t \) is an \( \mathcal{L} \)-term, then the \keyword{complexity} of \( t \) (denoted as \( \folcomplexityofterm{t} \))
    is the number of occurrences of function symbols with non-zero arity in \( t \).
\end{dfn}

\begin{rem}
    Note that the complexity of a term counts the number of \emph{occurrences} of function symbols with zero arity, not just the number of such function symbols occurring
    in \( t \). For instance, a term \( F(G(x), G(y)) \) has a complexity of \( 3 \) not \( 2 \) even though only two function symbols occurred in it in total.
    The function symbol \( G \) needs to be counted twice.
\end{rem}

\begin{cor}
    \phantomsection \label{cor:first-order-language-every-term-has-finite-complexity}
    If \( t \) is an \( \mathcal{L} \)-term, then \( \folcomplexityofterm{t} \in \naturalset \).
\end{cor}

\begin{cor}
    \phantomsection \label{cor:first-order-language-terms-with-non-zero-complexity-must-have-a-parent-function-symbol}
    If \( t \) is an \( \mathcal{L} \)-term and \( \folcomplexityofterm{t} \neq 0 \), then \( t \) must have the form
    \( F^{\mathcal{L}}_{i}(\tau) \) for some \( i \in \dom F^{\mathcal{L}} \) where \( \tau \) is a tuple of \( \mathcal{L} \)-terms
    and \( \abs{\tau} = \Phi^{\mathcal{L}}(i) \neq 0 \).
\end{cor}

\begin{eg}
    Thus, in a first-order language \( \mathcal{L} \) with function symbols \( F \), \( G \), and \( H \) (with arities 0, 1, and 2 respectively) and variable symbols
    \( x \), \( y \), and \( z \), we have the following:
    \begin{align*}
        \folcomplexityofterm{x} &= 0\\
        \folcomplexityofterm{F()} &= 0\\
        \folcomplexityofterm{G(x)} &= 1\\
        \folcomplexityofterm{H(F(), y)} &= 1\\
        \folcomplexityofterm{H(G(x), z)} &= 2\\
        \folcomplexityofterm{H(H(x, F()), y)} &= 2
    \end{align*}
\end{eg}

\begin{dfn}[Subterms]
    \phantomsection \label{dfn:first-order-language-subterms}
    Let \( s \) and \( t \) denote \( \mathcal{L} \)-terms.
    Then we say that \( s \) is a \keyword{subterm} of \( t \) (denoted as \( s \folsubterm t \))
    if and only if the following conditions are met:
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item \( t \) has the form \( F^{\mathcal{L}}_{i}(\tau) \) for some \( i \in \dom F^{\mathcal{L}} \) and \( \tau \) is a tuple of \( \mathcal{L} \)-terms
            such that \( \abs{\tau} = \Phi^{\mathcal{L}}(i) \neq 0 \)
        \item \( s \in \ran \tau \) or \( s \folsubterm r \) for some \( r \in \ran \tau \)
    \end{enumerate}
\end{dfn}

\begin{cor}
    \phantomsection \label{cor:first-order-language-the-complexity-of-a-subterm-is-less-than-the-complexity-of-the-parent-term}
    If \( t \) is an \( \mathcal{L} \)-term and \( s \) is a subterm of \( t \), then \( \folcomplexityofterm{s} < \folcomplexityofterm{t} \).
\end{cor}

\begin{dfn}[Atoms]
    \phantomsection \label{dfn:first-order-language-atom}
    An \( \mathcal{L} \)-expression \( \varphi \) is an \keyword{atom} of \( \mathcal{L} \) if and only if
    \( \varphi \) is of either one of the following forms:
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item \( t_{0} = t_{1} \) where \( t_{0} \) and \( t_{1} \) are \aref{dfn:first-order-language-construction-of-terms}{\( \mathcal{L} \)-terms}
        \item \( P^{\mathcal{L}}_{i}(\tau) \) for some \( i \in \dom P^{\mathcal{L}} \) where
            \( \tau \) is a tuple of \( \mathcal{L} \)-terms such that \( \abs{\tau} = \Pi^{\mathcal{L}}(i) \)
    \end{enumerate}
\end{dfn}

\begin{dfn}[Formulas]
    \phantomsection \label{dfn:first-order-language-construction-of-formulas}
    An \( \mathcal{L} \)-expression is an \( \mathcal{L} \)-formula\footnote{\,Also called a \keyword{well-formed formula} or wff in \( \mathcal{L} \).}
    if and only if it is constructible from an atom in a finite number of steps via the application of the following rules:
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item If \( \varphi \) is an atom of \( \mathcal{L} \), then \( \varphi \) is an \( \mathcal{L} \)-formula
        \item If \( \varphi \) is an \( \mathcal{L} \)-formula, then \( \mleft( \varphi \mright) \) is also an \( \mathcal{L} \)-formula
        \item If \( \varphi \) is an \( \mathcal{L} \)-formula, then \( \mleft[ \, \varphi \, \mright] \) is also an \( \mathcal{L} \)-formula
        \item If \( \varphi \) is an \( \mathcal{L} \)-formula, then \( \lnot \, \varphi \) is also an \( \mathcal{L} \)-formula
        \item If \( \varphi \) and \( \psi \) are \( \mathcal{L} \)-formulas, then \( \varphi \,\,{\land}\,\, \psi \) is also an \( \mathcal{L} \)-formula
        \item If \( \varphi \) and \( \psi \) are \( \mathcal{L} \)-formulas, then \( \varphi \,\,{\lor}\,\, \psi \) is also an \( \mathcal{L} \)-formula
        \item If \( \varphi \) is an \( \mathcal{L} \)-formula and \( v \) is any \( \mathcal{L} \)-variable,
            then \( \forall v \, \varphi \) is also an \( \mathcal{L} \)-formula
        \item If \( \varphi \) is an \( \mathcal{L} \)-formula and \( v \) is any \( \mathcal{L} \)-variable,
            then \( \exists v \, \varphi \) is also an \( \mathcal{L} \)-formula
    \end{enumerate}
\end{dfn}

\begin{notation}
    The familiar abbreviations from propositional logic do apply here, of course. For instance, if \( \varphi \) and \( \psi \) are
    \( \mathcal{L} \)-formulas, then we write \( \varphi {\implies} \psi \) to abbreviate \( \lnot \, \varphi \,\,{\lor}\,\, \psi \),
    \( \varphi {\iff} \psi \) to abbreviate \( \mleft( \varphi {\implies} \psi \mright) \,\,{\land}\,\, \mleft( \psi {\implies} \varphi \mright)  \), and so on.
\end{notation}

\begin{dfn}[Syntactic Operations]
    \phantomsection \label{dfn:first-order-language-syntactic-operations}
    If \( \mathcal{L} \) is a first-order language, then \( \mathcal{R}_{\ast} \) denotes an operation that can be applied to formulas of \( \mathcal{L} \)
    to construct a new formulas of \( \mathcal{L} \).
    The operation \( \mathcal{R}_{\ast} \) is also called a \keyword{syntactic operation} of \( \mathcal{L} \).
\end{dfn}

\begin{rem}
    Every first-order language has the same set of syntactic operations: \( \mathcal{R}_{(} \), \( \mathcal{R}_{[} \),
    \( \mathcal{R}_{\lnot} \), \( \mathcal{R}_{\land} \), \( \mathcal{R}_{\lor} \), \( \mathcal{R}_{\forall} \), and \( \mathcal{R}_{\exists} \).
\end{rem}

\begin{eg}
    If \( \varphi \) and \( \psi \) are \( \mathcal{L} \)-formulas, then \( \varphi \,\,{\land}\,\, \psi \) is the formula obtained by applying
    the syntactic operation \( \mathcal{R}_{\land} \) to \( \varphi \) and \( \psi \), \( \varphi \,\,{\lor}\,\, \psi \) is the formula obtained
    by applying the syntactic operation \( \mathcal{R}_{\lor} \) to \( \varphi \) and \( \psi \), etc.
\end{eg}

\begin{dfn}[Complexity]
    \phantomsection \label{dfn:first-order-language-complexity-of-formulas}
    If \( \varphi \) is an \( \mathcal{L} \)-formula, then the \keyword{complexity} of \( \varphi \) (denoted as \( \folcomplexityofformula{\varphi} \))
    is the total number of syntactic operations required to construct \( \varphi \) from atoms of \( \mathcal{L} \).
    More precisely:
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item If \( \varphi \coloneq \mleft( \psi \mright) \), then \( \folcomplexityofformula{\varphi} = \folcomplexityofformula{\psi} + 1 \)
        \item If \( \varphi \coloneq \mleft[ \,\psi\, \mright] \), then \( \folcomplexityofformula{\varphi} = \folcomplexityofformula{\psi} + 1 \)
        \item If \( \varphi \coloneq \lnot \, \psi \), then \( \folcomplexityofformula{\varphi} = \folcomplexityofformula{\psi} + 1 \)
        \item If \( \varphi \coloneq \psi \,\,{\land}\,\, \phi \), then \( \folcomplexityofformula{\varphi} = \folcomplexityofformula{\psi} + \folcomplexityofformula{\phi} \)
        \item If \( \varphi \coloneq \psi \,\,{\lor}\,\, \phi \), then \( \folcomplexityofformula{\varphi} = \folcomplexityofformula{\psi} + \folcomplexityofformula{\phi} \)
        \item If \( \varphi \coloneq \forall x \, \psi \), then \( \folcomplexityofformula{\varphi} = \folcomplexityofformula{\psi} + 1 \)
        \item If \( \varphi \coloneq \exists x \, \psi \), then \( \folcomplexityofformula{\varphi} = \folcomplexityofformula{\psi} + 1 \)
    \end{enumerate}
\end{dfn}

\begin{eg}
    If \( \varphi \) and \( \psi \) are atoms of \( \mathcal{L} \) and \( x \) is an \( \mathcal{L} \)-variable, then we have the following:
    \begin{align*}
        \folcomplexityofformula{\varphi} &= 0\\
        \folcomplexityofformula{\forall x \, \varphi} &= 1\\
        \folcomplexityofformula{\varphi \,\,{\land}\,\, \psi} &= 1\\
        \folcomplexityofformula{\lnot \, \varphi \,\,{\land}\,\, \psi} &= 2\\
        \folcomplexityofformula{\varphi \,\,{\lor}\,\, \psi \,\,{\land}\,\, \psi} &= 2\\
        \folcomplexityofformula{\mleft( \varphi \,\,{\lor}\,\, \psi \mright) \,\,{\land}\,\, \psi} &= 3
    \end{align*}
\end{eg}

\begin{cor}
    \phantomsection \label{cor:first-order-language-every-formula-has-finite-complexity}
    It follows immediately from \cref{dfn:first-order-language-construction-of-formulas} that every \( \mathcal{L} \)-formula can be constructed
    from \aref{dfn:first-order-language-atom}{atoms of \( \mathcal{L} \)} via the application of a finite number of
    \aref{dfn:first-order-language-syntactic-operations}{syntactic operations}; that is, for every \( \mathcal{L} \)-formula \( \varphi \),
    \( \folcomplexityofformula{\varphi} \in \naturalset \).
\end{cor}

\begin{cor}
    \phantomsection \label{cor:first-order-language-only-atoms-can-have-zero-complexity}
    If \( \varphi \) is an \( \mathcal{L} \)-formula, then \( \folcomplexityofformula{\varphi} = 0 \) iff \( \varphi \) is an atom.
\end{cor}

\begin{cor}
    \phantomsection \label{cor:first-order-language-every-formula-with-non-zero-complexity-can-be-decomposed}
    If \( \varphi \) is an \( \mathcal{L} \)-formula such that \( \folcomplexityofformula{\varphi} \neq 0 \), then exactly one of the following must be true:
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item \( \varphi \coloneq \mleft( \psi \mright) \) where \( \psi \) is an \( \mathcal{L} \)-formula
        \item \( \varphi \coloneq \mleft[ \, \psi \, \mright] \) where \( \psi \) is an \( \mathcal{L} \)-formula
        \item \( \varphi \coloneq \lnot \, \psi \) where \( \psi \) is an \( \mathcal{L} \)-formula
        \item \( \varphi \coloneq \psi \,\,{\land}\,\, \phi \) or \( \varphi \coloneq \phi \,\,{\land}\,\, \psi \)
            where \( \psi \) and \( \phi \) are \( \mathcal{L} \)-formulas
        \item \( \varphi \coloneq \psi \,\,{\lor}\,\, \phi \) or \( \varphi \coloneq \phi \,\,{\lor}\,\, \psi \) where \( \psi \) and \( \phi \)
            are \( \mathcal{L} \)-formulas
        \item \( \varphi \coloneq \forall x \, \psi \) where \( \psi \) is an \( \mathcal{L} \)-formula and \( x \) is an \( \mathcal{L} \)-variable
        \item \( \varphi \coloneq \exists x \, \psi \) where \( \psi \) is an \( \mathcal{L} \)-formula and \( x \) is an \( \mathcal{L} \)-variable
    \end{enumerate}
\end{cor}

\begin{dfn}[Subformulas]
    \phantomsection \label{dfn:first-order-language-subformulas}
    If \( \varphi \) and \( \psi \) are \( \mathcal{L} \)-formulas, then
    we say that \( \psi \) is a \keyword{subformula} of \( \varphi \) (denoted as \( \psi \folsubformula \varphi \))
    if and only if one of the following conditions is met:
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item \( \varphi \coloneq \mleft( \psi \mright) \)
        \item \( \varphi \coloneq \mleft[ \, \psi \, \mright] \)
        \item \( \varphi \coloneq \lnot \, \psi \),
        \item \( \varphi \coloneq \phi \,\,{\land}\,\, \psi \) or \( \varphi \coloneq \psi \,\,{\land}\,\, \phi \) for some \( \mathcal{L} \)-formula \( \phi \)
        \item \( \varphi \coloneq \phi \,\,{\lor}\,\, \psi \) or \( \varphi \coloneq \psi \,\,{\lor}\,\, \phi\) for some \( \mathcal{L} \)-formula \( \phi \)
        \item \( \varphi \coloneq \forall x \, \psi \) for some \( \mathcal{L} \)-variable \( x \)
        \item \( \varphi \coloneq \exists x \, \psi \) for some \( \mathcal{L} \)-variable \( x \)
        \item \( \psi \folsubformula \phi \) for some \( \mathcal{L} \)-formula \( \phi \) such that \( \phi \folsubformula \varphi \)
    \end{enumerate}
\end{dfn}

\begin{cor}
    \phantomsection \label{cor:first-order-language-the-complexity-of-a-subformula-is-less-than-the-complexity-of-the-formula}
    If \( \psi \folsubformula \varphi \), then \( \folcomplexityofformula{\psi} < \folcomplexityofformula{\varphi} \).
\end{cor}

\begin{rem}
    The converse of \cref{cor:first-order-language-the-complexity-of-a-subformula-is-less-than-the-complexity-of-the-formula} is obviously false.
\end{rem}

\begin{notation}
    If \( \varphi \) is an \( \mathcal{L} \)-formula and \( \psi \folsubformula \varphi \), then \( \folsubformulasubstitution{\varphi}{\psi}{\phi} \)
    denotes the \( \mathcal{L} \)-formula obtained by replacing all occurrences of \( \psi \) in \( \varphi \) with \( \phi \).
\end{notation}

\begin{dfn}
    \phantomsection \label{dfn:first-order-language-variable-occurrence-in-formulas}
    If \( \varphi \) is an \( \mathcal{L} \)-formula and \( v \) is an \( \mathcal{L} \)-variable, then we say that ``\( v \) occurs in \( \varphi \),'' 
    ``\( v \) is in \( \varphi \),'' or ``\( \varphi \) contains \( v \)'' if and only if \( v \in \folvars{\varphi} \).
\end{dfn}

\begin{dfn}[Free Variables]
    \phantomsection \label{dfn:first-order-language-recursive-definition-of-free-variables}
    If \( \varphi \) and \( \psi \) are \( \mathcal{L} \)-formulas, \( x \) and \( v \) are \( \mathcal{L} \)-variables, and \( \tau \) is a tuple of \( \mathcal{L} \)-variables,
    then the following holds:
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item If \( \varphi \) is an atom of \( \mathcal{L} \), then \( x \) is a free variable of \( \varphi \) if and only if \( x \in \folvars{\varphi} \)
        \item \( x \) is a free variable of \( \lnot \, \varphi \) if and only if \( x \) is a free variable of \( \varphi \)
        \item \( x \) is a free variable of \( \varphi \,\,{\land}\,\, \psi \) if and only if \( x \) is a free variable of \( \varphi \) or \( \psi \)
        \item \( x \) is a free variable of \( \varphi \,\,{\lor}\,\, \psi \) if and only if \( x \) is a free variable of \( \varphi \) or \( \psi \)
        \item \( x \) is a free variable of \( \forall v \, \varphi \) if and only if \( x \) is a free variable of \( \varphi \) and \( x \neq v \)
        \item \( x \) is a free variable of \( \exists v \, \varphi \) if and only if \( x \) is a free variable of \( \varphi \) and \( x \neq v \)
    \end{enumerate}
\end{dfn}

\begin{dfn}[Bound Variables]
    \phantomsection \label{dfn:first-order-language-bound-variables}
    An \( \mathcal{L} \)-variable occurring in \( \varphi \) that is not free is called a \keyword{bound variable} of \( \varphi \). 
\end{dfn}

\begin{rem}
    If \( x \) is a free variable of \( \varphi \), we also say that \( x \) \keyword{occurs freely} or \keyword{occurs free} in \( \varphi \).
\end{rem}

\begin{notation}
    \phantomsection \label{notation:first-order-language-the-set-of-all-free-variables-in-a-formula}
    If \( \varphi \) is an \( \mathcal{L} \)-formula, then \( \folfreevars{\varphi} \) denotes the set of all free variables in \( \varphi \).
\end{notation}

\section{Semantics}
\noindent
The syntactic rules of \( \mathcal{L} \) are fully defined in the previous section;
that is, given any finite string of symbols in \aref{dfn:first-order-language}{\( \mathcal{L} \)}, we
now have the information at our disposal to determine whether that string of symbols is a valid formula of \( \mathcal{L} \).
Nevertheless, our description of \( \mathcal{L} \) is far from complete. For any language to be useful, having a bunch of symbols
and a set of rules to construct valid formulas using those symbols is not enough; there also needs to be a set of rules that allows
\emph{meaning} to be assigned to all valid formulas---just as English words like ``cats'', ``apples'', and ``oranges'' need to have meaning
for English to have any value at all as a language.

\vspace{0.3cm}
\noindent
The assignment of meaning to the expressions of a formal language is called the \keyword{semantics} of a formal language. This typically involves
associating the symbols in the language (and the valid expressions constructed using those symbols) with existing concepts or mental imageries.
These mental imageries can be either concrete or abstract. For instance, when a native English speaker sees the word ``apple,'' their minds
construct the image of a fruit with red skin. In other words, the semantics of a language allows the language to be \keyword{interpreted};
it provides a connection between a string of seemingly meaningless symbols and concepts from the real world.

\vspace{0.3cm}
\noindent
In first-order languages, meaning is assigned to formulas via a special class of \aref{dfn:general-mathematical-structures}{mathematical structures}
known as \keyword{first-order structures}.

\begin{dfn}
    A \keyword{first-order structure} is any \aref{dfn:general-mathematical-structures}{mathematical structure} \( \mathfrak{A} \) such that
    \aref{notation:individual-components-of-mathematical-structures}{\( S^{\mathfrak{A}} = \emptyset \)}
    and \aref{notation:individual-components-of-mathematical-structures}{\( G^{\mathfrak{A}} = \emptyset \)}.
    In other words, a first-order structure is any mathematical structure that is endowed with no other features
    aside from n-ary \aref{dfn:n-ary-relation-on-a-set}{relations} and \aref{dfn:n-ary-operation}{operations}.
\end{dfn}

\noindent
Of course, not every first-order structure can be used to assign meaning to \( \mathcal{L} \)-formulas.
In fact, for a first-order structure to be capable of interpreting formulas of \( \mathcal{L} \),
its \aref{dfn:signature-of-a-structure}{signature} must be compatible with the arity functions
\aref{dfn:first-order-language}{\( \Pi^{\mathcal{L}} \) and \( \Phi^{\mathcal{L}} \)},
which leads us to the following definition.

\begin{dfn}
    An \( \mathcal{L} \)-structure is any first-order structure with \aref{dfn:signature-of-a-structure}{signature}
    \( \tuple{\tuple{}, \Pi^{\mathcal{L}}, \Phi^{\mathcal{L}}, \tuple{}} \).
\end{dfn}

\begin{rem}
    \phantomsection \label{one-to-one-correspondences-between-first-order-language-and-first-order-structure}
    Note that \( F^{\mathcal{L}} \), \( P^{\mathcal{L}} \), \( F^{\mathfrak{A}} \), and \( R^{\mathfrak{A}} \) are all bijective tuples
    because the symbols associated with \( \mathcal{L} \) and \( \mathfrak{A} \) are bound to be unique.
    As a result, \( \composition{F^{\mathfrak{A}}}{\mleft( F^{\mathcal{L}} \mright)^{-1}} \) and
    \( \composition{R^{\mathfrak{A}}}{\mleft( P^{\mathcal{L}} \mright)^{-1}} \) are both bijections.
    In other words, there exist one-to-one correspondences between the function symbols of \( \mathcal{L} \) and the n-ary operations of \( \mathfrak{A} \)
    as well as between the predicate symbols of \( \mathcal{L} \) and the n-ary relation of \( \mathfrak{A} \).
\end{rem}

\noindent
The stage is now set for the semantics of \( \mathcal{L} \) to be defined.

\begin{dfn}
    \phantomsection \label{dfn:first-order-language-variable-assignment-function}
    If \( \mathfrak{A} \) is an \( \mathcal{L} \)-structure, then a function \( \mathfrak{a} \) is a \keyword{variable assignment function}
    of \( \mathcal{L} \) under \( \mathfrak{A} \) if and only if \( \function{\mathfrak{a}}{\folvars{\mathcal{L}}}{\abs{\mathfrak{A}}} \).
    In other words, any function on \( \folvars{\mathcal{L}} \) into \( \abs{\mathfrak{A}} \) is a variable assignment function of \( \mathcal{L} \) under \( \mathfrak{A} \).
\end{dfn}

\begin{dfn}
    \phantomsection \label{dfn:first-order-language-modified-variable-assignment-function}
    Let \( \mathfrak{A} \) be an \( \mathcal{L} \)-structure and \( \function{\mathfrak{a}}{\folvars{\mathcal{L}}}{\abs{\mathfrak{A}}} \) be a variable assignment function of
    \( \mathcal{L} \) under \( \mathfrak{A} \).
    If \( v \) is an \( \mathcal{L} \)-variable and \( e \in \abs{\mathfrak{A}} \), then \( \folmodifiedvariableassignmentfunction{\mathfrak{a}}{v}{e} \) denotes
    the function \( \restrictedfunction{\mathfrak{a}}{ \folvars{\mathcal{L}} - \mleft\{ v \mright\} } \union \big\{ \orderedpair{v}{e} \big\} \).
\end{dfn}

\begin{rem}
    Note that \( \folmodifiedvariableassignmentfunction{\mathfrak{a}}{v}{e} \) is still a variable assignment function because
    \( \dom \big( \folmodifiedvariableassignmentfunction{\mathfrak{a}}{v}{e} \big) = \folvars{\mathcal{L}} \)
    and \( \ran \big( \folmodifiedvariableassignmentfunction{\mathfrak{a}}{v}{e} \big) \subseteq \abs{\mathfrak{A}} \).
\end{rem}

\begin{thm}
    \phantomsection \label{thm:first-order-language-existence-of-variable-assignment-function}
    For any first-order language \( \mathcal{L} \) and any \( \mathcal{L} \)-structure \( \mathfrak{A} \),
    a variable assignment function of \( \mathcal{L} \) under \( \mathfrak{A} \) exists.
\end{thm}

\begin{proof}
    Let \( \mathfrak{A} \) be an arbitrary \( \mathcal{L} \)-structure.
    Since \( \folvars{\mathcal{L}} \) and \( \abs{\mathfrak{A}} \) are non-empty sets,
    it is easy to define a function on \( \folvars{\mathcal{L}} \) into \( \abs{\mathfrak{A}} \).
    E.g., given an arbitrary element \( \alpha \) of \( \abs{\mathfrak{A}} \),
    let \( \mathfrak{a} \) be a function such that \( \dom \mathfrak{a} = \folvars{\mathcal{L}} \) and \( \mathfrak{a}(x) = \alpha \) for all \( x \in \folvars{\mathcal{L}} \).
    By \cref{dfn:first-order-language-variable-assignment-function}, \( \mathfrak{a} \) is a variable assignment function of \( \mathcal{L} \) under \( \mathfrak{A} \).
\end{proof}

\begin{dfn}
    \phantomsection \label{dfn:first-order-language-interpretation-of-terms}
    If \( t \) is an \( \mathcal{L} \)-term and \( \mathfrak{a} \) is a variable assignment function of \( \mathcal{L} \) under some \( \mathcal{L} \)-structure \( \mathfrak{A} \),
    then an \keyword{interpretation} of \( t \) in \( \mathfrak{A} \) with \( \mathfrak{a} \)
    (denoted as \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}}) is defined as follows:
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item If \( t \) consists of a single \( \mathcal{L} \)-variable \( x \), then \( \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}} = \mathfrak{a}(x) \)
        \item If \( t \) has the form \( F^{\mathcal{L}}_{i}(\tau) \) for some \( i \in \dom F^{\mathcal{L}} \) and \( \tau \) is a tuple of \( \mathcal{L} \)-terms
            such that \( \dom \tau = \Phi^{\mathcal{L}}(i) \),
            then \( \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}} = 
            F^{\mathfrak{A}}_{i} \sequence{\folterminterpretation{\tau_{k}}{\mathfrak{A}}{\mathfrak{a}}}{k \in \dom \tau}\)
    \end{enumerate}
\end{dfn}

\begin{dfn}[Satisfaction]
    \phantomsection \label{dfn:first-order-language-satisfaction-of-formulas}
    Let \( \varphi \) be an \( \mathcal{L} \)-formula and \( \mathfrak{a} \) be a \aref{dfn:first-order-language-variable-assignment-function}{variable assignment function}
    of \( \mathcal{L} \) under an \( \mathcal{L} \)-structure \( \mathfrak{A} \).
    We now define precisely what it means for \( \mathfrak{A} \) to satisfy \( \varphi \) with \( \mathfrak{a} \)
    (denoted as \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}}):
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item If \( \varphi \) is an atom of the form \( t_{0} = t_{1} \) where \( t_{0} \) and \( t_{1} \) are \( \mathcal{L} \)-terms,
            then \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \) if and only if
            \( \folterminterpretation{t_{0}}{\mathfrak{A}}{\mathfrak{a}} = \folterminterpretation{t_{1}}{\mathfrak{A}}{\mathfrak{a}} \)
        \item If \( \varphi \) is an atom of the form \( P^{\mathcal{L}}_{i}(\tau) \) for some \( i \in \dom P^{\mathcal{L}} \), then
            \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \) if and only if
            \( \sequence{ \folterminterpretation{\tau_{j}}{\mathfrak{A}}{\mathfrak{a}} }{j \in \dom \tau} \in P^{\mathfrak{A}}_{i} \)
        \item If \( \varphi \) has the form \( \mleft( \psi \mright) \) or \( \mleft[ \,\psi\, \mright] \) where \( \psi \) is another \( \mathcal{L} \)-formula,
            then \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \) if and only if \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \)
        \item If \( \varphi \) has the form \( \lnot \, \psi \) where \( \psi \) is another \( \mathcal{L} \)-formula, then
            \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \) if and only if \( \folwffnonsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \)
        \item If \( \varphi \) has the form \( \psi \,\,{\land}\,\, \phi \) where \( \psi \) and \( \phi \) are \( \mathcal{L} \)-formulas,
            then \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \) if and only if \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \)
            and \( \folwffsatisfaction{\mathfrak{A}}{\phi}{\mathfrak{a}} \)
        \item If \( \varphi \) has the form \( \psi \,\,{\lor}\,\, \phi \) where \( \psi \) and \( \phi \) are \( \mathcal{L} \)-formulas,
            then \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \) if and only if \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \)
            or \( \folwffsatisfaction{\mathfrak{A}}{\phi}{\mathfrak{a}} \)
        \item If \( \varphi \) has the form \( \forall v \, \psi \) where \( v \) is an \( \mathcal{L} \)-variable and \( \psi \) is an \( \mathcal{L} \)-formula,
            then \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \) if and only if
            \( \folwffsatisfaction{\mathfrak{A}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{v}{e} } \) for all \( e \in \abs{\mathfrak{A}} \)
        \item If \( \varphi \) has the form \( \exists v \, \psi \) where \( v \) is an \( \mathcal{L} \)-variable and \( \psi \) is an \( \mathcal{L} \)-formula,
            then \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \) if and only if there exists an \( e \in \abs{\mathfrak{A}} \) such that
            \( \folwffsatisfaction{\mathfrak{A}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{v}{e} } \)
    \end{enumerate}
    The tuple \( \tuple{\mathfrak{A}, \mathfrak{a}} \) is called an \keyword{interpretation} of \( \varphi \) since it contains all the information needed to
    interpret \( \varphi \).
\end{dfn}



\begin{lem}
    \phantomsection \label{lem:first-order-language-different-variable-assignment-functions-yield-the-same-interpretation-if-they-agree-for-all-variable-symbols-in-a-term}
    Let \( t \) be an \( \mathcal{L} \)-term and \( \mathfrak{A} \) be an \( \mathcal{L} \)-structure.
    If \( \mathfrak{a}_{0} \) and \( \mathfrak{a}_{1} \) are \aref{dfn:first-order-language-variable-assignment-function}{variable assignment functions}
    of \( \mathcal{L} \) under \( \mathfrak{A} \) such that
    \( \restrictedfunctionsmall{\mathfrak{a}_{0}}{\folvars{t}} = \restrictedfunctionsmall{\mathfrak{a}_{1}}{\folvars{t}} \),
    then \( \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{0}} = \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{1}} \).
\end{lem}

\begin{proof}
    Since every \( \mathcal{L} \)-term has a finite \aref{dfn:first-order-language-complexity-of-terms}{complexity}, we can use an inductive proof.
    We begin by defining the statement \( P(n) \) as follows: for all \( \mathcal{L} \)-term \( t \) with \( \folcomplexityofterm{t} = n \),
    \( \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{0}} = \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{1}} \) holds.
    We will now state the induction hypothesis for an arbitrary \( n \in \naturalset \).

    \vspace{0.3cm}
    \noindent
    \underline{\scshape Induction Hypothesis:}
    \( P(k) \) holds for all \( k < n \).

    \vspace{0.3cm}
    \noindent
    Let \( t \) be any \( \mathcal{L} \)-term such that \( \folcomplexityofterm{t} = n \).
    Since \( n \in \naturalset \), either \( n = 0 \) or \( n \neq 0 \).
    We have to show that \( \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{0}} = \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{1}} \) is implied in both cases.

    \vspace{0.3cm}
    \noindent
    {\scshape Case} \( n = 0 \):
    Clearly \( t \) consists of either a single \( \mathcal{L} \)-variable or a 0-ary function symbol.
    If \( t \) consists of the variable symbol \( x \), then we have \( \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{0}} = \mathfrak{a}_{0}(x) \)
    and \( \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{1}} = \mathfrak{a}_{1}(x) \) by \cref{dfn:first-order-language-interpretation-of-terms}.
    Since \( x \in \folvars{t} \), necessarily \( \mathfrak{a}_{0}(x) = \mathfrak{a}_{1}(x) \) by our initial assumption.
    Thus, \( \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{0}} = \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{1}} \).
    If \( t \) consists of a 0-ary function symbol \( F^{\mathcal{L}}_{i}() \), then \cref{dfn:first-order-language-interpretation-of-terms}
    requires that \( \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{0}} = F^{\mathfrak{A}}_{i}() = \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{1}} \).

    \vspace{0.3cm}
    \noindent
    {\scshape Case} \( n \neq 0 \):
    It follows from \cref{cor:first-order-language-terms-with-non-zero-complexity-must-have-a-parent-function-symbol} that
    \( t \) has the form \( F^{\mathcal{L}}_{i}(\tau) \) for some \( i \in \dom F^{\mathcal{L}} \) where \( \tau \) is a tuple of \( \mathcal{L} \)-terms
    and \( \abs{\tau} = \Phi^{L}(i) \). Thus we have by \cref{dfn:first-order-language-interpretation-of-terms}
    \begin{align}
        \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{0}}
        &= F^{\mathfrak{A}}_{i} \sequence{ \folterminterpretation{\tau_{j}}{\mathfrak{A}}{\mathfrak{a}_{0}} }{j \in \dom \tau} \phantomsection \label{eq:interpret-with-delta0}\\
        \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{1}}
        &= F^{\mathfrak{A}}_{i} \sequence{ \folterminterpretation{\tau_{j}}{\mathfrak{A}}{\mathfrak{a}_{1}} }{j \in \dom \tau} \phantomsection \label{eq:interpret-with-delta1}
    \end{align}
    Let \( m \in \dom \tau \) be arbitrary. It follows then that \( \tau_{m} \) is a \aref{dfn:first-order-language-subterms}{subterm} of \( t \); thus,
    \cref{cor:first-order-language-the-complexity-of-a-subterm-is-less-than-the-complexity-of-the-parent-term} guarantees that \( \folcomplexityofterm{\tau_{m}} < n \).
    By the induction hypothesis, we further deduce that
    \( \folterminterpretation{\tau_{m}}{\mathfrak{A}}{\mathfrak{a}_{0}} = \folterminterpretation{\tau_{m}}{\mathfrak{A}}{\mathfrak{a}_{1}} \).
    This means that the tuples in \eqref{eq:interpret-with-delta0} and \eqref{eq:interpret-with-delta1} are equal and that
    \( \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{0}} = \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{1}} \).

    \vspace{0.3cm}
    \noindent
    Evidently, \( \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{0}} = \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{1}} \) is true in all possible cases.
    From this we arrive at the conclusion that \( P(n) \) is implied by the induction hypothesis, which completes the inductive proof.
\end{proof}

\begin{thm}
    \phantomsection \label{thm:first-order-language-different-variable-assignment-functions-yield-the-same-interpretation-if-they-agree-on-the-free-variables-of-a-formula}
    Let \( \varphi \) be an \( \mathcal{L} \)-formula and \( \mathfrak{A} \) be an \( \mathcal{L} \)-structure.
    If \( \mathfrak{a}_{0} \) and \( \mathfrak{a}_{1} \) are \aref{dfn:first-order-language-variable-assignment-function}{variable assignment functions} of \( \mathcal{L} \)
    under \( \mathfrak{A} \) such that \( \restrictedfunctionsmall{\mathfrak{a}_{0}}{\folfreevars{\varphi}} = \restrictedfunctionsmall{\mathfrak{a}_{1}}{\folfreevars{\varphi}} \),
    then \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{0}} \) if and only if \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{1}} \).
\end{thm}

\begin{proof}
    Since \aref{cor:first-order-language-every-formula-has-finite-complexity}{every \( \mathcal{L} \)-formula has a finite complexity}, we can establish
    the theorem inductively.
    We begin by defining the statement \( P(n) \) as follows: for all \( \mathcal{L} \)-formula \( \varphi \) such that
    \( \folcomplexityofformula{\varphi} = n \) and for all variable assignment functions (of \( \mathcal{L} \) under \( \mathfrak{A} \))
    \( \mathfrak{a}_{0} \) and \( \mathfrak{a}_{1} \) such that
    \( \restrictedfunctionsmall{\mathfrak{a}_{0}}{ \folfreevars{\varphi} } = \restrictedfunctionsmall{\mathfrak{a}_{1}}{ \folfreevars{\varphi} } \),
    \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{0}} \) if and only if \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{1}} \).
    Then we make the following hypothesis for some arbitrary \( n \in \naturalset \).

    \vspace{0.3cm}
    \noindent
    \underline{\scshape Induction Hypothesis:} \( P(k) \) holds for all \( k \in \naturalset \) such that \( k < n \).

    \vspace{0.3cm}
    \noindent
    The goal is to show that \( P(n) \) is implied by the induction hypothesis so that we can conclude by the strong induction principle
    that \( P(n) \) holds for all \( n \in \naturalset \) (and thus for all \( \mathcal{L} \)-formulas).

    \vspace{0.3cm}
    \noindent
    Let \( \varphi \) be an arbitrary \( \mathcal{L} \)-formula such that \( \folcomplexityofformula{\varphi} = n \).
    Let \( \mathfrak{a}_{0} \) and \( \mathfrak{a}_{1} \) be arbitrary variable assignment functions of \( \mathcal{L} \) under \( \mathfrak{A} \)
    such that \( \restrictedfunctionsmall{\mathfrak{a}_{0}}{ \folfreevars{\varphi} } = \restrictedfunctionsmall{\mathfrak{a}_{1}}{ \folfreevars{\varphi} } \).
    Since \( n \in \naturalset \), there are only two cases to consider: \( n = 0 \) or \( n \neq 0 \).

    \vspace{0.3cm}
    \noindent
    {\scshape Case} \( n = 0 \):
    In this case, \( \varphi \) must be an atom of \( \mathcal{L} \); that is, \( \varphi \) either has the form \( s = t \) where \( s \) and \( t \)
    are \( \mathcal{L} \)-terms, or it has the form \( P^{\mathcal{L}}_{i}(\tau) \) for some \( i \in \dom P^{\mathcal{L}} \) where \( \tau \) is a tuple of
    \( \mathcal{L} \)-terms such that \( \dom \tau = \Pi^{\mathcal{L}}(i) \).
    Suppose that \( \varphi \) has the form \( s = t \).
    By \cref{dfn:first-order-language-recursive-definition-of-free-variables}, we have
    \( \folfreevars{\varphi} = \folvars{\varphi} = \folvars{s} \union \folvars{t}  \) and thus
    \( \folvars{s} \subseteq \folfreevars{\varphi} \) and \( \folvars{t} \subseteq \folfreevars{\varphi} \).
    Since \( \restrictedfunctionsmall{\mathfrak{a}_{0}}{\folfreevars{\varphi}} = \restrictedfunctionsmall{\mathfrak{a}_{1}}{\folfreevars{\varphi}} \),
    it follows that \( \restrictedfunctionsmall{\mathfrak{a}_{0}}{\folvars{s}} = \restrictedfunctionsmall{\mathfrak{a}_{1}}{\folvars{s}} \)
    and \( \restrictedfunctionsmall{\mathfrak{a}_{0}}{\folvars{t}} = \restrictedfunctionsmall{\mathfrak{a}_{1}}{\folvars{t}} \).
    In other words, \cref{lem:first-order-language-different-variable-assignment-functions-yield-the-same-interpretation-if-they-agree-for-all-variable-symbols-in-a-term}
    is applicable to \( \mathfrak{a}_{0} \) and \( \mathfrak{a}_{1} \), from which we infer:
    \begin{align}
        \folterminterpretation{s}{\mathfrak{A}}{\mathfrak{a}_{0}} &=
        \folterminterpretation{s}{\mathfrak{A}}{\mathfrak{a}_{1}} \phantomsection \label{eq:interpreting-s-yields-the-same-entity}\\
        \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{0}} &=
        \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{1}} \phantomsection \label{eq:interpreting-t-yields-the-same-entity}
    \end{align}
    From \cref{dfn:first-order-language-satisfaction-of-formulas}, we know that \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{0}} \)
    if and only if \( \folterminterpretation{s}{\mathfrak{A}}{\mathfrak{a}_{0}} = \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{0}} \).
    But from \eqref{eq:interpreting-s-yields-the-same-entity} and \eqref{eq:interpreting-t-yields-the-same-entity}, it is clear that
    \( \folterminterpretation{s}{\mathfrak{A}}{\mathfrak{a}_{0}} = \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{0}} \) if and only if
    \( \folterminterpretation{s}{\mathfrak{A}}{\mathfrak{a}_{1}} = \folterminterpretation{t}{\mathfrak{A}}{\mathfrak{a}_{1}} \).
    Applying \cref{dfn:first-order-language-satisfaction-of-formulas} once again allows us to conclude that
    \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{0}} \) if and only if \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{1}} \).

    \vspace{0.3cm}
    \noindent
    Similarly, if \( \varphi \) has the form \( P^{\mathcal{L}}_{i}(\tau) \),
    then \cref{lem:first-order-language-different-variable-assignment-functions-yield-the-same-interpretation-if-they-agree-for-all-variable-symbols-in-a-term}
    guarantees that
    \begin{equation*}
        \sequence{ \folterminterpretation{\tau_{j}}{\mathfrak{A}}{\mathfrak{a}_{0}} }{j \in \dom \tau}
        =
        \sequence{ \folterminterpretation{\tau_{j}}{\mathfrak{A}}{\mathfrak{a}_{1}} }{j \in \dom \tau}
    \end{equation*}
    since \( \restrictedfunctionsmall{\mathfrak{a}_{0}}{ \folvars{\tau_{j}} } = \restrictedfunctionsmall{\mathfrak{a}_{1}}{ \folvars{\tau_{j}} } \) for all
    \( j \in \dom \tau \). It should be clear from \cref{dfn:first-order-language-satisfaction-of-formulas} that
    \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{0}} \) iff \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{1}} \) holds in this case as well.

    \vspace{0.3cm}
    \noindent
    {\scshape Case} \( n \neq 0 \): From \cref{cor:first-order-language-every-formula-with-non-zero-complexity-can-be-decomposed}, we know that
    \( \varphi \) must take one of seven forms.
    Suppose that \( \varphi \coloneq \mleft( \psi \mright) \) for some \( \mathcal{L} \)-formula \( \psi \).
    Clearly, \( \psi \folsubformula \varphi \),
    so we have \( \folcomplexityofformula{\psi} < n \) by \cref{cor:first-order-language-the-complexity-of-a-subformula-is-less-than-the-complexity-of-the-formula}.
    Moreover, we also have \( \restrictedfunctionsmall{\mathfrak{a}_{0}}{ \folfreevars{\psi} } = \restrictedfunctionsmall{\mathfrak{a}_{1}}{ \folfreevars{\psi} } \)
    since \( \folfreevars{\psi} = \folfreevars{\varphi} \) and
    \( \restrictedfunctionsmall{\mathfrak{a}_{0}}{\folfreevars{\varphi}} = \restrictedfunctionsmall{\mathfrak{a}_{1}}{\folfreevars{\varphi}} \) by our initial assumption.
    From the induction hypothesis, it thus follows that \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{0}} \) iff \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}_{0}} \)
    iff \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}_{1}} \) iff \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{1}} \).
    Note that the proof is exactly the same if \( \varphi \) has the form \( \mleft[ \, \psi \, \mright] \) or \( \lnot \, \psi \).

    \vspace{0.3cm}
    \noindent
    Next, we consider the case where \( \varphi \) takes on the form \( \psi \,\,{\land}\,\, \phi \).
    Once again, \( \folcomplexityofformula{\psi} < n \) and \( \folcomplexityofformula{\phi} < n \) are both true
    by \cref{cor:first-order-language-the-complexity-of-a-subformula-is-less-than-the-complexity-of-the-formula}
    because \( \psi \) and \( \phi \) are subformulas of \( \varphi \).
    Then from \cref{dfn:first-order-language-satisfaction-of-formulas}, we know that \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{0}} \)
    if and only if \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}_{0}} \) and \( \folwffsatisfaction{\mathfrak{A}}{\phi}{\mathfrak{a}_{0}} \) both hold.
    Since \( \folcomplexityofformula{\psi} \) and \( \folcomplexityofformula{\phi} \) are both less than \( n \) and the variable assignment functions
    \( \mathfrak{a}_{0} \) and \( \mathfrak{a}_{1} \) overlap for all free variables of \( \varphi \), it follows from the induction hypothesis that
    \begin{align*}
        \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}_{0}} &\text{ iff } \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}_{1}}\\
        \folwffsatisfaction{\mathfrak{A}}{\phi}{\mathfrak{a}_{0}} &\text{ iff } \folwffsatisfaction{\mathfrak{A}}{\phi}{\mathfrak{a}_{1}}
    \end{align*}
    Thus, necessarily \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{0}} \) iff \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{1}} \).
    The same argument applies if \( \varphi \) has the form \( \psi \,\,{\lor}\,\, \phi \).

    \vspace{0.3cm}
    \noindent
    Finally, we consider the case where \( \varphi \) is constructed by quantification (\( \mathcal{R}_{\forall} \) or \( \mathcal{R}_{\exists} \)).
    If \( \varphi \coloneq \forall x \, \psi \), then \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{0}} \) if and only if
    \( \folwffsatisfaction{\mathfrak{A}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}_{0}}{x}{e} } \) for all \( e \in \abs{\mathfrak{A}} \).
    Now let \( e \in \abs{\mathfrak{A}} \) be arbitrary.
    We know from \cref{dfn:first-order-language-recursive-definition-of-free-variables} that
    \begin{equation*}
        \folfreevars{\varphi} =
        \begin{cases}
            \folfreevars{\psi} - \mleft\{ x \mright\} & \text{\normalfont if }  x \in \folfreevars{\psi} \\
            \folfreevars{\psi} & \text{\normalfont if } x \notin \folfreevars{\psi}
        \end{cases}
    \end{equation*}
    If \( x \notin \folfreevars{\psi} \), then obviously \( \restrictedfunctionsmall{\mathfrak{a}_{0}}{ \folfreevars{\psi} } = \restrictedfunctionsmall{\mathfrak{a}_{1}}{ \folfreevars{\psi} }  \) because \( \restrictedfunctionsmall{\mathfrak{a}_{0}}{ \folfreevars{\varphi} } = \restrictedfunctionsmall{\mathfrak{a}_{1}}{ \folfreevars{\varphi} } \) by our initial assumption
    and \( \folfreevars{\varphi} \) and \( \folfreevars{\psi} \) are the same set.
    We can deduce from this the fact that
    \begin{equation}
        \phantomsection \label{eq:modified-variable-assignment-function-also-agree}
        \restrictedfunctionsmall{ \folmodifiedvariableassignmentfunction{\mathfrak{a}_{0}}{x}{e} }{ \folfreevars{\psi} }
        =
        \restrictedfunctionsmall{ \folmodifiedvariableassignmentfunction{\mathfrak{a}_{1}}{x}{e} }{ \folfreevars{\psi} }
    \end{equation} 
    \noindent
    On the other hand, if \( x \in \folfreevars{\psi} \), then \( \folfreevars{\psi} = \folfreevars{\varphi} \union \mleft\{ x \mright\} \).
    If we were to construct the restricted functions \( \restrictedfunctionsmall{\mathfrak{a}_{0}}{\folfreevars{\psi}} \)
    and \( \restrictedfunctionsmall{\mathfrak{a}_{1}}{\folfreevars{\psi}} \), we will end up with
    \begin{align*}
        \restrictedfunctionsmall{\mathfrak{a}_{0}}{\folfreevars{\psi}} = \restrictedfunctionsmall{\mathfrak{a}_{0}}{\folfreevars{\varphi}} \union \big\{ \orderedpair{x}{e_{0}} \big\}\\
        \restrictedfunctionsmall{\mathfrak{a}_{1}}{\folfreevars{\psi}} = \restrictedfunctionsmall{\mathfrak{a}_{1}}{\folfreevars{\varphi}} \union \big\{ \orderedpair{x}{e_{1}} \big\}
    \end{align*}
    where \( e_{0} \) and \( e_{1} \) are elements of \( \abs{\mathfrak{A}} \). But for \( \folmodifiedvariableassignmentfunction{\mathfrak{a}_{0}}{x}{e} \) and
    \( \folmodifiedvariableassignmentfunction{\mathfrak{a}_{1}}{x}{e} \), the values \( e_{0} \) and \( e_{1} \) will simply be replaced with the same value \( e \)
    regardless of what they are; and given that \( \restrictedfunctionsmall{\mathfrak{a}_{0}}{\folfreevars{\varphi}} =  \restrictedfunctionsmall{\mathfrak{a}_{1}}{\folfreevars{\varphi}} \)
    by our initial assumption, we arrive at \eqref{eq:modified-variable-assignment-function-also-agree} once again. In other words, \eqref{eq:modified-variable-assignment-function-also-agree} holds for all possible cases.

    \vspace{0.3cm}
    \noindent
    Now recall that \( \folcomplexityofformula{\psi} < n \) since \( \psi \) is a subformula of \( \varphi \).
    Together, \eqref{eq:modified-variable-assignment-function-also-agree} and the induction hypothesis implies that
    \( \folwffsatisfaction{\mathfrak{A}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}_{0}}{x}{e} } \)
    iff \( \folwffsatisfaction{\mathfrak{A}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}_{1}}{x}{e} } \),
    which in turn implies that 
    \( \folwffsatisfaction{\mathfrak{A}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}_{0}}{x}{e} } \) 
    for all \( e \in \abs{\mathfrak{A}} \) iff \( \folwffsatisfaction{\mathfrak{A}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}_{1}}{x}{e} } \)
    for all \( e \in \abs{\mathfrak{A}} \).
    In other words, \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{0}} \) iff \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{1}} \)
    by \cref{dfn:first-order-language-satisfaction-of-formulas}.
    If \( \varphi \) is existentially quantified, then the proof is exactly the same except for the conclusion drawn at the end.
    The key is to establish that \( \folwffsatisfaction{\mathfrak{A}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}_{0}}{x}{e} } \)
    iff \( \folwffsatisfaction{\mathfrak{A}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}_{1}}{x}{e} } \), and the rest follows readily from
    \cref{dfn:first-order-language-satisfaction-of-formulas}.

    \vspace{0.3cm}
    \noindent
    We have now established that the statement \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{0}} \) iff \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{1}} \)
    is true regardless of the form \( \varphi \) takes. Since \( \varphi \), \( \mathfrak{a}_{0} \), and \( \mathfrak{a}_{1} \) are all arbitrary, we conclude by
    \aref{rule-of-inference:universal-generalization}{Universal Generalization} that \( P(n) \) holds.
    Therefore, \( P(n) \) is implied by the induction hypothesis for any arbitrary \( n \in \naturalset \).
    \cref{thm:first-order-language-different-variable-assignment-functions-yield-the-same-interpretation-if-they-agree-on-the-free-variables-of-a-formula} follows
    readily from the strong induction principle.
\end{proof}

\begin{thm}
    \phantomsection \label{thm:first-order-language-sentences-must-be-satisfied-by-all-variable-assignment-functions}
    Let \( \varphi \) be an \( \mathcal{L} \)-sentence and \( \mathfrak{A} \) be any \( \mathcal{L} \)-structure.
    If \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \) for some variable assignment function \( \mathfrak{a} \),
    then \( \mathfrak{A} \) satisfies \( \varphi \) for all variable assignment functions.
\end{thm}

\begin{proof}
    Let \( \mathfrak{a}_{0} \) be a variable assignment function of \( \mathcal{L} \) under \( \mathfrak{A} \) such that
    \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{0}} \).
    Let \( \mathfrak{a}_{1} \) denote an arbitrary variable assignment function of \( \mathcal{L} \) under \( \mathfrak{A} \).
    Since \( \varphi \) is a sentence in the language of \( \mathcal{L} \), it follows that \( \folfreevars{\varphi} = \emptyset \).
    Therefore,
    \begin{equation*}
        \restrictedfunctionsmall{\mathfrak{a}_{0}}{ \folfreevars{\varphi} }
        = \restrictedfunctionsmall{\mathfrak{a}_{0}}{ \emptyset }
        = \emptyset
        = \restrictedfunctionsmall{\mathfrak{a}_{1}}{ \emptyset }
        = \restrictedfunctionsmall{\mathfrak{a}_{1}}{ \folfreevars{\varphi} }
    \end{equation*}
    By \cref{thm:first-order-language-different-variable-assignment-functions-yield-the-same-interpretation-if-they-agree-on-the-free-variables-of-a-formula},
    it follows that \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \) iff \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}_{1}} \).
    Since \( \mathfrak{a}_{1} \) is arbitrary, it follows that \( \mathfrak{A} \) satisfies \( \varphi \) for all variable assignment functions of \( \mathcal{L} \)
    under \( \mathfrak{A} \).
\end{proof}

\begin{thm}
    \phantomsection \label{thm:first-order-language-replacing-variable-symbols-of-term-with-new-ones-does-not-affect-interpretation-with-modified-assignment-function}
    Let \( t \) be any \( \mathcal{L} \)-term. If \( x \) and \( y \) denote two \( \mathcal{L} \)-variables such that \( x \in \folvars{t} \)
    and \( y \notin \folvars{t} \) and \( t_{\ast} \) denotes the \( \mathcal{L} \)-term obtained after replacing all occurrences of \( x \) in \( t \)
    with \( y \), then \( \folterminterpretation{t}{\mathfrak{A}}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e} } =
    \folterminterpretation{t_{\ast}}{\mathfrak{A}}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{y}{e} } \) for all \( \mathcal{L} \)-structure \( \mathfrak{A} \),
    variable assignment function \( \mathfrak{a} \), and \( e \in \abs{\mathfrak{A}} \).
\end{thm}

\begin{proof}
    Since every \( \mathcal{L} \)-term has finite complexity, we can prove the lemma inductively. Let \( P(n) \) be a statement about a natural number \( n \)
    with the following definition: for all \( \mathcal{L} \)-term \( t \) such that \( \folcomplexityofterm{t} = n \) and any two \( \mathcal{L} \)-variables \( x \) and \( y \)
    such that \( x \in \folvars{t} \) and \( y \notin \folvars{t} \), if \( t_{\ast} \) denotes the \( \mathcal{L} \)-term obtained by replacing all occurrences of \( x \)
    in \( t \) with \( y \), then \( \folterminterpretation{t}{\mathfrak{A}}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e} } =
    \folterminterpretation{t_{\ast}}{\mathfrak{A}}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{y}{e} } \) for all \( \mathcal{L} \)-structure \( \mathfrak{A} \),
    variable assignment function \( \mathfrak{a} \), and \( e \in \abs{\mathfrak{A}} \).
    We now state our induction hypothesis for an arbitrary \( n \in \naturalset \).

    \vspace{0.3cm}
    \noindent
    \underline{\scshape Induction Hypothesis}: \( P(k) \) holds for all \( k < n \).

    \vspace{0.3cm}
    \noindent
    Let \( t \) be any \( \mathcal{L} \)-term with complexity \( n \) such that \( x \in \folvars{t} \) and \( y \notin \folvars{t} \) for
    \( \mathcal{L} \)-variables \( x \) and \( y \). Clearly, we must consider two cases: \( n = 0 \) or \( n \neq 0 \).

    \vspace{0.3cm}
    \noindent
    {\scshape Case} \( n = 0 \):
    Clearly, \( t \) consists solely of the variable symbol \( x \)
    (\( t \) cannot have the form \( F^{\mathcal{L}}_{i}() \) because \( x \in \folvars{t} \) by our initial assumption).
    Therefore, \( t_{\ast} \) must consist solely of the variable symbol \( y \). Let \( \mathfrak{A} \) be any \( \mathcal{L} \)-structure
    and \( \mathfrak{a} \) be any variable assignment function of \( \mathcal{L} \) under \( \mathfrak{A} \). Let \( e_{0} \) be any
    element of \( \abs{\mathfrak{A}} \). By \cref{dfn:first-order-language-interpretation-of-terms}, it follows that
    \( \folterminterpretation{t}{\mathfrak{A}}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e_{0}} }
    = \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e_{0}}(x)
    = e_{0}
    = \folmodifiedvariableassignmentfunction{\mathfrak{a}}{y}{e_{0}}(y)
    = \folterminterpretation{t_{\ast}}{\mathfrak{A}}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{y}{e_{0}} } \).

    \vspace{0.3cm}
    \noindent
    {\scshape Case} \( n \neq 0 \):
    By \cref{cor:first-order-language-terms-with-non-zero-complexity-must-have-a-parent-function-symbol}, it follows that \( t \)
    must have the form \( F^{\mathcal{L}}_{i}(\tau) \) for some \( i \in \dom F^{\mathcal{L}} \) where \( \tau \) is a non-empty tuple of
    \( \mathcal{L} \)-terms. Clearly, \( t_{\ast} = F^{\mathcal{L}}_{i}(\sigma) \) where \( \sigma \) is a tuple of \( \mathcal{L} \)-terms
    such that \( \dom \sigma = \dom \tau \) and
    \begin{equation*}
        \sigma_{j} =
        \begin{cases}
            \tau_{j} \ &\text{if} \ x \notin \folvars{\tau_{j}}\\
            (\tau_{j})_{\ast} &\text{if} \ x \in  \folvars{\tau_{j}}
        \end{cases}
    \end{equation*}
    for all \( k \in \dom \tau \). Now let \( \mathfrak{B} \) be any \( \mathcal{L} \)-structure and \( \mathfrak{b} \) be a variable assignment function
    of \( \mathcal{L} \) under \( \mathfrak{B} \). Let \( e_{1} \) be any element of \( \abs{\mathfrak{B}} \).
    By \cref{dfn:first-order-language-interpretation-of-terms}, it follows that
    \begin{align*}
        \folterminterpretation{t}{\mathfrak{B}}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{1}} } &=
        F^{\mathfrak{B}}_{i}\sequence{ \folterminterpretation{\tau_{j}}{\mathfrak{B}}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{1}} } }{ j \in \dom \tau }\\
        \folterminterpretation{t_{\ast}}{\mathfrak{B}}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{y}{e_{1}} } &=
        F^{\mathfrak{B}}_{i}\sequence{ \folterminterpretation{\sigma_{j}}{\mathfrak{B}}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{y}{e_{1}} } }{ j \in \dom \tau }
    \end{align*}
    It remains to show that
    \begin{equation}
        \phantomsection \label{eq:two-internal-sequences-are-equal}
        \sequence{ \folterminterpretation{\tau_{j}}{\mathfrak{B}}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{1}} } }{ j \in \dom \tau }
      = \sequence{ \folterminterpretation{\sigma_{j}}{\mathfrak{B}}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{y}{e_{1}} } }{ j \in \dom \tau }
    \end{equation}
    Let \( j \in \dom \tau \) be arbitrary. Since \( x \in \folvars{t} \) by our initial assumption, it follows that either \( x \in \folvars{\tau_{j}} \) or
    \( x \notin \folvars{\tau_{j}} \). If \( x \notin \folvars{\tau_{j}} \), then clearly \( \sigma_{j} = \tau_{j} \) so that
    \( \folvars{\sigma_{j}} = \folvars{\tau_{j}} \); i.e., they are literally the same term.
    Since \( y \notin \folvars{\tau_{j}} \), it is also necessary that \( y \notin \folvars{\tau_{j}} \).
    Consequently,
    \begin{equation*}
        \restrictedfunction{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{1}} }{ \folvars{\tau_{j}} }
      = \restrictedfunction{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{y}{e_{1}} }{ \folvars{\tau_{j}} }
    \end{equation*}
    Since \( \sigma_{j} = \tau_{j} \), necessarily
    \( \folterminterpretation{\tau_{j}}{\mathfrak{B}}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{1}} } =
    \folterminterpretation{\sigma_{j}}{\mathfrak{B}}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{y}{e_{1}} } \).
    
    \vspace{0.3cm}
    \noindent
    On the other hand, if \( x \in \folvars{\tau_{j}} \). Then \( \sigma_{j} = (\tau_{j})_{\ast} \).
    By \cref{dfn:first-order-language-subterms}, it follows that \( \tau_{j} \) is a subterm of \( t \).
    But \( \folcomplexityofterm{\tau_{j}} < n \) by \cref{cor:first-order-language-the-complexity-of-a-subterm-is-less-than-the-complexity-of-the-parent-term},
    which means that the induction hypothesis applies to \( \tau_{j} \).
    Once again, this implies that
    \( \folterminterpretation{\tau_{j}}{\mathfrak{B}}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{1}} } =
    \folterminterpretation{\sigma_{j}}{\mathfrak{B}}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{y}{e_{1}} } \).

    \vspace{0.3cm}
    \noindent
    The above analysis proves that the two sequences in \eqref{eq:two-internal-sequences-are-equal} have the same value for all \( j \in \dom \tau \).
    This suffices to establish \eqref{eq:two-internal-sequences-are-equal}. 

    \vspace{0.3cm}
    \noindent
    We have thus proven that the induction hypothesis implies \( P(n) \).
    Since \( n \in \naturalset \) is arbitrary, the theorem follows from strong induction.
\end{proof}

\section{Entailment}
\noindent
The notion of satisfaction can be used to define the logical relationship between formulas; specifically the notions of
logical consequence and logical equivalence.
Informally, if a formula \( \varphi \) is a logical consequence of another formula \( \psi \) (or a set of formulas), then \( \varphi \)
is implied by \( \psi \) for all possible interpretations; that is, the implication holds by virtue of the logical form of the formulas alone.

\begin{dfn}[Logical Consequence]
    \phantomsection \label{dfn:first-order-language-logical-consequence}
    Let \( \Gamma \) be a set of \( \mathcal{L} \)-formulas and \( \varphi \) be an \( \mathcal{L} \)-formula.
    We say that \( \varphi \) is a \keyword{logical consequence} of \( \Gamma \) (denoted as \( \Gamma \models \varphi \))
    if and only if for all \( \mathcal{L} \)-structure \( \mathfrak{A} \) and for all variable assignment function \( \mathfrak{a} \),
    if \( \mathfrak{A} \) satisfies every formula in \( \Gamma \) with \( \mathfrak{a} \), then \( \mathfrak{A} \) also satisfies \( \varphi \) with \( \mathfrak{a} \).
\end{dfn}

\begin{notation}
    If \( \Gamma \) consists of a single formula \( \psi \) (i.e., \( \Gamma = \mleft\{ \psi \mright\} \)), then we abbreviate the notation
    \( \mleft\{ \psi \mright\} \models \varphi \) with \( \psi \models \varphi \).
\end{notation}

\begin{dfn}[Tautology]
    \phantomsection \label{dfn:first-order-language-tautology}
    An \( \mathcal{L} \)-formula \( \varphi \) is a \keyword{tautology} if and only if \( \emptyset \models \varphi \);
    that is, \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \) for all \( \mathcal{L} \)-structure \( \mathfrak{A} \)
    and for all variable assignment function \( \mathfrak{a} \).
    By contrast, we say that \( \varphi \) is a \keyword{contradiction} if and only if \( \folwffnonsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \)
    for all structure \( \mathfrak{A} \) and all assignment function \( \mathfrak{a} \).
\end{dfn}

\begin{notation}
    We can abbreviate the notation \( \emptyset \models \varphi \) as \( \models \varphi \).
\end{notation}

\begin{dfn}[Logical Equivalence]
    \phantomsection \label{dfn:first-order-language-logical-equivalence}
    If \( \varphi \) and \( \psi \) are \( \mathcal{L} \)-formulas, then \( \varphi \) and \( \psi \) are \keyword{logically equivalent}
    (denoted as \( \varphi \equiv \psi \)) if and only if \( \varphi \models \psi \) and \( \psi \models \varphi \) are both true.
\end{dfn}

\noindent
We now prove some basic properties involving the \( \models \) and \( \equiv \) relations.

\begin{thm}[Transitivity]
    \phantomsection \label{thm:first-order-language-transitivity-of-logical-consequence-and-equivalence}
    The following properties hold for all \( \mathcal{L} \)-formulas \( \varphi \), \( \psi \), and \( \phi \):
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item If \( \varphi \models \psi \) and \( \psi \models \phi \), then \( \varphi \models \phi \) \phantomsection \label{enum:transitivity-of-consequence}
        \item If \( \varphi \equiv \psi \) and \( \psi \equiv \phi \), then \( \varphi \equiv \phi \) \phantomsection \label{enum:transitivity-of-equivalence}
    \end{enumerate}
\end{thm}

\begin{proof}
    Suppose that \( \varphi \models \psi \) and \( \psi \models \phi \) are true.
    Let \( \mathfrak{A} \) be an \( \mathcal{L} \)-structure and \( \mathfrak{a} \) be any
    variable assignment function of \( \mathcal{L} \) under \( \mathfrak{A} \) such that
    \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \).
    Since \( \varphi \models \psi \), \aref{dfn:first-order-language-logical-consequence}{necessarily \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \)}.
    But \( \psi \models \phi \) is also true, so \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \)
    implies \( \folwffsatisfaction{\mathfrak{A}}{\phi}{\mathfrak{a}} \). In other words, \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \)
    implies \( \folwffsatisfaction{\mathfrak{A}}{\phi}{\mathfrak{a}} \) for any structure \( \mathfrak{A} \) and any variable assignment function \( \mathfrak{a} \).
    Thus, \( \varphi \models \phi \) as stated in \ref{enum:transitivity-of-consequence}.

    \vspace{0.3cm}
    \noindent
    We now use \ref{enum:transitivity-of-consequence} to prove \ref{enum:transitivity-of-equivalence}.
    Suppose that \( \varphi \equiv \psi \) and \( \psi \equiv \phi \) are both true.
    By \cref{dfn:first-order-language-logical-equivalence}, it follows that \( \varphi \models \psi \) and \( \psi \models \phi \) are both true, which
    then implies \( \varphi \models \phi \) as shown in \ref{enum:transitivity-of-consequence}.
    Similarly, \( \phi \models \psi \) and \( \psi \models \varphi \) are also true, which implies \( \phi \models \varphi \).
    Since \( \varphi \models \phi \) and \( \phi \models \varphi \) are both true, necessarily \( \varphi \equiv \phi \).
\end{proof}

\begin{thm}
    \phantomsection \label{thm:first-order-language-tautological-conjuncts-iff-tautological-conjunction}
    Let \( \varphi \) and \( \psi \) be \( \mathcal{L} \)-formulas. Then \( \models \mleft( \varphi \,\,{\land}\,\, \psi \mright) \) if and only if
    \( \models \varphi \) and \( \models \psi \) are both true.
\end{thm}

\begin{proof}
    Suppose that \( \models \mleft( \varphi \,\,{\land}\,\, \psi \mright) \) holds.
    Let \( \mathfrak{A} \) be any \( \mathcal{L} \)-structure.
    Let \( \mathfrak{a} \) be any variable assignment function of \( \mathcal{L} \) under \( \mathfrak{A} \).
    Since \( \models \mleft( \varphi \,\,{\land}\,\, \psi \mright) \),
    it follows that \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \varphi \,\,{\land}\,\, \psi \mright)}{\mathfrak{a}} \)
    and thus \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \) and \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \).
    Since \( \mathfrak{A} \) and \( \mathfrak{a} \) are arbitrary, we conclude that \( \models \varphi \) and \( \models \psi \) are both true.

    \vspace{0.3cm}
    \noindent
    Conversely, suppose that \( \models \varphi \) and \( \models \psi \) are both true.
    Let \( \mathfrak{B} \) be any \( \mathcal{L} \)-structure.
    Let \( \mathfrak{b} \) be any variable assignment function of \( \mathcal{L} \) under \( \mathfrak{B} \).
    Since \( \models \varphi \) and \( \models \psi \) are both true, necessarily \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{\mathfrak{b}} \)
    and \( \folwffsatisfaction{\mathfrak{B}}{\psi}{\mathfrak{b}} \).
    Thus, \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \varphi \,\,{\land}\,\, \psi \mright)}{\mathfrak{b}} \)
    by \cref{dfn:first-order-language-satisfaction-of-formulas}.
    Since \( \mathfrak{B} \) and \( \mathfrak{a} \) are arbitrary, we conclude that \( \varphi \,\,{\land}\,\, \psi \) is a tautology.
\end{proof}

\begin{thm}
    \phantomsection \label{thm:first-order-language-equivalent-formulations-of-logical-consequence-and-logical-equivalence}
    For all \( \mathcal{L} \)-formulas \( \varphi \) and \( \psi \), the following holds:
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item \( \varphi \models \psi \) if and only if \( \models \mleft( \varphi \,{\implies} \psi \mright) \) \phantomsection \label{enum:logical-consequence-iff-tautological-implication}
        \item \( \varphi \,\,{\equiv}\,\, \psi \) if and only if \( \models \mleft( \varphi {\iff} \psi \mright) \) \phantomsection \label{enum:logical-equivalence-iff-tautological-biconditional}
    \end{enumerate}
\end{thm}

\begin{proof}
    We begin by proving \ref{enum:logical-consequence-iff-tautological-implication}, which we can then use to prove \ref{enum:logical-equivalence-iff-tautological-biconditional}.

    \vspace{0.3cm}
    \noindent
    Suppose that \( \varphi \models \psi \). Let \( \mathfrak{A} \) be any \( \mathcal{L} \)-structure and \( \mathfrak{a} \) be any
    variable assignment function of \( \mathcal{L} \) under \( \mathfrak{A} \).
    If \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \), then \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \)
    must follow because \( \varphi \models \psi \).
    Therefore, \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \varphi {\implies} \psi \mright)}{\mathfrak{a}} \) because
    \( \varphi {\implies} \psi \) is simply an abbreviation for \( \lnot \, \varphi \,\,{\lor}\,\, \psi \).
    Since \( \mathfrak{A} \) and \( \mathfrak{a} \) are arbitrary, we conclude that \( \models \mleft( \varphi {\implies} \psi \mright) \).
    To prove the converse, suppose that \( \models \mleft( \varphi {\implies} \psi \mright) \) holds.
    This means that
    \begin{equation}
        \label{eq:implication-holds-for-all-structures-and-assignments}
        \folwffsatisfaction{\mathfrak{A}}{\mleft( \lnot \, \varphi \,\,{\lor}\,\, \psi \mright)}{\mathfrak{a}} \ \ \  \text{for all \( \mathfrak{A} \) and \( \mathfrak{a} \) }
    \end{equation}
    Let \( \mathfrak{B} \) and \( \mathfrak{b} \) be any structure and variable assignment function such that \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{\mathfrak{b}} \).
    But by \eqref{eq:implication-holds-for-all-structures-and-assignments}, either \( \folwffnonsatisfaction{\mathfrak{B}}{\varphi}{\mathfrak{b}} \)
    or \( \folwffsatisfaction{\mathfrak{B}}{\psi}{\mathfrak{b}} \) is true.
    But we already know that \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{\mathfrak{b}} \) is true by assumption; therefore,
    \( \folwffsatisfaction{\mathfrak{B}}{\psi}{\mathfrak{b}} \) follows by \aref{rule-of-inference:disjunctive-syllogism}{Disjunctive Syllogism}.
    In other words, \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{\mathfrak{b}} \) implies \( \folwffsatisfaction{\mathfrak{B}}{\psi}{\mathfrak{b}} \)
    for any \( \mathfrak{B} \) and \( \mathfrak{b} \); i.e., \( \varphi \models \psi \).
    This suffices to prove \ref{enum:logical-consequence-iff-tautological-implication}.

    \vspace{0.3cm}
    \noindent
    To prove \ref{enum:logical-equivalence-iff-tautological-biconditional}, suppose that \( \varphi \equiv \psi \) holds;
    that is, \( \varphi \models \psi \) and \( \psi \models \varphi \) are both true.
    But \( \varphi \models \psi \) iff \( \models \mleft( \varphi {\implies} \psi \mright) \)
    and \( \psi \models \varphi \) iff \( \models \mleft( \psi {\implies} \varphi \mright) \)
    by \ref{enum:logical-consequence-iff-tautological-implication}.
    From this, we infer that \( \varphi \equiv \psi \) iff \( \models \mleft( \varphi {\implies} \psi \mright) \) and \( \models \mleft( \psi {\implies} \varphi \mright) \)
    are both true; and by \cref{thm:first-order-language-tautological-conjuncts-iff-tautological-conjunction}, this statement can be equivalently expressed as
    \( \varphi \equiv \psi \) iff \( \models \mleft( \varphi {\implies} \psi \,\,{\land}\,\, \psi {\implies} \varphi \mright) \).
    The formula \( \varphi {\iff} \psi \) is simply an abbreviation for \( \varphi {\implies} \psi \,\,{\land}\,\, \psi {\implies} \varphi \).
\end{proof}

\begin{thm}
    \phantomsection \label{logical-equivalence:parenthesized-formulas-are-logically-equivalent}
    For any \( \mathcal{L} \)-formula \( \varphi \), \( \varphi \equiv \mleft( \varphi \mright) \) and \( \varphi \equiv \mleft[ \,\varphi\, \mright] \).
\end{thm}

\begin{proof}
    Let \( \psi \coloneq \mleft( \varphi \mright) \). The goal is to prove that \( \varphi \models \psi \) and \( \psi \models \varphi \) are both true.

    \vspace{0.3cm}
    \noindent
    Let \( \mathfrak{A} \) be any \( \mathcal{L} \)-structure and \( \mathfrak{a} \) be any variable assignment function of \( \mathcal{L} \) under \( \mathfrak{A} \).
    If \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \), then necessarilly \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \)
    by \cref{dfn:first-order-language-satisfaction-of-formulas}.
    Since \( \mathfrak{A} \) and \( \mathfrak{a} \) are arbitrary, \( \varphi \models \psi \) follows immediately from
    \aref{dfn:first-order-language-logical-consequence}{the definition of logical consequence}.

    \vspace{0.3cm}
    \noindent
    To prove the converse, let \( \mathfrak{B} \) and \( \mathfrak{b} \) denote an arbitrary \( \mathcal{L} \)-structure and variable assignment function respectively.
    Suppose that \( \folwffsatisfaction{\mathfrak{B}}{\psi}{\mathfrak{b}} \). Once again, \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{\mathfrak{b}} \)
    follows from \cref{dfn:first-order-language-satisfaction-of-formulas}. Since \( \mathfrak{B} \) and \( \mathfrak{b} \) are arbitrary,
    we then conclude that \( \psi \models \varphi \).
\end{proof}

\begin{rem}
    The proof for \( \varphi \equiv \mleft[ \,\varphi\, \mright] \) is exactly the same.
\end{rem}

\begin{thm}[Idempotence]
    \phantomsection \label{logical-equivalence:idempotence-of-conjunctions-and-disjunctions}
    For any \( \mathcal{L} \)-formula \( \varphi \), the following hold:
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item \( \varphi \,\,{\land}\,\, \varphi \equiv \varphi \)
        \item \( \varphi \,\,{\lor}\,\, \varphi \equiv \varphi \)
    \end{enumerate}
\end{thm}

\begin{proof}
    Let \( \mathfrak{A} \) be any structure and \( \mathfrak{a} \) be any variable assignment function of \( \mathcal{L} \) under \( \mathfrak{A} \).
    Suppose that \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \varphi \,\,{\land}\,\, \varphi \mright)}{\mathfrak{a}} \).
    By \cref{dfn:first-order-language-satisfaction-of-formulas}, it follows that \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \).
    Thus, \( \varphi \,\,{\land}\,\, \varphi \models \varphi \).
    Similarly, let \( \mathfrak{B} \) and \( \mathfrak{b} \) be an arbitrary structure and a variable assignment function of \( \mathcal{L} \) under \( \mathfrak{B} \)
    respectively. Suppose that \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{\mathfrak{b}} \).
    Clearly, \( \folwffsatisfaction{\mathfrak{B}}{\varphi \,\,{\land}\,\, \varphi}{\mathfrak{b}} \) is also true.
    Thus, \( \varphi \models \varphi \,\,{\land}\,\, \varphi \).
    Since \( \varphi \,\,{\land}\,\, \varphi \models \varphi \) and \( \varphi \,\,{\land}\,\, \varphi \models \varphi \) are both true,
    it follows that \( \varphi \equiv \varphi \,\,{\land}\,\, \varphi \).

    \vspace{0.3cm}
    \noindent
    Let \( \mathfrak{C} \) be any structure and \( \mathfrak{c} \) be any variable assignment function of \( \mathcal{L} \) under \( \mathfrak{C} \).
    Suppose that \( \folwffsatisfaction{\mathfrak{C}}{\varphi}{\mathfrak{c}} \). By \cref{dfn:first-order-language-satisfaction-of-formulas},
    clearly \( \folwffsatisfaction{\mathfrak{C}}{\mleft( \varphi \,\,{\lor}\,\, \varphi \mright)}{\mathfrak{c}} \) holds.
    Thus, \( \varphi \models \varphi \,\,{\lor}\,\, \varphi \).
    For the converse, let \( \mathfrak{D} \) and \( \mathfrak{d} \) denote an arbitrary structure and an arbitrary variable assignment function
    of \( \mathcal{L} \) under \( \mathfrak{D} \) respectively. If \( \folwffsatisfaction{\mathfrak{D}}{\mleft( \varphi \,\,{\lor}\,\, \varphi \mright)}{\mathfrak{d}} \),
    then clearly \( \folwffsatisfaction{\mathfrak{D}}{\varphi}{\mathfrak{d}} \) also hold.
    Thus, \( \varphi \,\,{\lor}\,\, \varphi \models \varphi \).
\end{proof}

\begin{thm}[Commutativity]
    \phantomsection \label{logical-equivalence:commutativity-of-conjunctions-and-disjunctions}
    For any two \( \mathcal{L} \)-formulas \( \varphi \) and \( \psi \), the following hold:
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item \( \varphi \,\,{\land}\,\, \psi \equiv \psi \,\,{\land}\,\, \varphi \)
        \item \( \varphi \,\,{\lor}\,\, \psi \equiv \psi \,\,{\lor}\,\, \varphi \)
    \end{enumerate}
\end{thm}

\begin{proof}
    Let \( \mathfrak{A} \) be any \( \mathcal{L} \)-structure and \( \mathfrak{a} \) be any variable assignment function of \( \mathcal{L} \) under
    \( \mathfrak{A} \). Suppose that \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \varphi \,\,{\land}\,\, \psi \mright)}{\mathfrak{a}} \).
    Then it follows from \cref{dfn:first-order-language-satisfaction-of-formulas} that \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \)
    and \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \) are both true.
    But this also means that \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \psi \,\,{\land}\,\, \varphi \mright)}{\mathfrak{a}} \) is true.
    In other words, \( \varphi \,\,{\land}\,\, \psi \models \psi \,\,{\land}\,\, \varphi \).
    We can prove \( \psi \models \varphi \) with the exact same argument but with the roles of \( \varphi \) and \( \psi \) swapped.

    \vspace{0.3cm}
    \noindent
    Next, let \( \mathfrak{B} \) be any \( \mathcal{L} \)-structure and \( \mathfrak{b} \) be any variable assignment function of \( \mathcal{L} \) under
    \( \mathfrak{B} \).
    Suppose that \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \varphi \,\,{\lor}\,\, \psi \mright)}{\mathfrak{b}} \); that is, either
    \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{\mathfrak{b}} \) or \( \folwffsatisfaction{\mathfrak{B}}{\psi}{\mathfrak{b}} \) is true.
    Clearly, either case implies \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \psi \,\,{\lor}\,\, \varphi \mright)}{\mathfrak{b}} \),
    so we have \( \varphi \,\,{\lor}\,\, \psi \models \psi \,\,{\lor}\,\, \varphi \).
    The converse can be proven with the same argument but with the roles of \( \varphi \) and \( \psi \) swapped.
\end{proof}

\begin{thm}[Distributivity]
    \phantomsection \label{logical-equivalence:distributivity-of-conjunctions-and-disjunctions}
    For all \( \mathcal{L} \)-formulas \( \varphi \), \( \psi \), and \( \phi \), the following hold:
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item \( \varphi \,\,{\land}\,\, \mleft( \psi \,\,{\lor}\,\, \phi \mright) \equiv \mleft( \varphi \,\,{\land}\,\, \psi \mright) \,\,{\lor}\,\, \mleft( \varphi \,\,{\land}\,\, \phi \mright) \)
        \item \( \varphi \,\,{\lor}\,\, \mleft( \psi \,\,{\land}\,\, \phi \mright) \equiv \mleft( \varphi \,\,{\lor}\,\, \psi \mright) \,\,{\land}\,\, \mleft( \varphi \,\,{\lor}\,\, \phi \mright) \)
    \end{enumerate}
\end{thm}

\begin{proof}
    We begin by defining the following notation:
    \begin{align*}
        \theta_{a} &\coloneq \psi \,\,{\lor}\,\, \phi\\
        \theta_{b} &\coloneq \psi \,\,{\land}\,\, \phi\\
        \lambda_{0} &\coloneq \varphi \,\,{\land}\,\, \psi\\
        \lambda_{1} &\coloneq \varphi \,\,{\land}\,\, \phi\\
        \lambda_{2} &\coloneq \varphi \,\,{\lor}\,\, \psi\\
        \lambda_{3} &\coloneq \varphi \,\,{\lor}\,\, \phi
    \end{align*}
    Let \( \mathfrak{A} \) be an \( \mathcal{L} \)-structure and \( \mathfrak{a} \) be any variable assignment function of \( \mathcal{L} \) under \( \mathfrak{A} \).
    Suppose that \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \varphi \,\,{\land}\,\, \theta_{a} \mright)}{\mathfrak{a}} \).
    By \cref{dfn:first-order-language-satisfaction-of-formulas}, it follows that \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \)
    and \( \folwffsatisfaction{\mathfrak{A}}{\theta_{a}}{\mathfrak{a}} \) are both true.
    But for \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \) to hold, either \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \)
    or \( \folwffsatisfaction{\mathfrak{A}}{\phi}{\mathfrak{a}} \) must be true.
    If \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \), then clearly \( \folwffsatisfaction{\mathfrak{A}}{\lambda_{0}}{\mathfrak{a}} \)
    because \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \) and \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \) are both true.
    Similarly, if \( \folwffsatisfaction{\mathfrak{A}}{\phi}{\mathfrak{a}} \), then \( \folwffsatisfaction{\mathfrak{A}}{\lambda_{1}}{\mathfrak{a}} \)
    is implied.
    By \aref{rule-of-inference:constructive-dilemma}{Constructive Dilemma}, we conclude that
    \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \varphi \,\,{\land}\,\, \theta_{a} \mright)}{\mathfrak{a}} \)
    implies \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \lambda_{0} \,\,{\lor}\,\, \lambda_{1} \mright)}{\mathfrak{a}} \).
    Since \( \mathfrak{A} \) and \( \mathfrak{a} \) are arbitrary, we have \( \varphi \,\,{\lor}\,\, \theta_{a} \models \lambda_{0} \,\,{\lor}\,\, \lambda_{1} \).

    \vspace{0.3cm}
    \noindent
    Now for the converse.
    Let \( \mathfrak{B} \) be any \( \mathcal{L} \)-structure. Let \( \mathfrak{b} \) be any variable assignment function of \( \mathcal{L} \)
    under \( \mathfrak{B} \).
    Suppose that \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \lambda_{0} \,\,{\lor}\,\, \lambda_{1} \mright)}{\mathfrak{b}} \);
    that is, either \( \folwffsatisfaction{\mathfrak{B}}{\lambda_{0}}{\mathfrak{b}} \) or \( \folwffsatisfaction{\mathfrak{B}}{\lambda_{1}}{\mathfrak{b}} \) is true.
    If \( \folwffsatisfaction{\mathfrak{B}}{\lambda_{0}}{\mathfrak{b}} \), then \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{\mathfrak{b}} \)
    and \( \folwffsatisfaction{\mathfrak{B}}{\psi}{\mathfrak{b}} \) are both true.
    Therefore, \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \varphi \,\,{\land}\,\, \theta_{a} \mright)}{\mathfrak{b}} \) holds.
    Similarly, if \( \folwffsatisfaction{\mathfrak{B}}{\lambda_{1}}{\mathfrak{b}} \),
    then \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{\mathfrak{b}} \) and \( \folwffsatisfaction{\mathfrak{B}}{\phi}{\mathfrak{b}} \) both hold.
    Again, this implies that \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \varphi \,\,{\land}\,\, \theta_{a} \mright)}{\mathfrak{b}} \) holds.
    In other words, \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \varphi \,\,{\land}\,\, \theta_{a} \mright)}{\mathfrak{b}} \) follows from
    \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \lambda_{0} \,\,{\lor}\,\, \lambda_{1} \mright)}{\mathfrak{b}} \) for any structure
    \( \mathfrak{B} \) and any variable assignment function \( \mathfrak{b} \).
    Thus, \( \lambda_{0} \,\,{\lor}\,\, \lambda_{1} \models \varphi \,\,{\land}\,\, \theta_{a} \).

    \vspace{0.3cm}
    \noindent
    Let \( \mathfrak{C} \) be an arbitrary \( \mathcal{L} \)-structure.
    Let \( \mathfrak{c} \) be any variable assignment function of \( \mathcal{L} \) under \( \mathfrak{C} \).
    Suppose that \( \folwffsatisfaction{\mathfrak{C}}{\mleft( \varphi \,\,{\lor}\,\, \theta_{b} \mright)}{\mathfrak{c}} \);
    that is, either \( \folwffsatisfaction{\mathfrak{C}}{\varphi}{\mathfrak{c}} \) or
    \( \folwffsatisfaction{\mathfrak{C}}{\theta_{b}}{\mathfrak{c}} \) is true.
    If \( \folwffsatisfaction{\mathfrak{C}}{\varphi}{\mathfrak{c}} \), then clearly \( \folwffsatisfaction{\mathfrak{C}}{\lambda_{2}}{\mathfrak{c}} \)
    and \( \folwffsatisfaction{\mathfrak{C}}{\lambda_{3}}{\mathfrak{c}} \) are both true, which also means that
    \( \folwffsatisfaction{\mathfrak{C}}{\mleft( \lambda_{2} \,\,{\land}\,\, \lambda_{3} \mright)}{\mathfrak{c}} \) holds.
    On the other hand, if \( \folwffsatisfaction{\mathfrak{C}}{\theta_{b}}{\mathfrak{c}} \) is true, then necessarily
    \( \folwffsatisfaction{\mathfrak{C}}{\psi}{\mathfrak{c}} \) and \( \folwffsatisfaction{\mathfrak{C}}{\phi}{\mathfrak{c}} \) are both true.
    Once again, this implies that \( \folwffsatisfaction{\mathfrak{C}}{\lambda_{2}}{\mathfrak{c}} \) and \( \folwffsatisfaction{\mathfrak{C}}{\lambda_{3}}{\mathfrak{c}} \)
    are both true, and thus \( \folwffsatisfaction{\mathfrak{C}}{\mleft( \lambda_{2} \,\,{\land}\,\, \lambda_{3} \mright)}{\mathfrak{c}} \) holds.
    It has been shown that \( \folwffsatisfaction{\mathfrak{C}}{\mleft( \lambda_{2} \,\,{\land}\,\, \lambda_{3} \mright)}{\mathfrak{c}} \)
    follows from \( \folwffsatisfaction{\mathfrak{C}}{\mleft( \varphi \,\,{\lor}\,\, \theta_{b} \mright)}{\mathfrak{c}} \) in all possible cases.
    We then conclude \( \varphi \,\,{\lor}\,\, \theta_{b} \models \lambda_{2} \,\,{\land}\,\, \lambda_{3} \) from the fact that \( \mathfrak{C} \)
    and \( \mathfrak{c} \) are both arbitrarily selected.

    \vspace{0.3cm}
    \noindent
    Finally, for the converse, we let \( \mathfrak{D} \) and \( \mathfrak{d} \) be an arbitrarily selected \( \mathcal{L} \)-structure
    and variable assignment function respectively.
    Suppose that \( \folwffsatisfaction{\mathfrak{D}}{\mleft( \lambda_{2} \,\,{\land}\,\, \lambda_{3} \mright)}{\mathfrak{d}} \);
    i.e., \( \folwffsatisfaction{\mathfrak{D}}{\lambda_{2}}{\mathfrak{d}} \) and \( \folwffsatisfaction{\mathfrak{D}}{\lambda_{3}}{\mathfrak{d}} \)
    both hold. By the \aref{rule-of-inference:law-of-excluded-middle}{Law of Excluded Middle}, either
    \( \folwffsatisfaction{\mathfrak{D}}{\varphi}{\mathfrak{d}} \) or \( \folwffnonsatisfaction{\mathfrak{D}}{\varphi}{\mathfrak{d}} \).
    If \( \folwffsatisfaction{\mathfrak{D}}{\varphi}{\mathfrak{d}} \), then clearly
    \( \folwffsatisfaction{\mathfrak{D}}{\mleft( \varphi \,\,{\lor}\,\, \theta_{b} \mright)}{\mathfrak{d}} \) is also true.
    On the other hand, if \( \folwffnonsatisfaction{\mathfrak{D}}{\varphi}{\mathfrak{d}} \), then both
    \( \folwffsatisfaction{\mathfrak{D}}{\psi}{\mathfrak{d}} \) and \( \folwffsatisfaction{\mathfrak{D}}{\phi}{\mathfrak{d}} \);
    otherwise, \( \folwffsatisfaction{\mathfrak{D}}{\lambda_{2}}{\mathfrak{d}} \) and \( \folwffsatisfaction{\mathfrak{D}}{\lambda_{3}}{\mathfrak{d}} \)
    cannot both be true as required by our supposition.
    Therefore, \( \folwffsatisfaction{\mathfrak{D}}{\theta_{b}}{\mathfrak{d}} \) must be true,
    which in turn implies \( \folwffsatisfaction{\mathfrak{D}}{\mleft( \varphi \,\,{\lor}\,\, \theta_{b} \mright)}{\mathfrak{d}} \).
    Evidently, \( \folwffsatisfaction{\mathfrak{D}}{\mleft( \varphi \,\,{\lor}\,\, \theta_{b} \mright)}{\mathfrak{d}} \) follows from
    \( \folwffsatisfaction{\mathfrak{D}}{\mleft( \lambda_{2} \,\,{\land}\,\, \lambda_{3} \mright)}{\mathfrak{d}} \) in all possible scenarios.
    Since \( \mathfrak{D} \) and \( \mathfrak{d} \) are arbitrarily selected, we conclude that
    \( \lambda_{2} \,\,{\land}\,\, \lambda_{3} \models \varphi \,\,{\lor}\,\, \theta_{b} \).
\end{proof}

\begin{thm}[Negation]
    \phantomsection \label{logical-equivalence:two-formulas-are-logically-equivalent-iff-their-negations-are-logically-equivalent}
    For all \( \mathcal{L} \)-formulas \( \varphi \) and \( \psi \), \( \varphi \equiv \psi \) if and only if \( \lnot \, \varphi \equiv \lnot \, \psi \).
\end{thm}

\begin{proof}
    Suppose that \( \varphi \equiv \psi \). It follows immediately from \cref{thm:first-order-language-equivalent-formulations-of-logical-consequence-and-logical-equivalence}
    that \( \models \mleft( \varphi {\iff} \psi \mright) \) holds.
    Therefore, for every structure \( \mathfrak{A} \) and every variable assignment function \( \mathfrak{a} \),
    either \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \) and \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \)
    or \( \folwffnonsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \) and \( \folwffnonsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \);
    i.e., either \( \folwffnonsatisfaction{\mathfrak{A}}{\mleft( \lnot \, \varphi \mright)}{\mathfrak{a}} \)
    and \( \folwffnonsatisfaction{\mathfrak{A}}{\mleft( \lnot \, \psi \mright)}{\mathfrak{a}} \) or
    \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \lnot \, \varphi \mright)}{\mathfrak{a}} \)
    and \( \folwffsatisfaction{\mathfrak{A}}{\mleft(\lnot \, \psi \mright)}{\mathfrak{a}} \).
    This means that \( \lnot \, \varphi {\iff} \lnot \, \psi \) is a tautology and thus \( \lnot \, \varphi \equiv \lnot \, \psi \)
    by \cref{thm:first-order-language-equivalent-formulations-of-logical-consequence-and-logical-equivalence}.
\end{proof}

\begin{thm}
    \phantomsection \label{logical-equivalence:replacing-logically-equivalent-conjuncts-or-disjuncts-yields-logically-equivalent-conjunctions-and-disjunctions}
    If \( \varphi \) and \( \varphi^{\prime} \) are \( \mathcal{L} \)-formulas such that \( \varphi \equiv \varphi^{\prime} \),
    then the following hold:
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item \( \varphi \,\,{\land}\,\, \psi \equiv \varphi^{\prime} \,\,{\land}\,\, \psi \) for any formula \( \psi \) \phantomsection \label{enum:conjunct1}
        \item \( \psi \,\,{\land}\,\, \varphi \equiv \psi \,\,{\land}\,\, \varphi^{\prime} \) for any formula \( \psi \)\phantomsection \label{enum:conjunct2}
        \item \( \varphi \,\,{\lor}\,\, \psi \equiv \varphi^{\prime} \,\,{\lor}\,\, \psi \) for any formula \( \psi \) \phantomsection \label{enum:disjunct1}
        \item \( \psi \,\,{\lor}\,\, \varphi \equiv \psi \,\,{\lor}\,\, \varphi^{\prime} \) for any formula \( \psi \) \phantomsection \label{enum:disjunct2}
    \end{enumerate}
\end{thm}

\begin{proof}
    For \ref{enum:conjunct1}, let \( \mathfrak{A} \) be a structure and \( \mathfrak{a} \) be a variable assignment function such that
    \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \varphi \,\,{\land}\,\, \psi \mright)}{\mathfrak{a}} \).
    By \cref{dfn:first-order-language-satisfaction-of-formulas}, it follows that \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \)
    and \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \) are both true.
    Since \( \varphi \equiv \varphi^{\prime} \), \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\mathfrak{a}} \)
    implies \( \folwffsatisfaction{\mathfrak{A}}{\varphi^{\prime}}{\mathfrak{a}} \).
    In other words, \( \folwffsatisfaction{\mathfrak{A}}{\varphi^{\prime}}{\mathfrak{a}} \) and \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \) both hold;
    therefore, \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \varphi^{\prime} \,\,{\land}\,\, \psi \mright)}{\mathfrak{a}} \)
    by \cref{dfn:first-order-language-satisfaction-of-formulas}.
    Since \( \mathfrak{A} \) and \( \mathfrak{a} \) are arbitrary, it follows that \( \varphi \,\,{\land}\,\, \psi \models \varphi^{\prime} \,\,{\land}\,\, \psi \).
    To prove the converse, we let \( \mathfrak{B} \) be a structure and \( \mathfrak{b} \) be a variable assignment function such that
    \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \varphi^{\prime} \,\,{\land}\,\, \psi \mright)}{\mathfrak{b}} \). Again, this means that
    \( \folwffsatisfaction{\mathfrak{B}}{\varphi^{\prime}}{\mathfrak{b}} \) and \( \folwffsatisfaction{\mathfrak{B}}{\psi}{\mathfrak{b}} \) are both true;
    and \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{\mathfrak{b}} \) follows immediately from \( \varphi \equiv \varphi^{\prime} \).
    Thus \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \varphi \,\,{\land}\,\, \psi \mright)}{\mathfrak{b}} \) follows from
    \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \varphi^{\prime} \,\,{\land}\,\, \psi \mright)}{\mathfrak{b}} \) for all \( \mathfrak{B} \) and \( \mathfrak{b} \);
    i.e., \( \varphi^{\prime} \,\,{\land}\,\, \psi \models \varphi \,\,{\land}\,\, \psi \).

    \vspace{0.3cm}
    \noindent
    Next, we proceed to establish \ref{enum:conjunct2} using \ref{enum:conjunct1}.
    We start by applying \cref{logical-equivalence:commutativity-of-conjunctions-and-disjunctions} to the formula \( \psi \,\,{\land}\,\, \varphi \).
    Thus, \( \psi \,\,{\land}\,\, \varphi \equiv \varphi \,\,{\land}\,\, \psi \). But \( \varphi \equiv \varphi^{\prime} \), so we have
    \( \varphi \,\,{\land}\,\, \psi \equiv \varphi^{\prime} \,\,{\land}\,\, \psi \) by \ref{enum:conjunct1}.
    Applying \cref{logical-equivalence:commutativity-of-conjunctions-and-disjunctions} once again yields
    \( \varphi \,\,{\land}\,\, \psi \equiv \varphi^{\prime} \,\,{\land}\,\, \psi \equiv \psi \,\,{\land}\,\, \varphi^{\prime} \).
    Finally, we conclude by \cref{thm:first-order-language-transitivity-of-logical-consequence-and-equivalence} that
    \( \psi \,\,{\land}\,\, \varphi \equiv \psi \,\,{\land}\,\, \varphi^{\prime} \), which proves \ref{enum:conjunct2}.

    \vspace{0.3cm}
    \noindent
    For \ref{enum:disjunct1}, suppose that \( \folwffsatisfaction{\mathfrak{C}}{\mleft( \varphi \,\,{\lor}\,\, \psi \mright)}{\mathfrak{c}} \) for some structure
    \( \mathfrak{C} \) and some variable assignment function \( \mathfrak{c} \).
    By \cref{dfn:first-order-language-satisfaction-of-formulas}, it follows that either \( \folwffsatisfaction{\mathfrak{C}}{\varphi}{\mathfrak{c}} \)
    or \( \folwffsatisfaction{\mathfrak{C}}{\psi}{\mathfrak{c}} \) is true.
    If \( \folwffsatisfaction{\mathfrak{C}}{\psi}{\mathfrak{c}} \), then \( \folwffsatisfaction{\mathfrak{C}}{\mleft( \varphi^{\prime} \,\,{\lor}\,\, \psi \mright)}{\mathfrak{c}} \) is obviously true.
    On the other hand, if \( \folwffsatisfaction{\mathfrak{C}}{\varphi}{\mathfrak{c}} \), then \( \folwffsatisfaction{\mathfrak{C}}{\varphi^{\prime}}{\mathfrak{c}} \)
    follows immediately from the assumption that \( \varphi \equiv \varphi^{\prime} \). But if \( \folwffsatisfaction{\mathfrak{C}}{\varphi^{\prime}}{\mathfrak{c}} \),
    then \( \folwffsatisfaction{\mathfrak{C}}{\mleft( \varphi^{\prime} \,\,{\lor}\,\, \psi \mright)}{\mathfrak{c}} \) immediately follows from
    \cref{dfn:first-order-language-satisfaction-of-formulas}. Evidently, \( \folwffsatisfaction{\mathfrak{C}}{\mleft( \varphi^{\prime} \,\,{\lor}\,\, \psi \mright)}{\mathfrak{c}} \) is implied in all possibilities.
    Given that \( \mathfrak{C} \) and \( \mathfrak{c} \) are arbitrary, we can validly infer that \( \varphi \,\,{\lor}\,\, \psi \models \varphi^{\prime} \,\,{\lor}\,\, \psi \).
    The converse can be proven the same way.
    Suppose that \( \mathfrak{D} \) is a structure and \( \mathfrak{d} \) is a variable assignment function such that
    \( \folwffsatisfaction{\mathfrak{D}}{\mleft(\varphi^{\prime} \,\,{\lor}\,\, \psi\mright)}{\mathfrak{d}} \).
    By \cref{dfn:first-order-language-satisfaction-of-formulas}, it follows that either \( \folwffsatisfaction{\mathfrak{D}}{\varphi^{\prime}}{\mathfrak{d}} \)
    or \( \folwffsatisfaction{\mathfrak{D}}{\psi}{\mathfrak{d}} \) is true.
    Since \( \varphi \equiv \varphi^{\prime} \), \( \folwffsatisfaction{\mathfrak{D}}{\varphi^{\prime}}{\mathfrak{d}} \) necessarily implies
    \( \folwffsatisfaction{\mathfrak{D}}{\varphi}{\mathfrak{d}} \), which in turn implies \( \folwffsatisfaction{\mathfrak{D}}{\mleft( \varphi \,\,{\lor}\,\, \psi \mright)}{\mathfrak{d}} \).
    Obviously, if \( \folwffsatisfaction{\mathfrak{D}}{\psi}{\mathfrak{d}} \), then \( \folwffsatisfaction{\mathfrak{D}}{\mleft( \varphi \,\,{\lor}\,\, \psi \mright)}{\mathfrak{d}} \) also follows.

    \vspace{0.3cm}
    \noindent
    The proof for \ref{enum:disjunct2} is almost identical to that of \ref{enum:conjunct2}; simply replace the symbol \( \land  \) with the symbol \( \lor  \).
\end{proof}

\begin{thm}
    \phantomsection \label{logical-equivalence:replacing-logically-equivalent-formulas-in-quantified-formulas-yields-logically-equivalent-formulas}
    Let \( \varphi \) and \( \psi \) be \( \mathcal{L} \)-formulas and \( x \) be an arbitrary \( \mathcal{L} \)-variable.
    If \( \varphi \equiv \psi \), then the following hold:
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item \( \forall x \, \varphi \equiv \forall x \, \psi \)  \phantomsection \label{enum:equivalent-universal-formulas}
        \item \( \exists x \, \varphi \equiv \exists x \, \psi \) \phantomsection \label{enum:equivalent-existential-formulas}
    \end{enumerate}
\end{thm}

\begin{proof}
    Let \( \phi \coloneq \forall x \, \varphi \) and \( \theta \coloneq \forall x \, \psi  \).
    Furthermore, let \( \mathfrak{A} \) be an \( \mathcal{L} \)-structure and \( \mathfrak{a} \) be a variable assignment function such that
    \( \folwffsatisfaction{\mathfrak{A}}{\phi}{\mathfrak{a}} \).
    Now let \( e \in \abs{\mathfrak{A}} \) be arbitrary.
    Since \( \folwffsatisfaction{\mathfrak{A}}{\phi}{\mathfrak{a}} \), it follows from \cref{dfn:first-order-language-satisfaction-of-formulas}
    that \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e}} \).
    Clearly, \( \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e} \) is also a variable assignment function of \( \mathcal{L} \)
    under \( \mathfrak{A} \); and since \( \varphi \models \psi \),
    it follows that \( \folwffsatisfaction{\mathfrak{A}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e} } \) must also be true.
    Given that \( e \) is an arbitrarily selected element of \( \abs{\mathfrak{A}} \), we can validly infer that
    \( \folwffsatisfaction{\mathfrak{A}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e} } \) holds for all \( e \in \abs{\mathfrak{A}} \).
    Thus, \( \folwffsatisfaction{\mathfrak{A}}{\theta}{\mathfrak{a}} \) by \cref{dfn:first-order-language-satisfaction-of-formulas}.
    In other words, \( \folwffsatisfaction{\mathfrak{A}}{\phi}{\mathfrak{a}} \) implies \( \folwffsatisfaction{\mathfrak{A}}{\theta}{\mathfrak{a}} \)
    for any structure \( \mathfrak{A} \) and any variable assignment function \( \mathfrak{a} \); i.e., \( \phi \models \theta \).
    The converse can be proven with the same argument but with the roles of \( \varphi \) and \( \psi \) swapped.

    \vspace{0.3cm}
    \noindent
    To prove \ref{enum:equivalent-existential-formulas}, we follow the same procedure as above.
    Let \( \phi \coloneq \exists x \, \varphi \) and \( \theta \coloneq \exists x \, \psi \).
    Let \( \mathfrak{B} \) be any structure and \( \mathfrak{b} \) be any variable assignment function
    such that \( \folwffsatisfaction{\mathfrak{B}}{\phi}{\mathfrak{b}} \); that is, there exists some
    \( e \in \abs{\mathfrak{A}} \) such that \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e} } \) holds.
    But \( \varphi \models \psi \) by assumption, which means that
    \( \folwffsatisfaction{\mathfrak{B}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e} } \) also holds.
    Therefore, \( \folwffsatisfaction{\mathfrak{B}}{\theta}{\mathfrak{b}} \) must be true by \cref{dfn:first-order-language-satisfaction-of-formulas}.
    Since \( \mathfrak{B} \) and \( \mathfrak{b} \) are arbitrary, \( \phi \models \theta \) follows readilly from
    \cref{dfn:first-order-language-logical-consequence}.
    The converse can be proven in the same way but with the roles of \( \varphi \) and \( \psi \) interchanged.
\end{proof}

\section{Applications}
\noindent
We now have enough theoretical foundation to start proving many of the familiar results from logic.

\begin{thm}
    \phantomsection \label{logical-equivalence:the-order-of-quantifiers-of-the-same-type-does-not-matter}
    Let \( \varphi \) denote an \( \mathcal{L} \)-formula. Let \( x \) and \( y \) denote \( \mathcal{L} \)-variables.
    Then the following hold:
    \begin{enumerate}[topsep=2pt,label=\textup{\texttt{(\alph*)}}]
        \item \( \forall x \, \forall y \, \varphi \equiv \forall y \, \forall x \, \varphi \) \phantomsection \label{enum:swapping-universal-quantifiers}
        \item \( \exists x \, \exists y \, \varphi \equiv \exists y \, \exists x \, \varphi \) \phantomsection \label{enum:swapping-existential-quantifiers}
    \end{enumerate}
\end{thm}

\begin{proof}
    Let \( \psi \coloneq \forall y \, \varphi \) and \( \phi \coloneq \forall x \, \varphi \).
    To prove \ref{enum:swapping-universal-quantifiers}, we must demonstrate that \( \forall x \, \psi \equiv \forall y \, \phi \).
    Let \( \mathfrak{A} \) be any \( \mathcal{L} \)-structure. Let \( \mathfrak{a} \) be any variable assignment function of \( \mathcal{L} \)
    under \( \mathfrak{A} \).
    Suppose that \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \forall x \, \psi \mright)}{\mathfrak{a}} \).
    Let \( e_{0} \in \abs{\mathfrak{A}} \) be arbitrary.
    We now examine whether \( \folwffsatisfaction{\mathfrak{A}}{\phi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{y}{e_{0}} } \) holds.
    Let \( e_{1} \in \abs{\mathfrak{A}} \) be arbitrary. Consider the variable assignment function
    \( \folmodifiedvariableassignmentfunction{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{y}{e_{0}} }{x}{e_{1}} \).
    By \cref{dfn:first-order-language-modified-variable-assignment-function}, it follows that
    \begin{equation}
        \label{eq:conditional-values-for-assignment-function-a}
        \folmodifiedvariableassignmentfunction{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{y}{e_{0}} }{x}{e_{1}} =
        \begin{cases}
            \restrictedfunctionsmall{\mathfrak{a}}{ \folvars{\mathcal{L}} - \mleft\{ x \mright\} } \union \big\{ \orderedpair{x}{e_{1}} \big\}
            \ & \text{\normalfont if} \ x = y \\
            \restrictedfunctionsmall{\mathfrak{a}}{ \folvars{\mathcal{L}} - \mleft\{ x, y \mright\} } \union \big\{ \orderedpair{x}{e_{1}}, \orderedpair{y}{e_{0}} \big\}
            \ & \text{\normalfont if} \ x \neq y
        \end{cases}
    \end{equation}
    If \( x = y \), then clearly
    \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\folmodifiedvariableassignmentfunction{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{y}{e_{0}} }{x}{e_{1}}} \)
    because our assumption that \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \forall x \, \psi \mright)}{\mathfrak{a}} \) requires
    \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e} } \) to be true for all \( e \in \abs{\mathfrak{A}} \)
    (note that this is because \( x \) and \( y \) denote the same variable symbol).
    Since \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{\folmodifiedvariableassignmentfunction{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{y}{e_{0}} }{x}{e_{1}}} \)
    for an arbitrary \( e_{1} \in \abs{\mathfrak{A}} \),
    we can validly infer that \( \folwffsatisfaction{\mathfrak{A}}{\phi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{y}{e_{0}} } \).
    Since \( e_{0} \) is also arbitrary, we further conclude that \( \folwffsatisfaction{\mathfrak{A}}{\phi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{y}{e} } \)
    for all \( e \in \abs{\mathfrak{A}} \). Therefore, \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \forall y \, \phi \mright)}{\mathfrak{a}} \)
    by \cref{dfn:first-order-language-satisfaction-of-formulas}.

    \vspace{0.3cm}
    \noindent
    If \( x \neq y \), then we infer from \cref{dfn:first-order-language-modified-variable-assignment-function} and \eqref{eq:conditional-values-for-assignment-function-a} that
    \begin{equation}
        \label{eq:swapped-xy-yields-the-same-modified-assignment-function-a}
        \folmodifiedvariableassignmentfunction{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{y}{e_{0}} }{x}{e_{1}} =
        \folmodifiedvariableassignmentfunction{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e_{1}} }{y}{e_{0}}
    \end{equation}
    Since \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \forall x \, \psi \mright)}{\mathfrak{a}} \) by assumption,
    it follows that \( \folwffsatisfaction{\mathfrak{A}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{d} } \) for all \( d \in \abs{\mathfrak{A}} \).
    Since \( e_{1} \in \abs{\mathfrak{A}} \), it follows that \( \folwffsatisfaction{\mathfrak{A}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e_{1}} }\)
    must hold.
    By \cref{dfn:first-order-language-satisfaction-of-formulas}, it follows that
    \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{ \folmodifiedvariableassignmentfunction{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e_{1}} }{y}{e} } \)
    for all \( e \in \abs{\mathfrak{A}} \). Given that \( e_{0} \in \abs{\mathfrak{A}} \), we thus have
    \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{ \folmodifiedvariableassignmentfunction{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e_{1}} }{y}{e_{0}} } \),
    and by \eqref{eq:swapped-xy-yields-the-same-modified-assignment-function-a}, it follows that
    \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{ \folmodifiedvariableassignmentfunction{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{y}{e_{0}} }{x}{e_{1}} } \).
    Since \( e_{1} \) is an arbitrary element of \( \abs{\mathfrak{A}} \), we infer that
    \( \folwffsatisfaction{\mathfrak{A}}{\phi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{y}{e_{0}} } \).
    But \( e_{0} \) is also arbitrary, so necessarily \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \forall y \, \phi \mright)}{\mathfrak{a}} \).

    \vspace{0.3cm}
    \noindent
    Thus, \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \forall y \, \phi \mright)}{\mathfrak{a}} \) follows from
    \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \forall x \, \psi \mright)}{\mathfrak{a}} \) in all possible cases.
    Since \( \mathfrak{A} \) and \( \mathfrak{a} \) are arbitrary, we have \( \forall x \, \psi \models \forall y \, \phi \).

    \vspace{0.3cm}
    \noindent
    For the converse, let \( \mathfrak{B} \) be an \( \mathcal{L} \)-structure and \( \mathfrak{b} \) be a variable assignment function
    on \( \mathcal{L} \) under \( \mathfrak{B} \). Suppose that \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \forall y \, \phi \mright)}{\mathfrak{b}} \).
    Let \( e_{0} \in \abs{\mathfrak{B}} \) be arbitrary.
    We now verify that \( \folwffsatisfaction{\mathfrak{B}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{0}} } \) is indeed true.
    Let \( e_{1} \in \abs{\mathfrak{B}} \) be arbitrary. Now consider the variable assignment function
    \( \folmodifiedvariableassignmentfunction{\folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{0}}}{y}{e_{1}} \).
    By \cref{dfn:first-order-language-modified-variable-assignment-function}, we have
    \begin{equation}
        \label{eq:conditional-values-for-assignment-function-b}
        \folmodifiedvariableassignmentfunction{\folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{0}}}{y}{e_{1}} =
        \begin{cases}
            \restrictedfunctionsmall{\mathfrak{b}}{ \folvars{\mathcal{L}} - \mleft\{ x \mright\}} \union \big\{ \orderedpair{x}{e_{1}} \big\}
            &\ \text{\normalfont if} \ x = y\\
            \restrictedfunctionsmall{\mathfrak{b}}{ \folvars{\mathcal{L}} - \mleft\{ x, y \mright\} } \union \big\{ \orderedpair{x}{e_{0}}, \orderedpair{y}{e_{1}} \big\}
            &\ \text{\normalfont if} \ x \neq y
        \end{cases}
    \end{equation}
    If \( x = y \), then \( \folmodifiedvariableassignmentfunction{\folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{0}}}{y}{e_{1}}
    = \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{1}} \).
    But \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \forall y \, \phi \mright)}{\mathfrak{b}} \) by assumption, which means that
    \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{d} } \) for all \( d \in \abs{\mathfrak{B}} \).
    Since \( e_{1} \in \abs{\mathfrak{B}} \), it follows that
    \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{\folmodifiedvariableassignmentfunction{\folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{0}}}{y}{e_{1}}} \).
    Given that \( e_{1} \) is arbitrary, we thus have
    \( \folwffsatisfaction{\mathfrak{B}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{0}} } \);
    and consequently \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \forall x \, \psi \mright)}{\mathfrak{b}} \) because
    \( e_{0} \) is also arbitrary.

    \vspace{0.3cm}
    \noindent
    If \( x \neq y \), then it is clear from \cref{dfn:first-order-language-modified-variable-assignment-function} and \eqref{eq:conditional-values-for-assignment-function-b}
    that
    \begin{equation}
        \label{eq:swapped-xy-yields-the-same-modified-assignment-function-b}
        \folmodifiedvariableassignmentfunction{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{y}{e_{1}} }{x}{e_{0}} =
        \folmodifiedvariableassignmentfunction{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{0}} }{y}{e_{1}}
    \end{equation}
    Since \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \forall y \, \phi \mright)}{\mathfrak{b}} \) by assumption,
    it follows that \( \folwffsatisfaction{\mathfrak{B}}{\phi}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{y}{d} } \) for all \( d \in \abs{\mathfrak{B}} \).
    Since \( e_{1} \in \abs{\mathfrak{B}} \), it follows that
    \( \folwffsatisfaction{\mathfrak{B}}{\phi}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{y}{e_{1}} } \).
    In other words, \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{\folmodifiedvariableassignmentfunction{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{y}{e_{1}}}{x}{f}} \)
    for all \( f \in \abs{\mathfrak{B}} \). But \( e_{0} \in \abs{\mathfrak{B}} \), so necessarily
    \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{\folmodifiedvariableassignmentfunction{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{y}{e_{1}}}{x}{e_{0}}} \).
    By \eqref{eq:swapped-xy-yields-the-same-modified-assignment-function-b}, it follows that
    \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{\folmodifiedvariableassignmentfunction{\folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{0}}}{y}{e_{1}}} \)
    is also true. Given that \( e_{1} \) is arbitrary, it follows that
    \( \folwffsatisfaction{\mathfrak{B}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{0}} } \).
    But \( e_{0} \) is also arbitrary, so we can generalize once again to infer \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \forall x \, \psi \mright)}{\mathfrak{b}} \).

    \vspace{0.3cm}
    \noindent
    Evidently, \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \forall x \, \psi \mright)}{\mathfrak{b}} \) is implied by 
    the assumption \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \forall y \, \phi \mright)}{\mathfrak{b}} \) in all possible scenarios.
    Since \( \mathfrak{B} \) and \( \mathfrak{b} \) are arbitrary, we conclude that \( \forall y \, \phi \models \forall x \, \psi \).

    \vspace{0.3cm}
    \noindent
    Part \ref{enum:swapping-existential-quantifiers} is a lot easier to prove. Again, we begin by redefining our notation.
    Let \( \psi \coloneq \exists y \, \varphi \) and \( \phi \coloneq \exists x \, \varphi \).
    The goal, therefore, is to show that \( \exists x \, \psi \equiv \exists y \, \phi \).

    \vspace{0.3cm}
    \noindent
    Let \( \mathfrak{C} \) be an arbitrary \( \mathcal{L} \)-structure. Let \( \mathfrak{c} \) be any variable assignment function of \( \mathcal{L} \)
    under \( \mathfrak{C} \). Suppose that \( \folwffsatisfaction{\mathfrak{C}}{\mleft( \exists x \, \psi \mright)}{\mathfrak{c}} \);
    that is, there exists some \( e_{0} \in \abs{\mathfrak{C}} \) such that
    \( \folwffsatisfaction{\mathfrak{C}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{c}}{x}{e_{0}}}\).
    Since \( \psi \coloneq \exists y \, \varphi \), it follows that there exists some \( e_{1} \in \abs{\mathfrak{C}} \) such that
    \begin{equation*}
        \folwffsatisfaction{\mathfrak{C}}{\varphi}{\folmodifiedvariableassignmentfunction{\folmodifiedvariableassignmentfunction{\mathfrak{c}}{x}{e_{0}}}{y}{e_{1}}}
    \end{equation*}
    Clearly, this also means that
    \( \folwffsatisfaction{\mathfrak{C}}{\varphi}{\folmodifiedvariableassignmentfunction{\folmodifiedvariableassignmentfunction{\mathfrak{c}}{y}{e_{1}}}{x}{e_{0}}} \)
    regardless of whether \( x \) and \( y \) denote the same \( \mathcal{L} \)-variable.
    Thus, \( \folwffsatisfaction{\mathfrak{C}}{\phi}{ \folmodifiedvariableassignmentfunction{\mathfrak{c}}{y}{e_{1}} } \) and consequently,
    \( \folwffsatisfaction{\mathfrak{C}}{\mleft( \exists y \, \phi \mright)}{\mathfrak{c}} \).
    Since \( \mathfrak{C} \) and \( \mathfrak{c} \) are both arbitrary, we conclude that \( \exists x \, \psi \models \exists y \, \phi \).
    The proof for the converse is nearly identical, so we will omit it for brevity.
\end{proof}

\begin{thm}[Substitution Theorem]
    \phantomsection \label{logical-equivalence:substitution-theorem}
    Let \( \psi \), \( \psi^{\prime} \), and \( \varphi \) denote \( \mathcal{L} \)-formulas.
    If \( \psi \folsubformula \varphi \) and \( \psi \equiv \psi^{\prime} \),
    then
    \begin{equation*}
        \varphi \equiv \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}}
    \end{equation*}
\end{thm}

\begin{proof}
    \aref{cor:first-order-language-every-formula-has-finite-complexity}{Since every \( \mathcal{L} \)-formula has a finite complexity},
    we can prove the theorem inductively. We begin by defining a statement \( P(n) \) as follows: for all \( \mathcal{L} \)-formula
    \( \varphi \) such that \( \folcomplexityofformula{\varphi} = n \), \( \varphi \equiv \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} \)
    for all \( \psi \folsubformula \varphi \). Let \( n \in \naturalset \) be arbitrary and \( \varphi \) be any \( \mathcal{L} \)-formula with complexity
    \( n \). We then proceed to make the following induction hypothesis.

    \vspace{0.3cm}
    \noindent
    \underline{\scshape Induction Hypothesis}: \( P(k) \) is true for all \( k < n \).

    \vspace{0.3cm}
    \noindent
    Since \( n \in \naturalset \), there are two cases that we must consider: either \( n = 0 \) or \( n \neq 0 \).
    If \( n = 0 \), then \aref{cor:first-order-language-only-atoms-can-have-zero-complexity}{\( \varphi \) must be an atom of \( \mathcal{L} \)};
    that is, either \( \varphi \coloneq s = t \) where \( s \) and \( t \) are \( \mathcal{L} \)-terms or
    \( \varphi \coloneq P^{\mathcal{L}}_{i}(\tau) \) for some \( i \in \dom P^{\mathcal{L}} \) where \( \tau \) is a tuple of \( \mathcal{L} \)-terms.
    In either case, \( P(n) \) is vacuously true since \( \varphi \) has no subformulas.
    If \( n \neq 0 \), then \( \varphi \) must have one of the forms stated in
    \cref{cor:first-order-language-every-formula-with-non-zero-complexity-can-be-decomposed}.

    \vspace{0.3cm}
    \noindent
    {\scshape Case}: \( \varphi \coloneq \mleft( \phi \mright) \) or \( \varphi \coloneq \mleft[ \,\phi\, \mright] \) for some \( \mathcal{L} \)-formula \( \phi \).

    \vspace{0.1cm}
    \noindent
    Let \( \psi \) be any subformula of \( \varphi \) and \( \psi^{\prime} \) be any \( \mathcal{L} \)-formula such that \( \psi \equiv \psi^{\prime} \).
    By \cref{dfn:first-order-language-subformulas}, it follows that either \( \psi = \phi \) or \( \psi \folsubformula \phi \).
    If \( \psi = \phi \), then clearly \( \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} = \mleft( \psi^{\prime} \mright) \).
    Thus, we conclude that \( \varphi \equiv \mleft( \phi \mright) \equiv \phi \equiv \psi \equiv \psi^{\prime} \equiv \mleft( \psi^{\prime} \mright)
    \equiv \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} \)
    by \cref{logical-equivalence:parenthesized-formulas-are-logically-equivalent,thm:first-order-language-transitivity-of-logical-consequence-and-equivalence}.
    Similarly, if \( \psi \folsubformula \phi \), then clearly
    \( \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} = \mleft( \folsubformulasubstitution{\phi}{\psi}{\psi^{\prime}} \mright) \).
    Observe also that \( \folcomplexityofformula{\psi} < \folcomplexityofformula{\phi} < n \) by
    \cref{cor:first-order-language-the-complexity-of-a-subformula-is-less-than-the-complexity-of-the-formula}.
    Given that \( \psi \folsubformula \phi \) and \( \psi \equiv \psi^{\prime} \), it follows from the induction hypothesis,
    \( \varphi \equiv \mleft( \phi \mright) \equiv \phi \equiv \folsubformulasubstitution{\phi}{\psi}{\psi^{\prime}} \equiv
    \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} \).
    Since \( \varphi \), \( \psi \) and \( \psi^{\prime} \) are arbitrary, it follows that \( P(n) \) is true.

    \vspace{0.3cm}
    \noindent
    {\scshape Case}: \( \varphi \coloneq \lnot \, \phi \) for some \( \mathcal{L} \)-formula \( \phi \).
    
    \vspace{0.1cm}
    \noindent
    Let \( \psi \) be any subformula of \( \varphi \) and \( \psi^{\prime} \) be any \( \mathcal{L} \)-formula such that \( \psi \equiv \psi^{\prime} \).
    Given the syntactic form of \( \varphi \), it follows that either \( \psi = \phi \) or \( \psi \folsubformula \phi \).
    If \( \psi = \phi \), then \( \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} = \lnot \, \psi^{\prime} \).
    Since \( \psi \equiv \psi^{\prime} \), it follows from \cref{logical-equivalence:two-formulas-are-logically-equivalent-iff-their-negations-are-logically-equivalent}
    that \( \lnot \, \psi \equiv \lnot \, \psi^{\prime} \equiv \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} \).
    But \( \varphi \equiv \lnot \, \phi \equiv \lnot \, \psi \).
    Thus \( \varphi \equiv \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} \) by the \aref{enum:transitivity-of-equivalence}{transitivity of \( \equiv \)}.
    On the other hand, if \( \psi \folsubformula \phi \), then we have
    \( \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} \equiv \lnot \, \folsubformulasubstitution{\phi}{\psi}{\psi^{\prime}} \).
    Since \( \phi \) is a subformula of \( \varphi \), the induction hypothesis applies because \( \folcomplexityofformula{\phi} < n \).
    Thus, we have \( \phi \equiv \folsubformulasubstitution{\phi}{\psi}{\psi^{\prime}} \), which in turn implies
    \( \lnot \, \phi \equiv \lnot \, \folsubformulasubstitution{\phi}{\psi}{\psi^{\prime}} \) by
    \cref{logical-equivalence:two-formulas-are-logically-equivalent-iff-their-negations-are-logically-equivalent}.
    By the \aref{enum:transitivity-of-equivalence}{transitivity of \( \equiv \)}, we infer
    \( \varphi \equiv \lnot \, \phi \equiv \lnot \, \folsubformulasubstitution{\phi}{\psi}{\psi^{\prime}} \equiv \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} \).
    Since \( \varphi \), \( \psi \), and \( \psi^{\prime} \) are arbitrary, it follows that \( P(n) \) holds.

    \vspace{0.3cm}
    \noindent
    {\scshape Case}: \( \varphi \coloneq \phi \,\,{\land}\,\, \lambda \) for \( \mathcal{L} \)-formulas \( \phi \) and \( \lambda \).

    \vspace{0.1cm}
    \noindent
    Let \( \psi \) be any subformula of \( \varphi \) and \( \psi^{\prime} \) be any \( \mathcal{L} \)-formula such that \( \psi \equiv \psi^{\prime} \). There
    are a total of eight possibilities to consider.

    \vspace{0.3cm}
    \noindent
    \underline{\textbf{Subcase 1}}: \( \psi = \phi = \lambda \).

    \vspace{0.1cm}
    \noindent
    In this case, clearly \( \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}}
    = \psi^{\prime} \,\,{\land}\,\, \psi^{\prime} \).
    By \cref{logical-equivalence:idempotence-of-conjunctions-and-disjunctions}, we thus have
    \( \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} \equiv \psi^{\prime}\).
    Observe also that \( \varphi = \psi \,\,{\land}\,\, \psi \) so that \( \varphi \equiv \psi \).
    Since \( \psi \equiv \psi^{\prime} \), it follows from \cref{thm:first-order-language-transitivity-of-logical-consequence-and-equivalence}
    that \( \varphi \equiv \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} \).

    \vspace{0.3cm}
    \noindent
    \underline{\textbf{Subcase 2}}: \( \psi = \phi \) and \( \psi \folnotsubformula{\lambda} \).

    \vspace{0.1cm}
    \noindent
    Obviously, \( \varphi = \psi \,\,{\land}\,\, \lambda \) and \( \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} = \psi^{\prime} \,\,{\land}\,\, \lambda \).
    Since \( \psi \equiv \psi^{\prime} \), it follows from
    \cref{logical-equivalence:replacing-logically-equivalent-conjuncts-or-disjuncts-yields-logically-equivalent-conjunctions-and-disjunctions}
    that \( \varphi \equiv \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} \).

    \vspace{0.3cm}
    \noindent
    \underline{\textbf{Subcase 3}}: \( \psi = \phi \) and \( \psi \folsubformula{\lambda} \).

    \vspace{0.1cm}
    \noindent
    Again, \( \varphi = \psi \,\,{\land}\,\, \lambda \).
    However, \( \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} = \psi^{\prime} \,\,{\land}\,\, \folsubformulasubstitution{\lambda}{\psi}{\psi^{\prime}} \).
    Since \( \lambda \folsubformula \varphi \), it follows that \( \folcomplexityofformula{\lambda} < n \).
    Thus, it follows from the induction hypothesis that \( \lambda \equiv \folsubformulasubstitution{\lambda}{\psi}{\psi^{\prime}} \).
    By \cref{logical-equivalence:replacing-logically-equivalent-conjuncts-or-disjuncts-yields-logically-equivalent-conjunctions-and-disjunctions},
    we have \( \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} \equiv \psi^{\prime} \,\,{\land}\,\, \lambda \equiv \psi \,\,{\land}\,\, \lambda
    \equiv \varphi \).

    \vspace{0.3cm}
    \noindent
    \underline{\textbf{Subcase 4}}: \( \psi = \lambda \) and \( \psi \folnotsubformula{\phi} \).

    \vspace{0.1cm}
    \noindent
    In this case, we have \( \varphi =  \phi \,\,{\land}\,\, \psi \) and thus
    \( \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} = \phi \,\,{\land}\,\, \psi^{\prime} \).
    Since \( \psi \equiv \psi^{\prime} \) by assumption, we thus have \( \varphi \equiv \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} \)
    by \cref{logical-equivalence:replacing-logically-equivalent-conjuncts-or-disjuncts-yields-logically-equivalent-conjunctions-and-disjunctions}.

    \vspace{0.3cm}
    \noindent
    \underline{\textbf{Subcase 5}}: \( \psi = \lambda \) and \( \psi \folsubformula{\phi} \).

    \vspace{0.1cm}
    \noindent
    We have \( \varphi = \phi \,\,{\land}\,\, \psi \) and
    \( \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} = \folsubformulasubstitution{\phi}{\psi}{\psi^{\prime}} \,\,{\land}\,\, \psi^{\prime} \).
    Since \( \phi \folsubformula \varphi \), it follows that \( \folcomplexityofformula{\phi} < n \), and thus the induction hypothesis applies to \( \psi \).
    Thus, \( \phi \equiv \folsubformulasubstitution{\phi}{\psi}{\psi^{\prime}} \).
    By \cref{logical-equivalence:replacing-logically-equivalent-conjuncts-or-disjuncts-yields-logically-equivalent-conjunctions-and-disjunctions}, we infer that
    \( \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} \equiv \phi \,\,{\land}\,\, \psi^{\prime} \equiv \phi \,\,{\land}\,\, \psi \equiv \varphi \).

    \vspace{0.3cm}
    \noindent
    \underline{\textbf{Subcase 6}}: \( \psi \folsubformula \phi \) and \( \psi \folnotsubformula \lambda \).

    \vspace{0.1cm}
    \noindent
    Clearly \( \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} = \folsubformulasubstitution{\phi}{\psi}{\psi^{\prime}} \,\,{\land}\,\, \lambda \)
    and \( \phi \equiv \folsubformulasubstitution{\phi}{\psi}{\psi^{\prime}} \).
    \( \varphi \equiv \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} \) follows readilly from
    \cref{logical-equivalence:replacing-logically-equivalent-conjuncts-or-disjuncts-yields-logically-equivalent-conjunctions-and-disjunctions}.

    \vspace{0.3cm}
    \noindent
    \underline{\textbf{Subcase 7}}: \( \psi \folnotsubformula \phi \) and \( \psi \folsubformula \lambda \).
    
    \vspace{0.1cm}
    \noindent
    Same as previous subcase.

    \vspace{0.3cm}
    \noindent
    \underline{\textbf{Subcase 8}}: \( \psi \folsubformula \phi \) and \( \psi \folsubformula \lambda \).
    Replacing \( \psi \) with \( \psi^{\prime} \) in \( \varphi \) requires both \( \phi \) and \( \lambda \) to be replaced.
    More specifically, \( \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} =
    \folsubformulasubstitution{\phi}{\psi}{\psi^{\prime}} \,\,{\land}\,\,
    \folsubformulasubstitution{\lambda}{\psi}{\psi^{\prime}} \).
    Since \( \phi \) and \( \lambda \) are both subformulas of \( \varphi \), it follows that the induction hypothesis applies
    to the both of them.
    Thus, \( \varphi \equiv \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} \).

    \vspace{0.3cm}
    \noindent
    It has been demonstrated that \( \varphi \equiv \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} \) holds for all eight possibilities.
    Since \( \varphi \), \( \psi \), and \( \psi^{\prime} \) are arbitrary, we conclude that \( P(n) \) holds.

    \vspace{0.3cm}
    \noindent
    {\scshape Case:} \( \varphi \coloneq \phi \,\,{\lor}\,\, \lambda \) for \( \mathcal{L} \)-formulas \( \phi \) and \( \lambda \).

    \vspace{0.1cm}
    \noindent
    Exactly the same as the previous case.

    \vspace{0.3cm}
    \noindent
    {\scshape Case:} \( \varphi \coloneq \forall x \, \phi \) for some \( \mathcal{L} \)-formula \( \phi \) and some \( \mathcal{L} \)-variable \( x \).

    \vspace{0.1cm}
    \noindent
    Let \( \psi \) be any subformula of \( \varphi \).
    There are two possibilities to consider: \( \psi = \phi \) or \( \psi \folsubformula \phi \).
    If \( \psi = \phi \), then clearly \( \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} = \forall x \, \psi^{\prime} \) and
    \( \varphi \equiv \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} \) is merely a consequence of
    \cref{logical-equivalence:replacing-logically-equivalent-formulas-in-quantified-formulas-yields-logically-equivalent-formulas}.
    If \( \psi \folsubformula \phi \), then \( \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} =
    \forall x \, \folsubformulasubstitution{\phi}{\psi}{\psi^{\prime}} \).
    But \( \phi \) is a subformula of \( \varphi \), so the induction hypothesis applies to it;
    i.e., \( \phi \equiv \folsubformulasubstitution{\phi}{\psi}{\psi^{\prime}} \).
    Once again, we arrive at \( \varphi \equiv \folsubformulasubstitution{\varphi}{\psi}{\psi^{\prime}} \)
    by \cref{logical-equivalence:replacing-logically-equivalent-formulas-in-quantified-formulas-yields-logically-equivalent-formulas}.
    Since \( \varphi \), \( \psi \), and \( \psi^{\prime} \) are arbitrary, it follows that \( P(n) \) holds.

    \vspace{0.3cm}
    \noindent
    {\scshape Case:} \( \varphi \coloneq \exists x \, \phi \) for some \( \mathcal{L} \)-formula \( \phi \) and some \( \mathcal{L} \)-variable \( x \).

    \vspace{0.1cm}
    \noindent
    Same argument as the previous case.

    \vspace{0.3cm}
    \noindent
    Thus, \( P(n) \) holds in all possible cases. Since \( n \in \naturalset \), the theorem follows immediately from strong induction.
\end{proof}

\begin{thm}
    \phantomsection \label{logical-equivalence:changing-free-variables-does-not-affect-universal-quantification}
    If \( \varphi \) is any \( \mathcal{L} \)-formula such that \( x \in \folfreevars{\varphi} \) and \( y \notin \folfreevars{\varphi} \)
    for \( \mathcal{L} \)-variables \( x \) and \( y \) and \( \varphi^{\prime} \) is the \( \mathcal{L} \)-formula obtained by replacing all
    free occurrences of \( x \) in \( \varphi \) with \( y \), then
    \begin{equation*}
        \forall x \, \varphi \equiv \forall y \, \varphi^{\prime}
    \end{equation*}
\end{thm}

\begin{proof}
    We proceed by induction on the complexity of formulas. First, we define a statement \( P(n) \) as follows:
    for all \( \mathcal{L} \)-formula \( \varphi \) such that \( \folcomplexityofformula{\varphi} = n \)
    and \( x \in \folfreevars{\varphi} \) for some \( \mathcal{L} \)-variable \( x \),
    if \( y \) is an \( \mathcal{L} \)-variable such that \( y \notin \folfreevars{\varphi} \) and
    \( \varphi^{\prime} \) is the formula obtained by replacing every free occurrence of \( x \) with \( y \),
    then \( \forall x \, \varphi \equiv \forall y \, \varphi^{\prime} \).
    Let \( n \in \naturalset \) be arbitrary. We now state the following hypothesis for \( n \).

    \vspace{0.3cm}
    \noindent
    \underline{\scshape Induction Hypothesis}: \( P(k) \) is true for all \( k < n \).

    \vspace{0.3cm}
    \noindent
    Let \( \varphi \) be any \( \mathcal{L} \)-formula with complexity \( n \) such that \( x \in \folfreevars{\varphi} \) and \( y \notin \folfreevars{\varphi} \) for
    \( \mathcal{L} \)-variables \( x \) and \( y \).
    Moreoever, let \( \varphi^{\prime} \) be the formula obtained after every free occurrence of \( x \) in \( \varphi \) has been replaced with \( y \).
    Since \( n \in \naturalset \), there are only two cases to consider: either \( n = 0 \) or \( n \neq 0 \).

    \vspace{0.3cm}
    \noindent
    {\scshape Case:} \( n = 0 \).

    \vspace{0.1cm}
    \noindent
    \aref{cor:first-order-language-only-atoms-can-have-zero-complexity}{Clearly \( \varphi \) must be an atom of \( \mathcal{L} \)}.
    Therefore, either \( \varphi \coloneq s = t \) where \( s \) and \( t \) are \( \mathcal{L} \)-terms or
    \( \varphi \coloneq P^{\mathcal{L}}_{i}(\tau) \) for some \( i \in \dom P^{\mathcal{L}} \) where \( \tau \) is a tuple of \( \mathcal{L} \)-terms
    such that \( \dom \tau = \Pi^{\mathcal{L}}(i) \).
    If \( \varphi \) takes the first form, then \( \varphi^{\prime} \coloneq s_{\ast} = t_{\ast} \) where
    \( s_{\ast} \) and \( t_{\ast} \) are the terms obtained after every occurrence of \( x \) in \( s \) and \( t \) respectively has been replaced with \( y \).
    We infer from this two non-trivial facts:
    \begin{equation}
        \begin{aligned}
            \folvars{s_{\ast}} &= \\
            \folvars{t_{\ast}} &=
        \end{aligned}
        \begin{aligned}
            &\mleft( \folvars{s} - \mleft\{ x \mright\} \mright) \union \mleft\{ y \mright\}\\
            &\mleft( \folvars{t} - \mleft\{ x \mright\} \mright) \union \mleft\{ y \mright\}
        \end{aligned}
    \end{equation}
\end{proof}

\begin{thm}
    \phantomsection \label{logical-equivalence:universal-quantifiers-distributes-over-conjunction}
    If \( \varphi \) and \( \psi \) are \( \mathcal{L} \)-formulas and \( x \) is any \( \mathcal{L} \)-variable,
    then
    \begin{equation*}
        \forall x \, \mleft[ \varphi \,\,{\land}\,\, \psi \mright] \equiv \forall x \, \varphi \,\,{\land}\,\, \forall x \, \psi
    \end{equation*}
\end{thm}

\begin{proof}
    Let \( \mathfrak{A} \) be an arbitrary \( \mathcal{L} \)-structure. Let \( \mathfrak{a} \) be any variable assignment function of \( \mathcal{L} \) under \( \mathfrak{A} \).
    Suppose that \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \forall x \, \mleft[ \varphi \,\,{\land}\,\, \psi \mright] \mright)}{\mathfrak{a}} \).
    Thus, we have by \cref{dfn:first-order-language-satisfaction-of-formulas}
    \begin{equation}
        \label{eq:forall-x-phi-and-psi-is-satisfied}
        \folwffsatisfaction{\mathfrak{A}}{\varphi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e} } \ \text{and}
        \ \folwffsatisfaction{\mathfrak{A}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e} } \ \text{for all} \ e \in \abs{\mathfrak{A}}
    \end{equation}
    Let \( e_{0} \in \abs{\mathfrak{A}} \) be arbitrary.
    Clearly, \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e_{0}} } \) holds by
    \eqref{eq:forall-x-phi-and-psi-is-satisfied}.
    Since \( e_{0} \in \abs{\mathfrak{A}} \) is arbitrary,
    it follows that \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e} } \) holds for
    all \( e \in \abs{\mathfrak{A}} \).
    By \cref{dfn:first-order-language-satisfaction-of-formulas}, it follows that \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \forall x \, \varphi \mright)}{\mathfrak{a}} \) holds.
    The same reasoning can be applied to show that \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \forall x \, \psi \mright)}{\mathfrak{a}} \) also holds.
    In other words, \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \forall x \, \varphi \,\,{\land}\,\, \forall x \, \psi \mright)}{\mathfrak{a}} \) follows from
    \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \forall x \, \mleft[ \varphi \,\,{\land}\,\, \psi \mright] \mright)}{\mathfrak{a}} \)
    for any arbitrary \( \mathfrak{A} \) and \( \mathfrak{a} \).
    Therefore, \( \forall x \, \mleft[ \varphi \,\,{\land}\,\, \psi \mright] \models \forall x \, \varphi \,\,{\land}\,\, \forall x \, \psi \).

    \vspace{0.3cm}
    \noindent
    Now we prove the converse.
    Let \( \mathfrak{B} \) be an arbitrary \( \mathcal{L} \)-structure. Let \( \mathfrak{b} \) be any variable assignment function of \( \mathcal{L} \) under \( \mathfrak{B} \).
    Suppose that \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \forall x \, \varphi \,\,{\land}\,\, \forall x \, \psi \mright)}{\mathfrak{b}} \).
    By \cref{dfn:first-order-language-satisfaction-of-formulas}, we have
    \begin{equation}
        \label{eq:forall-x-phi-and-forall-x-psi-are-satisfied}
        \folwffsatisfaction{\mathfrak{B}}{\mleft( \forall x \, \varphi \mright)}{\mathfrak{b}} \ \text{and}
        \ \folwffsatisfaction{\mathfrak{B}}{\mleft( \forall x \, \psi \mright)}{\mathfrak{b}}
    \end{equation}
    Let \( e_{1} \in \abs{\mathfrak{B}} \) be arbitrary.
    By \eqref{eq:forall-x-phi-and-forall-x-psi-are-satisfied} and \cref{dfn:first-order-language-satisfaction-of-formulas}, it follows that
    \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{1}}} \)
    and \( \folwffsatisfaction{\mathfrak{B}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{1}}} \)
    are both true;
    i.e., \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \varphi \,\,{\land}\,\, \psi \mright)}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{1}} } \) holds.
    Since \( e_{1} \) is arbitrary, we can validly infer
    \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \forall x \, \mleft[ \varphi \,\,{\land}\,\, \psi \mright] \mright)}{\mathfrak{b}} \).
    In other words, \( \forall x \, \varphi \,\,{\land}\,\, \forall x \, \psi \models \forall x \, \mleft[ \varphi \,\,{\land}\,\, \psi \mright] \).
\end{proof}

\begin{thm}
    \phantomsection \label{logical-equivalence:existential-quantifier-distributes-over-disjunction}
    If \( \varphi \) and \( \psi \) are \( \mathcal{L} \)-formulas and \( x \) is any \( \mathcal{L} \)-variable, then
    \( \exists x \, \mleft[ \varphi \,\,{\lor}\,\, \psi \mright] \equiv \exists x \, \varphi \,\,{\lor}\,\, \exists x \, \psi \).
\end{thm}

\begin{proof}
    Let \( \mathfrak{A} \) be an arbitrary \( \mathcal{L} \)-structure. Let \( \mathfrak{a} \) be any variable assignment function of \( \mathcal{L} \) under \( \mathfrak{A} \).
    Suppose that \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \exists x \, \mleft[ \varphi \,\,{\lor}\,\, \psi \mright] \mright)}{\mathfrak{a}} \); that is, there exists
    some \( e_{0} \in \abs{\mathfrak{A}} \) such that
    \( \folwffsatisfaction{\mathfrak{A}}{ \mleft( \varphi \,\,{\lor}\,\, \psi \mright) }{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e_{0}} } \) holds.
    By \cref{dfn:first-order-language-satisfaction-of-formulas}, it follows that either
    \begin{equation}
        \label{eq:either-phi-or-psi-is-satisfied-with-e0}
        \folwffsatisfaction{\mathfrak{A}}{\varphi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e_{0}} } \ \text{or}
        \ \folwffsatisfaction{\mathfrak{A}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e_{0}} }
    \end{equation}
    If \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e_{0}} } \), then
    clearly \( \folwffsatisfaction{\mathfrak{A}}{ \mleft( \exists x \, \varphi \mright) }{\mathfrak{a}} \).
    Similarly, if \( \folwffsatisfaction{\mathfrak{A}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e_{0}} } \),
    then \( \folwffsatisfaction{\mathfrak{A}}{ \mleft( \exists x \, \psi \mright) }{\mathfrak{a}} \).
    Thus, we infer by \aref{rule-of-inference:constructive-dilemma}{Constructive Dilemma} that either
    \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \exists x \, \varphi \mright)}{\mathfrak{a}} \) or
    \( \folwffsatisfaction{\mathfrak{A}}{ \mleft( \exists x \, \psi \mright) }{\mathfrak{a}} \) is true;
    i.e., \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \exists x \, \varphi \,\,{\lor}\,\, \exists x \, \psi \mright)}{\mathfrak{a}} \) holds.
    Given that \( \mathfrak{A} \) and \( \mathfrak{a} \) are arbitrary,
    we conclude that \( \exists x \, \mleft[ \varphi \,\,{\lor}\,\, \psi \mright] \models \exists x \, \varphi \,\,{\lor}\,\, \exists x \, \psi \).

    \vspace{0.3cm}
    \noindent
    Now we proceed to establish the converse.
    Let \( \mathfrak{B} \) denote any \( \mathcal{L} \)-structure.
    Let \( \mathfrak{b} \) denote any variable assignment function of \( \mathcal{L} \) under \( \mathfrak{B} \).
    Suppose that \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \exists x \, \varphi \,\,{\lor}\,\, \exists x \, \psi \mright)}{\mathfrak{b}} \) holds;
    that is, either
    \begin{equation}
        \label{eq:either-exists-x-varphi-or-exists-x-psi}
        \folwffsatisfaction{\mathfrak{B}}{\mleft( \exists x \, \varphi \mright)}{\mathfrak{b}} \ \text{or}
        \ \folwffsatisfaction{\mathfrak{B}}{\mleft( \exists x \, \psi \mright)}{\mathfrak{b}}
    \end{equation}
    If \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \exists x \, \varphi \mright)}{\mathfrak{b}} \), then there clearly exists
    an \( e_{1} \in \abs{\mathfrak{B}} \) such that \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{1}} } \).
    But this implies that
    \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \varphi \,\,{\lor}\,\, \psi \mright)}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{1}} } \)
    and thus \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \exists x \, \mleft[ \varphi \,\,{\lor}\,\, \psi \mright] \mright)}{\mathfrak{b}} \).
    Similarly, if \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \exists x \, \psi \mright)}{\mathfrak{b}} \), then there exists some \( e_{2} \in \abs{\mathfrak{B}} \)
    such that \( \folwffsatisfaction{\mathfrak{B}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{2}} } \).
    Again, this implies \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \varphi \,\,{\lor}\,\, \psi \mright)}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e_{2}} } \)
    and thus \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \exists x \, \mleft[ \varphi \,\,{\lor}\,\, \psi \mright] \mright)}{\mathfrak{b}} \).
    Clearly, \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \exists x \, \mleft[ \varphi \,\,{\lor}\,\, \psi \mright] \mright)}{\mathfrak{b}} \) follows from
    \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \exists x \, \varphi \,\,{\lor}\,\, \exists x \, \psi \mright)}{\mathfrak{b}} \) for any structure
    \( \mathfrak{B} \) and any variable assignment function \( \mathfrak{b} \);
    i.e., \( \exists x \, \varphi \,\,{\lor}\,\, \exists x \, \psi \models \exists x \, \mleft[ \varphi \,\,{\lor}\,\, \psi \mright] \).
\end{proof}

\begin{thm}
    \phantomsection \label{logical-equivalence:grouping-disjuncts-into-universally-quantified-formula}
    Let \( \psi \) and \( \varphi \) be \( \mathcal{L} \)-formulas.
    Furthermore, let \( x \) be an \( \mathcal{L} \)-variable such that \( x \notin \folfreevars{\psi} \).
    Then \( \psi \,\,{\lor}\,\, \forall x \, \varphi \equiv \forall x \, \mleft[ \psi \,\,{\lor}\,\, \varphi \mright] \)
\end{thm}

\begin{proof}
    We begin by organizing our notation.
    Let \( \phi \coloneq \forall x \, \varphi \), \( \theta \coloneq \psi \,\,{\lor}\,\, \varphi \), and \( \lambda \coloneq \psi \,\,{\lor}\,\, \phi \).
    The goal is to show that \( \lambda \equiv \forall x \, \theta \).
    Let \( \mathfrak{A} \) be any \( \mathcal{L} \)-structure and \( \mathfrak{a} \) be any variable assignment function of \( \mathcal{L} \) under \( \mathfrak{A} \).
    Suppose that \( \folwffsatisfaction{\mathfrak{A}}{\lambda}{\mathfrak{a}} \); that is, either \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \)
    or \( \folwffsatisfaction{\mathfrak{A}}{\phi}{\mathfrak{a}} \).

    \vspace{0.3cm}
    \noindent
    {\scshape Case:} \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \).
    Let \( e_{0} \) be an arbitrary element of \( \abs{\mathfrak{A}} \).
    Since \( x \notin \folfreevars{\psi} \) by assumption, it follows that
    \( \restrictedfunctionsmall{\mathfrak{a}}{\folfreevars{\psi}} = \restrictedfunctionsmall{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e_{0}} }{\folfreevars{\psi}} \).
    By \cref{thm:first-order-language-different-variable-assignment-functions-yield-the-same-interpretation-if-they-agree-on-the-free-variables-of-a-formula},
    it follows that \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \) iff
    \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e_{0}}} \).
    Since \( \folwffsatisfaction{\mathfrak{A}}{\psi}{\mathfrak{a}} \) is true by assumption,
    \( \folwffsatisfaction{\mathfrak{A}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e_{0}} } \) must therefore also be true.
    By \cref{dfn:first-order-language-satisfaction-of-formulas}, it follows that
    \( \folwffsatisfaction{\mathfrak{A}}{\theta}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e_{0}} } \) holds.
    Since \( e_{0} \) is arbitrary, we may conclude that \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \forall x \, \theta \mright)}{\mathfrak{a}} \).

    \vspace{0.3cm}
    \noindent
    {\scshape Case:} \( \folwffsatisfaction{\mathfrak{A}}{\phi}{\mathfrak{a}} \).
    Let \( e_{1} \) be an arbitrary element of \( \abs{\mathfrak{A}} \).
    Thus, \( \folwffsatisfaction{\mathfrak{A}}{\varphi}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e_{1}} } \) is true by assumption.
    But this would mean that \( \folwffsatisfaction{\mathfrak{A}}{\theta}{ \folmodifiedvariableassignmentfunction{\mathfrak{a}}{x}{e_{1}} } \) is also true.
    With \( e_{1} \) being arbitrary, it follows that \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \forall x \, \theta \mright)}{\mathfrak{a}} \).

    \vspace{0.3cm}
    \noindent
    We have thus shown that \( \folwffsatisfaction{\mathfrak{A}}{\mleft( \forall x \, \theta \mright)}{\mathfrak{a}} \) is implied by
    \( \folwffsatisfaction{\mathfrak{A}}{\lambda}{\mathfrak{a}} \) in all possible scenarios.
    Since \( \mathfrak{A} \) and \( \mathfrak{a} \) are arbitrary, we can therefore conclude that \( \lambda \models \forall x \, \theta \).
    To prove the converse, we follow a similar procedure as above. Let \( \mathfrak{B} \) denote any \( \mathcal{L} \)-structure and \( \mathfrak{b} \)
    denote any variable assignment function of \( \mathcal{L} \) under \( \mathfrak{B} \).
    Suppose that \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \forall x \, \theta \mright)}{\mathfrak{b}} \).
    By \cref{dfn:first-order-language-satisfaction-of-formulas}, it follows that
    \begin{equation}
        \label{eq:satisfaction-of-forall-x-theta}
        \folwffsatisfaction{\mathfrak{B}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e} } \ \ \text{or}
        \ \ \folwffsatisfaction{\mathfrak{B}}{\varphi}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e} }
    \end{equation}
    for all \( e \in \abs{\mathfrak{B}} \).
    Recall that \( x \notin \folfreevars{\psi} \), which means that
    \( \restrictedfunctionsmall{\mathfrak{b}}{ \folfreevars{\psi} } =
       \restrictedfunctionsmall{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e} }{ \folfreevars{\psi} }
    \) for all \( e \in \abs{\mathfrak{B}} \).
    By \cref{thm:first-order-language-different-variable-assignment-functions-yield-the-same-interpretation-if-they-agree-on-the-free-variables-of-a-formula},
    it follows that
    \begin{equation}
        \label{eq:satisfaction-of-psi-is-independent-of-x}
        \folwffsatisfaction{\mathfrak{B}}{\psi}{\mathfrak{b}} \ \text{iff}
        \ \folwffsatisfaction{\mathfrak{B}}{\psi}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e} }
    \end{equation}
    for all \( e \in \abs{\mathfrak{B}} \).
    By the \aref{rule-of-inference:law-of-excluded-middle}{Law of Excluded Middle}, there are clearly only two possibilities:
    either \( \folwffsatisfaction{\mathfrak{B}}{\psi}{\mathfrak{b}} \) or \( \folwffnonsatisfaction{\mathfrak{B}}{\psi}{\mathfrak{b}} \).
    In the case of the former, then it is obvious that \( \folwffsatisfaction{\mathfrak{B}}{\lambda}{\mathfrak{b}} \) is also true.
    On the other hand, if the latter is true, then \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e} } \)
    must be true for all \( e \in \abs{\mathfrak{B}} \);
    otherwise, it would contradict \eqref{eq:satisfaction-of-forall-x-theta}.
    But if \( \folwffsatisfaction{\mathfrak{B}}{\varphi}{ \folmodifiedvariableassignmentfunction{\mathfrak{b}}{x}{e} } \)
    is true for all \( e \in \abs{\mathfrak{B}} \), then it must be that \( \folwffsatisfaction{\mathfrak{B}}{\phi}{\mathfrak{b}} \), which in turn implies
    \( \folwffsatisfaction{\mathfrak{B}}{\lambda}{\mathfrak{b}} \).

    \vspace{0.3cm}
    \noindent
    Thus, \( \folwffsatisfaction{\mathfrak{B}}{\lambda}{\mathfrak{b}} \) follows from \( \folwffsatisfaction{\mathfrak{B}}{\mleft( \forall x \, \theta \mright)}{\mathfrak{b}} \)
    in all possible scenarios. Since \( \mathfrak{B} \) and \( \mathfrak{b} \) are arbitrary, it follows that \( \forall x \, \theta \models \lambda \).
\end{proof}

\begin{thm}
    \phantomsection \label{logical-equivalence:grouping-conjuncts-into-universally-quantified-formula}
    Let \( \psi \) and \( \varphi \) be \( \mathcal{L} \)-formulas. Furthermore, let \( x \) denote any \( \mathcal{L} \)-variable such that
    \( x \notin \folfreevars{\psi} \). Then \( \psi \,\,{\land}\,\, \forall x \, \varphi \equiv \forall x \, \mleft[ \psi \,\,{\land}\,\, \varphi \mright] \).
\end{thm}

\chapter{Proof-Theoretic Concepts}
\begin{dfn}[Proof System]
    \phantomsection \label{dfn:proof-system}
    A \keyword{proof system} \( \proofsystem{P} \) for a first-order language \( \mathcal{L} \) consists of the following:
    \begin{enumerate}[topsep=2pt,noitemsep,label=\textup{\texttt{(\alph*)}}]
        \item A (possibly infinite) set \( \Lambda^{\proofsystem{P}} \) of logical axioms--- \( \mathcal{L} \)-formulas that are universally true regardless of
            interpretation.
        \item A finite set \( \Phi^{\proofsystem{P}} \) of rules that determine how new \( \mathcal{L} \)-formulas can be derived from an existing
            collection of \( \mathcal{L} \)-formulas. These rules are typically called the \keyword{rules of inference}.
    \end{enumerate}
\end{dfn}

\begin{rem}
    The set \( \Lambda^{\proofsystem{P}} \) is typically defined using a finite collection of \emph{axiom schemas}, which are essentially
    templates representing an entire family of infinitely many axioms.
    For instance, the axiom schema \phantomsection \label{anchor:axiom-schema}
    \( A {\implies} \mleft( B {\implies} A \mright) \) is part of many proof systems.
    For such proof systems, the formulas \( \varphi_{1} {\implies} \mleft( \psi_{1} {\implies} \varphi_{1} \mright) \) and
    \( \varphi_2 {\implies} \mleft( \psi_1 {\implies} \varphi_{2} \mright) \) are both part of \( \Lambda^{\proofsystem{P}} \) provided
    that \( \varphi_1 \), \( \varphi_2 \), \( \psi_1 \), and \( \psi_2 \) are all \( \mathcal{L} \)-formulas.
\end{rem}

\begin{dfn}[Proof]
    \phantomsection \label{dfn:formal-proof}
    Let \( \proofsystem{P} \) be a proof system for a first-order language \( \mathcal{L} \) and \( \varphi \)
    be an \( \mathcal{L} \)-formula. Furthermore, let \( \Gamma \) be a set of \( \mathcal{L} \)-formulas.
    A \keyword{formal proof} of \( \varphi \) from \( \Gamma \) in \( \proofsystem{P} \) is a finite sequence
    \( \sequence{\Sigma_i}{i \in n } \) of length \( n \in \naturalset \) such that the following conditions hold:
    \begin{enumerate}[topsep=2pt,noitemsep,label=\textup{\texttt{(\alph*)}}]
        \item For all \( i \in n \), at least one of the following is true:
            \begin{itemize}[topsep=2pt,noitemsep]
                \item \( \Sigma_i \in \Gamma \)
                \item \( \Sigma_i \in \Lambda^{\proofsystem{P}} \)
                \item \( \Sigma_i \) is derived using one of the rules in \( \Phi^{\proofsystem{P}} \)
            \end{itemize}
        \item \( \Sigma_{n-1} = \varphi \)
    \end{enumerate}
\end{dfn}

\begin{dfn}
    \phantomsection \label{dfn:provability}
    An \( \mathcal{L} \)-formula \( \varphi \) is \keyword{provable} from \( \Gamma \) in \( \proofsystem{P} \) if and only if
    there exists a formal proof of \( \varphi \) from \( \Gamma \) in \( \proofsystem{P} \).
\end{dfn}

\begin{notation}
    \phantomsection \label{notation:provable}
    We write \( \Gamma \derives_{\proofsystem{P}} \varphi \) to mean
    that \( \varphi \) is provable from \( \Gamma \) in \( \proofsystem{P} \).
    The subscript can be omitted if the context makes it clear which proof system is being used.
    We write \( \Gamma \models \varphi \) to mean that \( \varphi \) is a \emph{semantic consequence} or \keyword{logical consequence}
    of \( \Gamma \); that is, for every interpretation where all the \( \mathcal{L} \)-formulas in \( \Gamma \)
    is true, then the \( \mathcal{L} \)-formula \( \varphi \) is also true.
\end{notation}

\begin{dfn}[Soundness]
    \phantomsection \label{dfn:soundness-of-proof-system}
    A proof system \( \proofsystem{P} \) of a first-order language \( \mathcal{L} \) is \keyword{sound}
    if and only if for any set of \( \mathcal{L} \)-formulas \( \Gamma \) and any \( \mathcal{L} \)-formula
    \( \varphi \), \( \Gamma \derives_{\proofsystem{P}} \varphi {\implies} \Gamma \models \varphi \).
\end{dfn}

\begin{dfn}[Completeness]
    \phantomsection \label{dfn:completeness-of-proof-system}
    A proof system \( \proofsystem{P} \) of a first-order language \( \mathcal{L} \) is \keyword{complete}
    if and only if for any set of \( \mathcal{L} \)-formulas \( \Gamma \) and any \( \mathcal{L} \)-formula
    \( \varphi \), \( \Gamma \models \varphi {\implies} \Gamma \derives_{\proofsystem{P}} \varphi \). In other
    words, it is possible to construct a proof in \( \proofsystem{P} \) for every possible logical consequence.
\end{dfn}

\noindent
Most proof systems in practice fall into either one of the following categories:
\begin{enumerate*}[label=(\roman*)]
    \item Hilbert systems,
    \item natural deduction.
\end{enumerate*}
Hilbert systems gravitate towards the use of logical axioms to express the laws of deductive reasoning. The number of rules
of inference in such systems is typically kept at a bare minimum.
In fact, most Hilbert systems include only a single rule of inference---Modus Ponens.
Natural deduction, on the other hand, does the exact opposite by incorporating as many rules of inference as possible in order
to mimic the ``natural'' way of human reasoning.

\vspace{0.3cm}
\noindent
Both Hilbert systems and natural deduction have their merits.
In a nutshell, natural deduction is more human-friendly whereas Hilbert systems are more computer-friendly.
Rather than basing the reasoning process on a set of seemingly abstruse and esoteric axioms,
natural deduction revolves around a set of rules that model human deductive thought.
As such, proofs written within the framework of natural deduction are typically easier to read and understand by humans;
that, per se, is why mathematicians often prefer natural deduction for reading and writing proofs.
The only downside of natural deduction is that it renders the implementation of proof-checking software far more complicated than need be,
which is why most proof-checkers are implemented using Hilbert systems.

\vspace{0.3cm}
\noindent
This book adopts a natural deduction proof system for all its proofs.
Refer to \aref{appendices:natural-deduction}{Appendix }\ref{appendices:natural-deduction} for a complete specification of the proof system used in this book.

\begin{notation}
    Let \( \mathcal{L} \) be any first-order language and \( \proofsystem{P} \) be our proof system for \( \mathcal{L} \).
    The following notational conventions will be used in the specification of \( \proofsystem{P} \):
    \begin{enumerate}[topsep=2pt,noitemsep,label=\textup{\texttt{(\alph*)}}]
        \item Lowercase letters (\( x, y, z\)) denote variables.
        \item Uppercase letters (\( A,B,C,\dots \)) and \( \varphi \) denote \( \mathcal{L} \)-formulas.
        \item Uppercase Greek letters (\( \Delta, \Gamma \)) denote \emph{sets} of \( \mathcal{L} \)-formulas.
        \item \aref{dfn:functional-notation}{Functional notation} is used whenever there is a need to emphasize
            that an \( \mathcal{L} \)-formula has free variables (i.e., \( \varphi(x) \)).
        \item Ellipses are used to indicate that an \( \mathcal{L} \)-formula may contain any number of free variables other
            than the one indicated (i.e., \( \varphi(x,\dots) \)).
        \item We write \( \varphi(x \mapsto t) \) to denote the \( \mathcal{L} \)-formula obtained by replacing every free
            occurrence of \( x \) in \( \varphi \) with \( t \).
    \end{enumerate}
\end{notation}

\subsection*{Logical Axioms}
\( \proofsystem{P} \) has only one logical axiom, and that is the Law of Identity: \( \forall x \, \mleft[ x = x \mright] \).
The rest of \( \proofsystem{P} \) consists of rules for deriving new \( \mathcal{L} \)-formulas from a (possibly empty) set of
existing \( \mathcal{L} \)-formulas.

\subsection*{Rules of Inference}
\begin{enumerate}[topsep=2pt,itemsep=10pt,label=\textup{\texttt{(\arabic*)}}]
    \item {Assumption}: \phantomsection \label{rule-of-inference:assumption}

        If \( A \in \Gamma \), then \( \Gamma \derives A \).

    \item Monotonicity: \phantomsection \label{rule-of-inference:monotonicity}

        If \( \Gamma^{\prime} \derives A \) and \( \Gamma^{\prime} \subset \Gamma \), then \( \Gamma \derives A \).

    \item Substitution: \phantomsection \label{rule-of-inference:substitution}

        If \( \Gamma \derives \mleft( x = y \mright) \), then \( \Gamma \derives \mleft( \varphi(x, \dots) {\iff} \varphi(y, \dots) \mright) \)
        for any formula \( \varphi \).

    \item Double Negation: \phantomsection \label{rule-of-inference:double-negation}

        If \( \Gamma \derives A \), then \( \Gamma \derives \lnot \, \mleft( \lnot \, A \mright) \).

    \item Conjunction Introduction: \phantomsection \label{rule-of-inference:conjunction-introduction}

        If \( \Gamma \derives A \) and \( \Gamma \derives B \), then \( \Gamma \derives \mleft( A \,\,{\land}\,\, B \mright) \).

    \item Conjunction Elimination: \phantomsection \label{rule-of-inference:conjunction-elimination}

        If \( \Gamma \derives \mleft( A \,\,{\land}\,\, B \mright) \), then \( \Gamma \derives A \).

    \item Disjunction Introduction: \phantomsection \label{rule-of-inference:disjunction-introduction}

        If \( \Gamma \derives A \), then \( \Gamma \derives \mleft( A \,\,{\lor}\,\, B \mright) \).

    \item Disjunction Elimination: \phantomsection \label{rule-of-inference:disjunction-elimination}

        If \( \Gamma \derives \mleft( A \,\,{\lor}\,\, B \mright) \), \( \Gamma \derives \mleft( A {\implies} C \mright) \),
        and \( \Gamma \derives \mleft( B {\implies} C \mright) \), then \( \Gamma \derives C \).

    \item Commutation: \phantomsection \label{rule-of-inference:commutation}

        If \( \Gamma \derives \mleft( A \,\,{\land}\,\, B \mright) \), then \( \Gamma \derives \mleft( B \,\,{\land}\,\, A \mright) \).

        \noindent
        If \( \Gamma \derives \mleft( A \,\,{\lor}\,\, B \mright) \), then \( \Gamma \derives \mleft( B \,\,{\lor}\,\, A \mright) \).

    \item Association: \phantomsection \label{rule-of-inference:association}

        If \( \Gamma \derives \mleft( A \,\,{\land}\,\, \mleft( B \,\,{\land}\,\, C \mright) \mright) \), then
        \( \Gamma \derives \mleft( \mleft( A \,\,{\land}\,\, B \mright) \,\,{\land}\,\, C \mright) \).

        \noindent
        If \( \Gamma \derives \mleft( A \,\,{\lor}\,\, \mleft( B \,\,{\lor}\,\, C \mright) \mright) \), then
        \( \Gamma \derives \mleft( \mleft( A \,\,{\lor}\,\, B \mright) \,\,{\lor}\,\, C \mright) \).

        \noindent
        If \( \Gamma \derives \mleft( \mleft( A \,\,{\land}\,\, B \mright) \,\,{\land}\,\, C \mright) \), then
        \( \Gamma \derives \mleft( A \,\,{\land}\,\, \mleft( B \,\,{\land}\,\, C \mright) \mright) \).

        \noindent
        If \( \Gamma \derives \mleft( \mleft( A \,\,{\lor}\,\, B \mright) \,\,{\lor}\,\, C \mright) \), then
        \( \Gamma \derives \mleft( A \,\,{\lor}\,\, \mleft( B \,\,{\lor}\,\, C \mright) \mright) \).

    \item Distribution: \phantomsection \label{rule-of-inference:distribution}

        If \( \Gamma \derives \mleft( A \,\,{\land}\,\, \mleft( B \,\,{\lor}\,\, C \mright) \mright) \), then
        \( \Gamma \derives \mleft( \mleft( A \,\,{\land}\,\, B \mright) \,\,{\lor}\,\, \mleft( A \,\,{\land}\,\, C \mright) \mright) \).

        \noindent
        If \( \Gamma \derives \mleft( A \,\,{\lor}\,\, \mleft( B \,\,{\land}\,\, C \mright) \mright) \), then
        \( \Gamma \derives \mleft( \mleft( A \,\,{\lor}\,\, B \mright) \,\,{\land}\,\, \mleft( A \,\,{\lor}\,\, C \mright) \mright) \).

        \noindent
        If \( \Gamma \derives \mleft( \mleft( A \,\,{\land}\,\, B \mright) \,\,{\lor}\,\, \mleft( A \,\,{\land}\,\, C \mright) \mright) \),
        then \( \Gamma \derives \mleft( A \,\,{\land}\,\, \mleft( B \,\,{\lor}\,\, C \mright) \mright) \).

        \noindent
        If \( \Gamma \derives \mleft( \mleft( A \,\,{\lor}\,\, B \mright) \,\,{\land}\,\, \mleft( A \,\,{\lor}\,\, C \mright) \mright) \),
        then \( \Gamma \derives \mleft( A \,\,{\lor}\,\, \mleft( B \,\,{\land}\,\, C \mright) \mright) \).

    \item Disjunction Definition: \phantomsection \label{rule-of-inference:disjunction-definition}

        If \( \Gamma \derives \mleft( A \,\,{\lor}\,\, B \mright) \), then
        \( \Gamma \derives \mleft( \lnot \, \mleft( \lnot \, A \,\,{\land}\,\, \lnot \, B \mright) \mright) \).

        \noindent
        If \( \Gamma \derives \mleft( \lnot \, \mleft( \lnot \, A \,\,{\land}\,\, \lnot \, B \mright) \mright) \),
        then \( \Gamma \derives \mleft( A \,\,{\lor}\,\, B \mright) \).

    \item Conditional Definition: \phantomsection \label{rule-of-inference:conditional-definition}

        If \( \Gamma \derives \mleft( A {\implies} B \mright) \), then \( \Gamma \derives \mleft( \lnot \, A \,\,{\lor}\,\, B \mright) \).

        \noindent
        If \( \Gamma \derives \mleft( \lnot \, A \,\,{\lor}\,\, B \mright) \), then \( \Gamma \derives \mleft( A {\implies} B \mright) \).

    \item Contraposition: \phantomsection \label{rule-of-inference:contraposition}

        If \( \Gamma \derives \mleft( A {\implies} B \mright) \), then \( \Gamma \derives \mleft( \lnot \, B {\implies} \lnot \, A \mright) \).

        \noindent
        If \( \Gamma \derives \mleft( \lnot \, B {\implies} \lnot \, A \mright) \), then \( \Gamma \derives \mleft( A {\implies} B \mright) \).

    \item Biconditional Introduction: \phantomsection \label{rule-of-inference:biconditional-introduction}

        If \( \Gamma \derives \mleft( A {\implies} B \mright) \) and \( \Gamma \derives \mleft( B {\implies} A \mright) \),
        then \( \Gamma \derives \mleft( A {\iff} B \mright) \).

    \item Biconditional Elimination: \phantomsection \label{rule-of-inference:biconditional-elimination}

        If \( \Gamma \derives \mleft( A {\iff} B \mright) \), then \( \Gamma \derives \mleft( A {\implies} B \mright) \).

        \noindent
        If \( \Gamma \derives \mleft( A {\iff} B \mright) \), then \( \Gamma \derives \mleft( B {\implies} A \mright) \).

    \item Law of Excluded Middle: \phantomsection \label{rule-of-inference:law-of-excluded-middle}

        \( \Gamma \derives \mleft( A \,\,{\lor}\,\, \lnot \, A \mright) \).

    \item De Morgan's Laws: \phantomsection \label{rule-of-inference:de-morgans-laws}

        If \( \Gamma \derives \lnot \, \mleft( A \,\,{\land}\,\, B \mright) \), then
        \( \Gamma \derives \mleft( \lnot \, A \,\,{\lor}\,\, \lnot \, B \mright) \).

        \noindent
        If \( \Gamma \derives \lnot \, \mleft( A \,\,{\lor}\,\, B \mright) \), then
        \( \Gamma \derives \mleft( \lnot \, A \,\,{\land}\,\, \lnot \, B \mright) \).

        \noindent
        If \( \Gamma \derives \mleft( \lnot \, A \,\,{\lor}\,\, \lnot \, B \mright) \), then
        \( \Gamma \derives \lnot \, \mleft( A \,\,{\land}\,\, B \mright) \).

        \noindent
        If \( \Gamma \derives \mleft( \lnot \, A \,\,{\land}\,\, \lnot \, B \mright) \), then
        \( \Gamma \derives \lnot \, \mleft( A \,\,{\lor}\,\, B \mright) \).

    \item Absorption: \phantomsection \label{rule-of-inference:absorption}

        If \( \Gamma \derives \mleft( A {\implies} B \mright) \), then \( \Gamma \derives \mleft( A {\implies} \mleft( A \,\,{\land}\,\, B \mright) \mright) \).

    \item Exportation: \phantomsection \label{rule-of-inference:exportation}

        If \( \Gamma \derives \mleft( \mleft( A \,\,{\land}\,\, B \mright) {\implies} C \mright) \),
        then \( \Gamma \derives \mleft( A {\implies} \mleft( B {\implies} C \mright) \mright) \).

    \item Importation: \phantomsection \label{rule-of-inference:importation}

        If \( \Gamma \derives \mleft( A {\implies} \mleft( B {\implies} C \mright) \mright) \), then
        \( \Gamma \derives \mleft( \mleft( A \,\,{\land}\,\, B \mright) {\implies} C \mright) \).

    \item Modus Ponens: \phantomsection \label{rule-of-inference:modus-ponens}

        If \( \Gamma \derives \mleft( A {\implies} B \mright) \) and \( \Gamma \derives A \), then \( \Gamma \derives B \).

    \item Modus Tollens: \phantomsection \label{rule-of-inference:modus-tollens}

        If \( \Gamma \derives \mleft( A {\implies} B \mright) \) and \( \Gamma \derives \lnot \, B \), then \( \Gamma \derives \lnot \, A \).

    \item Disjunctive Syllogism: \phantomsection \label{rule-of-inference:disjunctive-syllogism}

        If \( \Gamma \derives \mleft( A \,\,{\lor}\,\, B \mright) \) and \( \Gamma \derives \lnot \, B \), then \( \Gamma \derives A \).

    \item Hypothetical Syllogism: \phantomsection \label{rule-of-inference:hypothetical-syllogism}

        If \( \Gamma \derives \mleft( A {\implies} B \mright) \) and \( \Gamma \derives \mleft( B {\implies} C \mright) \), then
        \( \Gamma \derives \mleft( A {\implies} C \mright) \).

    \item Constructive Dilemma: \phantomsection \label{rule-of-inference:constructive-dilemma}

        If \( \Gamma \derives \mleft( A {\implies} C \mright) \), \( \Gamma \derives \mleft( B {\implies} D \mright) \), and
        \( \Gamma \derives \mleft( A \,\,{\lor}\,\, B \mright) \), then \( \Gamma \derives \mleft( C \,\,{\lor}\,\, D \mright) \).

    \item Deduction Theorem: \phantomsection \label{rule-of-inference:deduction-theorem}

        If \( \Gamma \union \mleft\{ A \mright\} \derives B \), then \( \Gamma \derives \mleft( A {\implies} B \mright) \).

    \item Reductio ad Absurdum: \phantomsection \label{rule-of-inference:reductio-ad-absurdum}

        If \( \Gamma \union \mleft\{ A \mright\} \derives B \) and \( \Gamma \union \mleft\{ A \mright\} \derives \lnot \, B \), then
        \( \Gamma \derives \lnot \, A \).

    \item Universal Generalization: \phantomsection \label{rule-of-inference:universal-generalization}

        If \( \Gamma \derives \varphi(t, \dots) \) and \( t \) does not appear in any of the formulas in \( \Gamma \),
        then \( \Gamma \derives \forall x \, \varphi(t \mapsto x, \dots) \) provided that \( x \) does not appear in
        \( \varphi(t, \dots) \).

    \item Universal Instantiation: \phantomsection \label{rule-of-inference:universal-instantiation}

        If \( \Gamma \derives \forall x \, \varphi(x,\dots) \) and \( t \) is not a bound variable in \( \varphi(x, \dots) \),
        then \( \Gamma \derives \varphi(x \mapsto t,\dots) \).

    \item Existential Generalization: \phantomsection \label{rule-of-inference:existential-generalization}

        If \( \Gamma \derives \varphi(t,\dots) \) and \( x \) does not appear in \( \varphi(t, \dots) \),
        then \( \Gamma \derives \exists x \, \varphi(t \mapsto x, \dots )\).

    \item Existential Instantiation: \phantomsection \label{rule-of-inference:existential-instantiation}

        If \( \Gamma \derives \exists x \, \varphi(x, \dots) \) and \( t \) does not appear in \( \varphi(x, \dots) \),
        then \( \Gamma \derives \varphi(t \mapsto x, \dots) \).

    \item Existential Definition: \phantomsection \label{rule-of-inference:existential-definition}

        If \( \Gamma \derives \exists x \, \varphi(x, \dots) \), then \( \Gamma \derives \lnot \, \forall x \, \lnot \, \varphi(x, \dots) \).

        \noindent
        If \( \Gamma \derives \lnot \, \forall x \, \lnot \, \varphi(x, \dots) \), then \( \Gamma \derives \exists x \, \varphi(x, \dots) \).
\end{enumerate}

\vspace{0.3cm}
\noindent
The validity of rules \crefrange{rule-of-inference:assumption}{rule-of-inference:reductio-ad-absurdum} is evident from basic propositional logic
(i.e., to verify them one only has to look at the truth tables of the relevant logical operators).
On the other hand, the restrictions imposed on the rules involving quantifiers warrant further discussion.

\vspace{0.3cm}
\noindent
First, we have the \aref{rule-of-inference:universal-generalization}{Universal Generalization} rule, upon which the
following restrictions are imposed:
\begin{enumerate}[label=\textup{\texttt{(\alph*)}}]
    \item \( t \) is not a variable in any of the formulas in \( \Gamma \). \phantomsection \label{enum:universal-generalization-restriction1}
    \item \( x \) is not a variable in \( \varphi \). \phantomsection \label{enum:universal-generalization-restriction2}
\end{enumerate}
Both of these restrictions are necessary to ensure the \aref{dfn:soundness-of-proof-system}{soundness} of \( \proofsystem{P} \).
To illustrate why, suppose that \( \Gamma = \mleft\{ \varphi, \psi, \phi \mright\} \) is a set of \( \mathcal{L} \)-formulas with the formulas
\( \varphi \), \( \psi \), and \( \phi \) defined as follows:
\begin{align*}
    \varphi(t) &\coloneq \text{\( t \) is a cat}\\
    \psi(t,x) &\coloneq \text{\( t \) is an offspring of \( x \)}\\
    \phi &\coloneq \text{\( \forall x \, \exists t \, \psi(x,t) \)}
\end{align*}
Since \( \varphi(t) \in \Gamma \), we can validly infer by \aref{rule-of-inference:assumption}{Assumption} that \( \Gamma \derives \varphi(t) \).
However, without restriction \ref{enum:universal-generalization-restriction1}, we are then permitted to fallaciously apply
\aref{rule-of-inference:universal-generalization}{Universal Generalization} to conclude that
\( \Gamma \derives \forall x \, \varphi(x) \), which is tantamount to the following argument:
\begin{equation*}
    \begin{aligned}
        \text{\bfseries Premise:}&\\
        \text{\bfseries Conclusion:}&
    \end{aligned}
    \begin{aligned}
        &\quad\text{\( t \) is a cat.}\\
        &\quad\text{Therefore, every animal is a cat.}
    \end{aligned}
\end{equation*}
\noindent
Absurd! Just because some animal named \( t \) is a cat does not necessarily mean that every animal in existence is a cat.
In order to validly infer \( \forall z \, \varphi(z) \) from \( \Gamma \),
one must be able to derive \( \varphi \) from \( \Gamma \) using \emph{any}
variable symbol from the language of \( \mathcal{L} \), not just \( t \).
It is easy to see how restriction \ref{enum:universal-generalization-restriction1} enforces this rule.
Since the variable \( t \) is a free variable in at least one formula in \( \Gamma \),
the \aref{rule-of-inference:universal-generalization}{Universal Generalization} rule is not applicable,
so one cannot infer \( \forall z \, \varphi(z) \) from \( \varphi(t) \). Instead, one has to
derive \( \varphi \) from \( \Gamma \) not by mere \aref{rule-of-inference:assumption}{Assumption}, but
by showing that \( \varphi \) still holds even after replacing \( t \) with an arbitrary variable in the language of \( \mathcal{L} \).

\vspace{0.3cm}
\noindent
Restriction \ref{enum:universal-generalization-restriction2} is equally important for the soundness of \( \proofsystem{P} \).
If restriction \ref{enum:universal-generalization-restriction1}'s role is to prevent erroneous inferences, then restriction
\ref{enum:universal-generalization-restriction2}'s role is to preserve the \emph{meaning} of the original formula during the
generalization. To understand this, suppose that \( \Gamma \derives \psi(a,x) \). Since \( a \) is not mentioned in any of the
formulas in \( \Gamma \), restriction \ref{enum:universal-generalization-restriction1} is not violated, so one concludes (albeit fallaciously)
by \aref{rule-of-inference:universal-generalization}{Universal Generalization} that \( \forall x \, \psi(a \mapsto x, x) \).
Informally, this problematic inference translates to:
\begin{equation*}
    \begin{aligned}
        \text{\bfseries Premise:}&\\
        \text{\bfseries Conclusion:}&
    \end{aligned}
    \begin{aligned}
        &\quad \text{\( a \) is an offspring of \( x \).}\\
        &\quad\text{Therefore, every \( x \) is an offspring of \( x \).}
    \end{aligned}
\end{equation*}
\noindent
Again, absurd!
The fact that an animal \( a \) is an offspring of some other animal \( x \) does not logically imply that every animal is an offspring of itself.
There is simply no logical connection whatsoever between the conclusion and the premise. So what exactly went wrong?

\vspace{0.3cm}
\noindent
The problem here is that the variable \( x \) is already a free variable of \( \psi \) before the universal quantifier is applied,
and whenever we replace a free variable in a formula with another free variable in the same formula, we alter the meaning of the formula entirely.
\begin{equation*}
    \begin{aligned}
        \psi(a, x):&\\
        \psi(a \mapsto x, x):&
    \end{aligned}
    \begin{aligned}
        &\quad\text{\( a \) is an offspring of \( x \).}\\
        &\quad\text{\( x \) is an offspring of \( x \).}
    \end{aligned}
\end{equation*}
\noindent
Needless to say, \( \psi(a,x) \) and \( \psi(a \mapsto x, x) \) have completely different meanings.
\( \psi(a, x) \) asserts that the first animal \( a \) is an offspring of
the second animal \( x \), whereas \( \psi(a \mapsto x, x) \) asserts that an animal is an offspring of itself;
and when the universal quantifier is applied to \( \psi(a \mapsto x, x) \), we end up with a formula that says
\emph{every animal is an offspring of itself} instead of the intended
\emph{every animal is an offspring of \( x \)}.

\vspace{0.3cm}
\noindent
Of course, restriction \ref{enum:universal-generalization-restriction2} resolves this by prohibiting the reuse of variable names in the same formula;
that is, we are no longer allowed to infer \( \forall x \, \psi(a \mapsto x, x) \) from \( \psi(a, x) \)
because the symbol \( x \) is already a free variable in \( \psi(a, x) \).
Instead, we must use something other than \( x \) for the bound variable---i.e., \( \forall y \, \psi(a \mapsto y, x) \).

\vspace{0.3cm}
\noindent
To understand the restriction on \aref{rule-of-inference:universal-instantiation}{Universal Instantiation}, consider
the invalid inference from \( \forall x \, \exists t \, \psi(x,t) \) to \( \exists t \, \psi(x \mapsto t, t) \),
which results in the following nonsensical argument:
\begin{equation*}
    \begin{aligned}
        \text{\bfseries Premise}&:\\
        \text{\bfseries Conclusion}&:
    \end{aligned}
    \begin{aligned}
        &\quad\text{Every animal has a parent.}\\
        &\quad\text{There exists an animal that is its own parent.}
    \end{aligned}
\end{equation*}
Since \( t \) is a bound variable in \( \exists t \, \psi(x,t) \)
and the free variable \( x \) appears within the scope of the existential quantifier to which \( t \) is bound,
replacing all occurrences of \( x \) with \( t \) will alter the meaning of the formula because the variable that used
to be \( x \) will no longer be a free variable after the substitution.
\begin{equation*}
    \begin{aligned}
        \exists t \, \psi(x,t)&:\\
        \exists t \, \psi(x \mapsto t, t)&:
    \end{aligned}
    \begin{aligned}
        &\quad \exists t \, \big[ \text{\( x \) is an offspring of \( t \)} \big]\\
        &\quad \exists t \, \big[ \text{\( t \) is an offspring of \( t \)} \big]
    \end{aligned}
\end{equation*}
The variable that used to be \( x \) is supposed to still be a free variable after the substitution; but because
the new variable is called \( t \) and \( t \) is already bound to an existential quantifier, the resulting variable
becomes a bound variable too. A variable with a different name has to be used to retain the intended meaning of the formula:
\begin{equation*}
    \begin{aligned}
        \text{\bfseries Premise}&:\\
        \text{\bfseries Conclusion}&:
    \end{aligned}
    \begin{aligned}
        &\quad \forall x \, \exists t \, \psi(x,t)\\
        &\quad \exists t \, \psi(y, t)
    \end{aligned}
\end{equation*}
Note that the use of the variable \( t \) for \aref{rule-of-inference:universal-instantiation}{Universal Instantiation} is still allowed if \( t \) is a free variable
in the original formula. For example, if \( \Gamma \derives \forall x \, \psi(x, t) \), then \( \Gamma \derives \psi(t, t) \) is a valid inference because \( t \)
is a free variable in \( \psi(x,t) \) and not a bound variable. This is in contrast with restriction \ref{enum:universal-generalization-restriction2} for
\aref{rule-of-inference:universal-generalization}{Universal Generalization} (take note!)
which prohibits the reuse of any variable, free or bound, from the original formula.

\chapter{Logical Equivalences}
\begin{thm}
    \label{tautology:grouping-conjuncts-into-universally-quantified-formula}
    Given any well-formed formulas \( \Gamma \) and \( \varphi \)
    with free variables \( \alpha_{1}, \dots, \alpha_{n}\) and
    \(\beta_{1}, \dots, \beta_{m} \) respectively,
    \begin{equation*}
        \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
        \forall x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
        \equiv
        \forall x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
        \varphi(x, \beta_{1}, \dots, \beta_{m})\big)
    \end{equation*}
\end{thm}

\begin{proof}
    Let \( \Gamma \) and \( \varphi \) be well-formed formulas interpreted in the context
    of an arbitrary, non-empty domain \( D \). Let \( \alpha_{1}, \dots, \alpha_{n} \) and
    \( \beta_{1}, \dots, \beta_{m} \) represent objects in \( D \).

    \vspace{0.5cm}
    \noindent
    Suppose that \( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
    \forall x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big) \) holds.
    Now, let \( x_{0} \in D \) be arbitrary.
    Since \( \forall x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big) \) is true,
    \( \varphi(x_{0}, \beta_{1}, \dots, \beta_{m}) \) follows by Universal Instantiation.
    By Conjunction Introduction and Universal Generalization, it follows that,
    \begin{equation*}
        \forall x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
        \varphi(x, \beta_{1}, \dots, \beta_{m})\big)
    \end{equation*}
    Using the Deduction Theorem, we derive the implication:
    \begin{equation*}
        \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
        \forall x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
        \implies
        \forall x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
        \varphi(x, \beta_{1}, \dots, \beta_{m})\big)
    \end{equation*}
    Given that the interpretation of all the formulas is arbitrary, it follows that,
    \begin{equation}
        \label{eq:conjunct-into-universal-quantifier-forward-entailment}
        \Gamma(\alpha_{1}, \dots, \alpha_{n})
        \land
        \forall x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
        \models
        \forall x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n})
        \land
        \varphi(x, \beta_{1}, \dots, \beta_{m})\big)
    \end{equation}

    \vspace{0.5cm}
    \noindent
    Conversely, if \( \forall x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
    \varphi(x, \beta_{1}, \dots, \beta_{m})\big) \) holds, then \( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \)
    must be true. Let
    \( x_{0} \) be an arbitrary object in the domain. Again, \( \varphi(x_{0}, \beta_{1}, \dots, \beta_{m}) \)
    follows directly from the initial hypothesis. Since \( x_{0} \) is arbitrary, Universal Generalization permits
    us to conclude \( \forall x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big) \). By Conjunction Introduction,
    we have,
    \begin{equation*}
        \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
        \forall x \big( \varphi(x, \beta_{1}, \beta_{m}) \big)
    \end{equation*}
    Using the Deduction Theorem, we infer,
    \begin{equation*}
        \forall x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
        \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
        \implies
        \Gamma(\alpha_{1}, \dots, \alpha_{n})
        \land
        \forall x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
    \end{equation*}
    And since the interpretation is arbitrary,
    \begin{equation}
        \label{eq:conjunct-into-universal-quantifier-backward-entailment}
        \forall x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
        \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
        \models
        \Gamma(\alpha_{1}, \dots, \alpha_{n})
        \land
        \forall x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
    \end{equation}
    Together, \eqref{eq:conjunct-into-universal-quantifier-forward-entailment} and \eqref{eq:conjunct-into-universal-quantifier-backward-entailment}
    establish the logical equivalence.
\end{proof}

\begin{thm}
    \label{tautology:grouping-disjuncts-into-existentially-quantified-formula}
    For any well-formed formula \( \Gamma \) and \( \varphi \) with free variables
    \( \alpha_{1}, \dots \alpha_{n} \) and \( \beta_{1}, \dots, \beta_{m} \) respectively,
    \begin{equation*}
        \Gamma(\alpha_{1}, \dots, \alpha_{n})
        \lor
        \exists x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
        \equiv
        \exists x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \lor
        \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
    \end{equation*}
\end{thm}

\begin{proof}
    Let \( \Gamma \) and \( \varphi \) be well-formed formulas interpreted in the context of an arbitrary, non-empty domain \( D \).
    Let \( \alpha_{1}, \dots, \alpha_{n} \) and \( \beta_{1}, \dots, \beta_{m} \) represent objects in \( D \).

    \vspace{0.5cm}
    \noindent
    Suppose that \( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \lor \exists x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big) \)
    is true, which is only possible in three scenarios.

    \vspace{0.5cm}
    \noindent
    {\scshape Case 1}: \( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land \neg \, \exists x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big) \).

    \vspace{0.1cm}
    \noindent
    Since \( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \) is true, it follows that \( \forall x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n})
    \lor \varphi(x, \beta_{1}, \dots, \beta_{m}) \big) \) is true. Since the domain \( D \) is non-empty,
    \( \exists x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \lor
    \varphi(x, \beta_{1}, \dots, \beta_{m}) \big) \) is necessarily true.

    \vspace{0.5cm}
    \noindent
    {\scshape Case 2}: \( \neg \, \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
    \exists x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big) \)

    \vspace{0.1cm}
    \noindent
    Let \( x_{0} \) be an object in the domain such that \( \varphi(x_{0}, \beta_{1}, \dots, \beta_{m}) \). By Disjunction Introduction,
    clearly \( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \lor \varphi(x_{0}, \beta_{1}, \dots, \beta_{m}) \) is true. And by Existential
    Introduction, we have, \( \exists x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \lor \varphi(x, \beta_{1}, \dots, \beta_{m}) \big) \).

    \vspace{0.5cm}
    \noindent
    {\scshape Case 3}: \( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land \exists x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big) \)

    \vspace{0.1cm}
    \noindent
    Since \( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \) is true, this reduces to the first case.

    \vspace{0.5cm}
    \noindent
    Evidently, \( \exists x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \lor \varphi(x, \beta_{1}, \dots, \beta_{m}) \big) \) is implied
    in all three cases. Using the Deduction Theorem, we conclude,
    \begin{equation*}
        \Gamma(\alpha_{1}, \dots, \alpha_{n}) \lor
        \exists x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
        \implies
        \exists x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n})
        \lor
        \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
    \end{equation*}
    Since the implication was derived using an arbitrary interpretation, we can derive the following entailment:
    \begin{equation}
        \label{eq:disjunct-into-existential-quantifier-forward-entailment}
        \Gamma(\alpha_{1}, \dots, \alpha_{n}) \lor
        \exists x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
        \models
        \exists x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \lor
        \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
    \end{equation}

    \vspace{0.5cm}
    \noindent
    Proving the converse is also fairly straightforward. Naturally, if
    \begin{equation*}
        \exists x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \lor \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
    \end{equation*}
    is true, then there are only two possible scenarios to consider\,\footnote{Again, this is the Law of Excluded Middle at play.}.

    \vspace{0.5cm}
    \noindent
    {\scshape Case 1}: \( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \)

    \vspace{0.1cm}
    \noindent
    {\scshape Case 2}: \( \neg \, \Gamma(\alpha_{1}, \dots, \alpha_{n}) \)

    \vspace{0.5cm}
    \noindent
    In the first case, \( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \lor \exists x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big) \) follows
    readily from Disjunction Introduction.
    In the second case, we use Disjunctive Syllogism to conclude that,
    \begin{equation*}
        \exists x \big[ \varphi(x, \beta_{1}, \dots, \beta_{m}) \big]
    \end{equation*}
    Again, \( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \lor \exists x \mleft( \varphi(x, \beta_{1}, \dots, \beta_{m}) \mright) \) follows from Disjunction
    Introduction. Applying the Deduction Theorem, we infer,
    \begin{equation*}
        \exists x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \lor \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
        \implies
        \Gamma(\alpha_{1}, \dots, \alpha_{n}) \lor
        \exists x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
    \end{equation*}
    And since the implication is independent of implication, we have,
    \begin{equation}
        \label{eq:disjunct-into-existential-quantifier-backward-implication}
        \exists x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \lor \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
        \models
        \Gamma(\alpha_{1}, \dots, \alpha_{n}) \lor
        \exists x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
    \end{equation}
    Together, \eqref{eq:disjunct-into-existential-quantifier-forward-entailment} and \eqref{eq:disjunct-into-existential-quantifier-backward-implication}
    establish the logical equivalence.
\end{proof}

\begin{thm}
    \label{tautology:grouping-conjuncts-into-existentially-quantified-formula}
    For any well-formed formula \( \Gamma \) and \( \varphi \) with free variables
    \( \alpha_{1}, \dots, \alpha_{n} \) and \( \beta_{1}, \dots, \beta_{m} \) respectively,
    \begin{equation*}
        \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
        \exists x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
        \equiv
        \exists x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
        \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
    \end{equation*}
\end{thm}

\begin{proof}
    Let \( \Gamma \) and \( \varphi \) be well-formed fomulas interpreted in the context of an
    arbitrary, non-empty domain, \( D \). Let \( \alpha_{1}, \dots, \alpha_{n} \) and
    \( \beta_{1}, \dots, \beta_{m} \) represent objects in \( D \).

    \vspace{0.5cm}
    \noindent
    Suppose that \( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land \exists x \mleft( \varphi(x, \beta_{1}, \dots, \beta_{m}) \mright) \)
    is true. Clearly, \( \varphi(x_{0}, \beta_{1}, \dots, \beta_{m}) \) is true for some \( x_{0} \in D \). Thus, by Conjunction Introduction,
    \begin{equation*}
        \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land \varphi(x_{0}, \beta_{1}, \dots, \beta_{m})
    \end{equation*}
    And by Existential Instantiation,
    \begin{equation*}
        \exists x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
    \end{equation*}
    Thus,
    \begin{equation*}
        \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
        \exists x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
        \models
        \exists x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
        \varphi(x, \beta_{1}, \dots, \beta_{m})\big)
    \end{equation*}

    \vspace{0.5cm}
    \noindent
    Proving the converse is trivial. If \( \exists x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
    \varphi(x, \beta_{1}, \dots, \beta_{m})\big) \) is true, then clearly,
    \begin{equation*}
        \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land \varphi(x_{0}, \beta_{1}, \dots, \beta_{m})
    \end{equation*}
    for some \( x_{0} \in D \). By Conjunction Elimination, obviously \( \varphi(x_{0}, \beta_{1}, \dots, \beta_{m}) \)
    is true for some \( x_{0} \in D \). By Existential Generalization, we thus conclude,
    \begin{equation*}
        \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
        \exists x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
    \end{equation*}
    And therefore,
    \begin{equation*}
        \exists x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
        \varphi(x, \beta_{1}, \dots, \beta_{m})\big)
        \models
        \Gamma(\alpha_{1}, \dots, \alpha_{n}) \land
        \exists x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big) \qedhere
    \end{equation*}
\end{proof}

\begin{thm}
    \label{tautology:combining-unquantified-antecedent-and-universally-quantified-conseqeunt-into-a-single-quantified-formula}
    For any well-formed formula \( \Gamma \) and \( \varphi \) with free variables
    \( \alpha_{1}, \dots, \alpha_{n} \) and \( \beta_{1}, \dots, \beta_{m} \) respectively,
    \begin{equation*}
        \Gamma(\alpha_{1}, \dots, \alpha_{n})
        \implies
        \forall x \big[ \varphi(x, \beta_{1}, \dots, \beta_{m}) \big]
        \equiv
        \forall x \big[ \Gamma(\alpha_{1}, \dots, \alpha_{n}) \implies \varphi(x, \beta_{1}, \dots, \beta_{m}) \big]
    \end{equation*}
\end{thm}

\begin{proof}
    Let \( \Gamma \) and \( \varphi \) be well-formed formulas interpreted in the context of an arbitrary, non-empty
    domain \( D \). Let \( \alpha_{1}, \dots, \alpha_{n} \) and \( \beta_{1}, \dots, \beta_{m} \) be objects in \( D \).
    Using the truth table for logical implication and \cref{logical-equivalence:grouping-disjuncts-into-universally-quantified-formula},
    we transform one of the statement into the other like so:
    \begin{align*}
        \Gamma(\alpha_{1}, \dots, \alpha_{n})
        {\implies}
        \forall x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
        &\iff
        \neg \, \Gamma(\alpha_{1}, \dots, \alpha_{n})
        \lor
        \forall x \big( \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)\\
        &\iff
        \forall x \big( \neg \, \Gamma(\alpha_{1}, \dots, \alpha_{n})
        \lor
        \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)\\
        &\iff
        \forall x \big( \Gamma(\alpha_{1}, \dots, \alpha_{n})
        {\implies}
        \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
    \end{align*}
    The biconditional holds regardless of the interpretation of choice. Thus, the two statements are logically equivalent.
\end{proof}

\begin{thm}
    \label{tautology:using-a-universally-quantified-formula-to-express-the-fact-that-an-object-satisfies-a-formula}
    For any well-formed formula \( \varphi \) and free variables
    \( \alpha, \beta_{1}, \dots, \beta_{m} \),
    \begin{equation*}
        \forall x \big( x = \alpha \implies \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
        \equiv
        \varphi(\alpha, \beta_{1}, \dots, \beta_{m})
    \end{equation*}
\end{thm}

\begin{proof}
    Let \( \varphi \) be a well-formed formula interpreted in the context of a non-empty
    domain, \( D \). Let \( \alpha, \beta_{1}, \dots, \beta_{m} \) represent objects in \( D \).
    Suppose that the well-formed formula
    \( \forall x \big( x = \alpha \implies \varphi(x, \beta_{1}, \dots, \beta_{m}) \big) \) is true.
    Then by Universal Instantiation, we have,
    \begin{equation*}
        \alpha = \alpha \implies \varphi(\alpha, \beta_{1}, \dots, \beta_{m})
    \end{equation*}
    Since \( \alpha = \alpha \) is trivially true, we conclude by Modus Ponens that
    \( \varphi(\alpha, \beta_{1}, \dots, \beta_{m}) \) is also true.
    \begin{equation*}
        \therefore \, \forall x \big( x = \alpha \implies \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
        \implies
        \varphi(\alpha, \beta_{1}, \dots, \beta_{m})
    \end{equation*}
    Since the domain set \( D \) (the interpretation) is arbitrary selected, it follows that,
    \begin{equation}
        \label{eq:all-x-equals-alpha-implies-phi-x-entails-phi-alpha}
        \forall x \big( x = \alpha \implies \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
        \models
        \varphi(\alpha, \beta_{1}, \dots, \beta_{m})
    \end{equation}
    Conversely, suppose that \( \varphi(\alpha, \beta_{1}, \dots, \beta_{m}) \) is true for an arbitrary interpretation.
    Let \( x_{0} \in D \) such that \( x_{0} = \alpha \).
    Since \( \varphi(\alpha, \beta_{1}, \dots, \beta_{m}) \) is true and \( x_{0} = \alpha \) ,
    it follows that \( \varphi(x_{0}, \beta_{1}, \dots, \beta_{m}) \) is also true.
    Thus, \( x_{0} = \alpha \implies \varphi(x_{0}, \beta_{1}, \dots, \beta_{m})\).
    Since \( x_{0} \) is arbitrary, Universal Generalization yields
    \( \forall x \big( x = \alpha \implies \varphi(x, \beta_{1}, \dots, \beta_{m}) \big) \).
    \begin{equation*}
        \therefore \, \varphi(\alpha, \beta_{1}, \dots, \beta_{m})
        \implies
        \forall x \big( x = \alpha \implies \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
    \end{equation*}
    Again, since the interpretation is arbitrary, it follows that,
    \begin{equation}
        \label{eq:phi-alpha-entails-all-x-equals-alpha-implies-phi-x}
        \varphi(\alpha, \beta_{1}, \dots, \beta_{m})
        \models
        \forall x \big( x = \alpha \implies \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
    \end{equation}
    Together, \eqref{eq:all-x-equals-alpha-implies-phi-x-entails-phi-alpha} and \eqref{eq:phi-alpha-entails-all-x-equals-alpha-implies-phi-x}
    establish the equivalence as needed.
\end{proof}

\begin{thm}
    \label{tautology:using-an-existentially-quantified-formula-to-express-the-fact-that-an-object-satisfies-a-formula}
    For any well-formed formula \( \varphi \) and free variables \( \alpha, \beta_{1}, \dots, \beta_{m} \),
    \begin{equation*}
        \exists x \big( x = \alpha \land \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
        \equiv
        \varphi(\alpha, \beta_{1}, \dots, \beta_{m})
    \end{equation*}
\end{thm}

\begin{proof}
    Let \( \varphi \) be a well-formed formula interpreted in the context of a non-empty domain, \( D \).
    Let \( \alpha, \beta_{1}, \dots, \beta_{m} \) represent objects in \( D \). Supposed that the well-formed
    formula \( \exists x \mleft( x = \alpha \land \varphi(x, \beta_{1}, \dots, \beta_{m}) \mright) \)
    is true. Then clearly,
    \begin{equation*}
        x_{0} = \alpha \land \varphi(x_{0}, \beta_{1}, \dots, \beta_{m})
    \end{equation*}
    for some \( x_{0} \in D \). Since \( x_{0} = \alpha \), it follows that
    \( \varphi(\alpha, \beta_{1}, \dots, \beta_{m}) \) must also be true.
    In other words,
    \begin{equation*}
        \exists x \big( x = \alpha \land \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
        \models
        \varphi(\alpha, \beta_{1}, \dots, \beta_{m})
    \end{equation*}
    Conversely, if \( \varphi(\alpha, \beta_{1}, \dots, \beta_{m}) \) is true, then clearly
    \( \alpha = \alpha \land \varphi(\alpha, \beta_{1}, \dots, \beta_{m}) \). Since
    \( \alpha \in D \), then we can use Existential Generalization to conclude,
    \begin{equation*}
        \exists x \big( x = \alpha \land \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
    \end{equation*}
    Thus,
    \begin{equation*}
        \varphi(\alpha, \beta_{1}, \dots, \beta_{m})
        \models
        \exists x \big( x = \alpha \land \varphi(x, \beta_{1}, \dots, \beta_{m}) \big)
    \end{equation*}
    Since the entailment has been established in both directions, the two formulas are indeed logically
    equivalent.
\end{proof}
